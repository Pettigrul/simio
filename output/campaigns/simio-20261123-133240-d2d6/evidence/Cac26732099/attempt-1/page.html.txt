<script>
(function() {
    var tickerTrack = document.querySelector('#tickerBar .ticker-track');
    if (!tickerTrack) return;

    function fmtChange(pct) {
        if (pct > 0) return '<span class="ticker-up">+' + pct.toFixed(2).replace('.',',') + '%</span>';
        if (pct < 0) return '<span class="ticker-down">' + pct.toFixed(2).replace('.',',') + '%</span>';
        return '<span class="ticker-flat">0,00%</span>';
    }
    function fmtPrice(v) { return 'R$ ' + v.toLocaleString('pt-BR',{minimumFractionDigits:2,maximumFractionDigits:2}); }

    function updateTicker() {
        fetch('/eds/ticker-api.php?_=' + Date.now())
            .then(function(r) { return r.json(); })
            .then(function(d) {
                var html = '';
                // IBOV
                if (d.ibov && d.ibov.price > 0) {
                    html += '<span class="ticker-item"><strong>IBOV</strong> ' + d.ibov.price.toLocaleString('pt-BR') + ' ' + fmtChange(d.ibov.change) + '</span>';
                }
                // FX
                if (d.fx) {
                    html += '<span class="ticker-item"><strong>USD</strong> ' + d.fx.usd.toFixed(2).replace('.',',') + ' ' + fmtChange(d.fx.usdChange || 0) + '</span>';
                    html += '<span class="ticker-item"><strong>EUR</strong> ' + (d.fx.eur || 0).toFixed(2).replace('.',',') + '</span>';
                    html += '<span class="ticker-item"><strong>BTC</strong> ' + Math.round(d.fx.btc || 0).toLocaleString('pt-BR') + '</span>';
                }
                // Indicators
                html += '<span class="ticker-item"><strong>SELIC</strong> ' + (d.selic || 0).toFixed(2).replace('.',',') + '%</span>';
                html += '<span class="ticker-item"><strong>IPCA 12m</strong> ' + (d.ipca || 0).toFixed(2).replace('.',',') + '%</span>';
                // Stocks
                (d.stocks || []).forEach(function(s) {
                    html += '<span class="ticker-item"><strong>' + s.symbol + '</strong> ' + fmtPrice(s.price) + ' ' + fmtChange(s.change) + '</span>';
                });
                // News
                (d.news || []).forEach(function(n) {
                    html += '<span class="ticker-item ticker-news"><i class="bi bi-dot"></i><a href="' + n.url + '" target="_blank" rel="noopener">' + n.title + '</a><small>' + n.time + '</small></span>';
                });
                // Duplicate for loop
                html += html;

                tickerTrack.innerHTML = html;
            })
            .catch(function() {});
    }

    // Update cotações a cada 30s (1 min em cache durante mercado aberto, 5 min fechado)
    setInterval(updateTicker, 30000);
})();
</script>

<script>
var laisaChat = {
    open: false,
    userName: '',
    userPhone: '',
    askedContact: false,
    toggle: function() {
        this.open = !this.open;
        var w = document.getElementById('laisaChatWindow');
        w.classList.toggle('open', this.open);
        if (this.open) {
            document.getElementById('laisaChatInput').focus();
        }
    },
    addMsg: function(text, sent) {
        var body = document.getElementById('laisaChatBody');
        var div = document.createElement('div');
        div.className = 'laisa-msg ' + (sent ? 'sent' : 'received');
        var now = new Date();
        var time = now.getHours().toString().padStart(2,'0') + ':' + now.getMinutes().toString().padStart(2,'0');
        div.innerHTML = '<div class="laisa-msg-bubble">' + text + '</div><div class="laisa-msg-time">' + time + '</div>';
        // Insert before quick buttons if they exist
        var btns = document.getElementById('laisaQuickBtns');
        if (btns) body.insertBefore(div, btns);
        else body.appendChild(div);
        body.scrollTop = body.scrollHeight;
    },
    quickReply: function(text) {
        this.addMsg(text, true);
        if (!this.askedContact) {
            this.askedContact = true;
            setTimeout(function() {
                laisaChat.addMsg('Ótima escolha! 👍 Para eu te ajudar melhor, me conta seu <strong>nome</strong> e <strong>WhatsApp</strong>:', false);
            }, 600);
        } else {
            this.openWhatsApp(text);
        }
    },
    send: function() {
        var input = document.getElementById('laisaChatInput');
        var text = input.value.trim();
        if (!text) return;
        this.addMsg(text, true);
        input.value = '';
        
        if (!this.askedContact) {
            this.askedContact = true;
            setTimeout(function() {
                laisaChat.addMsg('Para prosseguir, me informe seu <strong>nome</strong> e <strong>WhatsApp</strong>:', false);
            }, 600);
        } else if (text.length > 5) {
            this.openWhatsApp(text);
        }
    },
    openWhatsApp: function(msg) {
        var encoded = encodeURIComponent(msg);
        var url = 'https://wa.me/5511920712121?text=' + encoded;
        // Abre em popup, mantendo o site aberto
        var w = 450, h = 600;
        var left = screen.width - w - 20;
        var top = screen.height - h - 100;
        window.open(url, 'laisa-whatsapp', 'width='+w+',height='+h+',left='+left+',top='+top+',menubar=no,toolbar=no,location=no');
    }
};
</script>

    <script>
    (function() {
        // Captura contexto da página e simulação para enviar à Laisa
        function getSimContext() {
            var ctx = [];
            var title = document.title.replace(' - EDS Serviços Financeiros', '').replace(' - EDS Financeiros', '').trim();
            ctx.push('Página: ' + title);

            // Busca valor de simulação nos sliders
            var sliders = document.querySelectorAll('input[type="range"]');
            sliders.forEach(function(s) {
                var label = s.previousElementSibling || s.parentElement.querySelector('strong');
                if (label && s.value) {
                    var text = label.textContent.replace(/[^a-zA-ZÀ-ú\s]/g, '').trim();
                    if (text && text.length < 50) ctx.push(text + ': ' + s.value);
                }
            });

            // Busca selects relevantes
            var selects = document.querySelectorAll('select[id]');
            selects.forEach(function(sel) {
                if (sel.selectedIndex > 0) {
                    var opt = sel.options[sel.selectedIndex];
                    ctx.push(sel.id.replace(/([A-Z])/g,' $1').trim() + ': ' + opt.text);
                }
            });

            return ctx.length > 0 ? '\n\n📊 Simulação no site:\n' + ctx.join('\n') : '';
        }

        // Atualiza links WhatsApp com contexto
        function updateWhatsAppLinks() {
            var ctx = getSimContext();
            if (!ctx) return;

            var links = document.querySelectorAll('a[href*="wa.me/5511920712121"]');
            links.forEach(function(link) {
                // Evita duplicar contexto
                if (link.href.indexOf('📊 Simulação') === -1) {
                    var sep = link.href.indexOf('?text=') > -1 ? '%0A%0A' : '?text=';
                    link.href = link.href.split('?text=')[0] + '?text=' + encodeURIComponent(
                        decodeURIComponent(link.href.split('?text=')[1] || 'Olá Laisa') + ctx.replace(/\n/g, '%0A')
                    );
                }
            });
        }

        // Executa ao carregar
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', updateWhatsAppLinks);
        } else {
            updateWhatsAppLinks();
        }

        // Atualiza também ao interagir com sliders
        document.addEventListener('input', function(e) {
            if (e.target.type === 'range' || e.target.tagName === 'SELECT') {
                clearTimeout(window._simUpdateTimeout);
                window._simUpdateTimeout = setTimeout(updateWhatsAppLinks, 500);
            }
        });
    })();
    </script>
