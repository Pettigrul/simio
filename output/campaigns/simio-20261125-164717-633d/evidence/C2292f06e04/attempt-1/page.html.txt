                    $(".owl-item:first").addClass('active');

                    $(".owl-carousel .owl-item").removeClass('active');
                    $(".owl-item:first").addClass('active');

                    prox = $(".owl-item:first");
                }
                else
                    prox = $(".owl-item.active").next();

                pos = pos + 1;
                if (pos >= $(".owl-carousel .owl-item").length) pos = 0;

                $(".owl-carousel").data("owlCarousel").jumpTo(pos);


                $(".featured-image img").attr("src", prox.find('.fotoSlide').attr("src"));
                $(".owl-carousel .owl-item").removeClass('active');
                prox.addClass('active');
            });

            $(".prev-photo").click(function () {
                if ($(".owl-item").hasClass("active") == false)
