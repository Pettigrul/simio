    function getFavs() {
      try { return JSON.parse(localStorage.getItem(FAV_KEY) || '[]'); } catch(e) { return []; }
    }
    function saveFavs(list) {
      try { localStorage.setItem(FAV_KEY, JSON.stringify(list)); } catch(e) {}
    }
    function syncFavUI() {
      var favs = getFavs();
      var badge = document.querySelector('[data-favorites-count]');
      if (badge) {
        badge.textContent = String(favs.length);

<script>
(function() {
  var vehicleSelect = document.getElementById('fin-vehicle-select');
  var priceInput = document.getElementById('fin-price-input');
  var entryInput = document.getElementById('fin-entry-input');
  var entrySlider = document.getElementById('fin-entry-slider');
  var priceDisplay = document.getElementById('fin-price-display');
  var entryDisplay = document.getElementById('fin-entry-display');
  var financedEl = document.getElementById('fin-financed-amount');
  var summaryEntryEl = document.getElementById('fin-summary-entry');
  var installmentEl = document.getElementById('fin-monthly-installment');
  var formSummaryInput = document.getElementById('fin-form-summary');
  var termBtns = Array.prototype.slice.call(document.querySelectorAll('[data-term]'));

  var selectedTerm = 48;
  var monthlyRate = 0.0159; // 1.59% a.m.

  function fmtCurrency(n) {
    return 'R$ ' + Number(n || 0).toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  }

  function recalc(source) {
    var price = Math.max(5000, Number(priceInput.value || 0));
    var entry = Number(entryInput.value || 0);

    if (source === 'slider' || source === 'vehicle') {
      var pct = Number(entrySlider.value || 30);
      entry = Math.round((price * pct) / 100);
      entryInput.value = String(entry);
    } else if (source === 'entry') {
      if (entry >= price) {
        entry = Math.round(price * 0.8);
        entryInput.value = String(entry);
      }
      var calcPct = price > 0 ? Math.round((entry / price) * 100) : 0;
      entrySlider.value = String(Math.min(80, Math.max(0, calcPct)));
    }

    var financed = Math.max(0, price - entry);
    var r = monthlyRate;
    var n = selectedTerm;
    var pmt = 0;
    if (financed > 0 && n > 0) {
      pmt = financed * (r * Math.pow(1 + r, n)) / (Math.pow(1 + r, n) - 1);
    }

    var pctDisplay = price > 0 ? Math.round((entry / price) * 100) : 0;
    priceDisplay.textContent = fmtCurrency(price);
    entryDisplay.textContent = fmtCurrency(entry) + ' (' + pctDisplay + '%)';
    financedEl.textContent = fmtCurrency(financed);
    summaryEntryEl.textContent = fmtCurrency(entry);
    installmentEl.textContent = n + 'x de ' + fmtCurrency(pmt);

    var selectedOption = vehicleSelect.options[vehicleSelect.selectedIndex];
    var vTitle = (selectedOption && selectedOption.getAttribute('data-title')) ? selectedOption.getAttribute('data-title') : ('Veículo de ' + fmtCurrency(price));
    if (formSummaryInput) {
      formSummaryInput.value = vTitle + ' | Entrada: ' + fmtCurrency(entry) + ' + ' + n + 'x de ' + fmtCurrency(pmt);
    }
  }

  if (vehicleSelect) {
    vehicleSelect.addEventListener('change', function() {
      var opt = vehicleSelect.options[vehicleSelect.selectedIndex];
      var p = Number(opt ? opt.getAttribute('data-price') : 0);
      if (p > 0) {
        priceInput.value = String(p);
        recalc('vehicle');
      }
    });
  }
  if (priceInput) priceInput.addEventListener('input', function() { recalc('price'); });
  if (entryInput) entryInput.addEventListener('input', function() { recalc('entry'); });
  if (entrySlider) entrySlider.addEventListener('input', function() { recalc('slider'); });

  termBtns.forEach(function(btn) {
    btn.addEventListener('click', function() {
      selectedTerm = Number(btn.getAttribute('data-term') || 48);
      termBtns.forEach(function(b) { b.classList.toggle('is-active', b === btn); });
      recalc('term');
    });
  });

  recalc('slider');

  // Submit Financing Lead Form -> POST /api/public/leads
  var form = document.getElementById('omega-financing-form');
  var feedback = document.getElementById('fin-lead-feedback');
  var submitBtn = document.getElementById('fin-submit-btn');

  if (form) {
    form.addEventListener('submit', async function(e) {
      e.preventDefault();
      var nameVal = (form.name.value || '').trim();
      var phoneVal = (form.phone.value || '').trim();
      if (!nameVal || !phoneVal) {
        feedback.className = 'omega-form-feedback is-visible is-error';
        feedback.textContent = 'Informe seu Nome e Telefone/WhatsApp para enviarmos sua simulação.';
        return;
      }

      submitBtn.disabled = true;
      var origHtml = submitBtn.innerHTML;
      submitBtn.innerHTML = '<span>Enviando simulação...</span>';

      var opt = vehicleSelect.options[vehicleSelect.selectedIndex];
      var vId = vehicleSelect.value || null;
      var vTitle = (opt && opt.getAttribute('data-title')) || '';

      var payload = {
        name: nameVal,
        phone: phoneVal,
        email: (form.email.value || '').trim(),
        vehicle_id: vId,
        vehicle_name: vTitle,
        value: Number(priceInput.value || 0),
        interest_type: 'financing',
        origin: 'Simulador de Financiamento',
        message: [
          form.simulation_summary.value,
          form.cpf.value ? ('CPF: ' + form.cpf.value) : '',
          'Troca: ' + (form.has_trade_in.value || 'Não')
        ].filter(Boolean).join(' | ')
      };

      try {
        var res = await fetch('/api/public/leads', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
        var data = await res.json();
        if (res.ok && data.success !== false) {
          feedback.className = 'omega-form-feedback is-visible is-success';
          feedback.textContent = data.message || 'Simulação enviada com sucesso! Nosso especialista em financiamento entrará em contato via WhatsApp.';
          form.name.value = '';
          form.phone.value = '';
          form.cpf.value = '';
          form.email.value = '';
        } else {
          throw new Error(data.message || data.error || 'Erro ao enviar simulação.');
        }
      } catch (err) {
        feedback.className = 'omega-form-feedback is-visible is-error';
        feedback.textContent = err.message || 'Não foi possível enviar no momento. Chame no WhatsApp.';
      } finally {
        submitBtn.disabled = false;
        submitBtn.innerHTML = origHtml;
      }
    });
  }
})();
</script>
