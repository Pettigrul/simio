        // Rolagem automática: um timer só, e só enquanto os depoimentos estão na tela.
        function rodar() {
            if (timer || !visivel || isDown) return;
            timer = setInterval(function () {
                carousel.scrollLeft += 1;
                if (carousel.scrollLeft >= carousel.scrollWidth / 2) carousel.scrollLeft = 0;
            }, 20);
        }
        function parar() { clearInterval(timer); timer = null; }

        if ('IntersectionObserver' in window) {
            new IntersectionObserver(function (entradas) {
                visivel = entradas[0].isIntersecting;
                if (visivel) rodar(); else parar();
            }).observe(carousel);
        } else {
            visivel = true;
            rodar();
        }

        carousel.addEventListener('mousedown', function (e) {
            isDown = true;
            startX = e.pageX - carousel.offsetLeft;
            scrollLeft = carousel.scrollLeft;
            parar();
        });
        carousel.addEventListener('mouseleave', function () { isDown = false; rodar(); });
        carousel.addEventListener('mouseup', function () { isDown = false; rodar(); });
        carousel.addEventListener('mousemove', function (e) {
            if (!isDown) return;

            startX = e.touches[0].pageX - carousel.offsetLeft;

            parar();

        carousel.addEventListener('touchend', rodar);
