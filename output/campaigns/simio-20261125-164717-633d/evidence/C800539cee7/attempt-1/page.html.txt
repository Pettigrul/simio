    function getFavs() {
      try { return JSON.parse(localStorage.getItem(FAV_KEY) || '[]'); } catch(e) { return []; }
    }
    function saveFavs(list) {
      try { localStorage.setItem(FAV_KEY, JSON.stringify(list)); } catch(e) {}

    function syncFavUI() {
      var favs = getFavs();
      var badge = document.querySelector('[data-favorites-count]');

(function() {
  // 1. Hero Banner Carousel
  var slides = Array.prototype.slice.call(document.querySelectorAll('[data-hero-slide]'));
  var dots = Array.prototype.slice.call(document.querySelectorAll('[data-hero-dot]'));
  var currHero = 0;
  function showHero(idx) {
    if (!slides.length) return;
    currHero = (idx + slides.length) % slides.length;
    slides.forEach(function(s, i) { s.classList.toggle('is-active', i === currHero); });
    dots.forEach(function(d, i) { d.classList.toggle('is-active', i === currHero); });
  }
  dots.forEach(function(d) {
    d.addEventListener('click', function() {
      showHero(Number(d.getAttribute('data-hero-dot') || 0));
    });
  });
  if (slides.length > 1) {
    setInterval(function() { showHero(currHero + 1); }, 6000);
  }

  // 2. Testimonials Carousel
  var tSlides = Array.prototype.slice.call(document.querySelectorAll('[data-testimonial-slide]'));
  var tDots = Array.prototype.slice.call(document.querySelectorAll('[data-testimonial-dot]'));
  var tPrev = document.querySelector('[data-testimonial-prev]');
  var tNext = document.querySelector('[data-testimonial-next]');
  var currT = 0;
  function showTestimonial(idx) {
    if (!tSlides.length) return;
    currT = (idx + tSlides.length) % tSlides.length;
    tSlides.forEach(function(s, i) {
      var active = i === currT;
      s.classList.toggle('active', active);
      s.hidden = !active;
    });
    tDots.forEach(function(d, i) { d.classList.toggle('active', i === currT); });
  }
  if (tPrev) tPrev.addEventListener('click', function() { showTestimonial(currT - 1); });
  if (tNext) tNext.addEventListener('click', function() { showTestimonial(currT + 1); });
  tDots.forEach(function(d) {
    d.addEventListener('click', function() {
      showTestimonial(Number(d.getAttribute('data-testimonial-dot') || 0));
    });
  });
})();
