jQuery(document).ready(function($) {

    let numeroSelecionado = null;

    function buscarProximoAtendente() {
        $.ajax({
            url: 'https://financiautocredito.com/wp-admin/admin-ajax.php',
            type: 'POST',
            async: false,
            data: {
                action: 'whatsapp_leads_proximo'
            },
            success: function(response) {
                if (response.success) {
                    numeroSelecionado = response.data;

                    const textoAtendente =
                        numeroSelecionado.nome + ' - ' + numeroSelecionado.numero;

                    $('input[name="form_fields[atendente_whatsapp]"]')
                        .val(textoAtendente);
                }
            }
        });
    }

    function preencherAtendente() {
        if (!numeroSelecionado) {
            buscarProximoAtendente();
        }

        if (numeroSelecionado) {
            const textoAtendente =
                numeroSelecionado.nome + ' - ' + numeroSelecionado.numero;

            $('input[name="form_fields[atendente_whatsapp]"]')
                .val(textoAtendente);
        }
    }

    $(document).on('submit', 'form.elementor-form', function() {
        preencherAtendente();
    });

    $('input, textarea, select').on(
        'input change focus',
        preencherAtendente
    );

    $(document).on('submit_success', function(event) {

        const form = $(event.target)
            .closest('form.elementor-form');

        if (!numeroSelecionado) {
            buscarProximoAtendente();
        }

        if (!numeroSelecionado) {
            return;
        }

        let mensagemTexto =
            '*NOVA SOLICITAÇÃO DE FINANCIAMENTO*\n\n';

        form.find('input, textarea, select').each(function() {

            let campo = $(this);
            let valor = campo.val();
            let nomeCampo = campo.attr('name');

            if (nomeCampo === 'form_fields[atendente_whatsapp]') {
                return;
            }

            let label = campo
                .closest('.elementor-field-group')
                .find('label')
                .text()
                .trim();

            if (!label) {
                label = campo.attr('placeholder');
            }

            if (valor && label) {
                mensagemTexto += '*' + label + ':* ' + valor + '\n';
            }
        });

        mensagemTexto +=
            '\n*Lead aguardando atendimento.*';

        const mensagem =
            encodeURIComponent(mensagemTexto);

        const url =
            'https://wa.me/' +
            numeroSelecionado.numero +
            '?text=' +
            mensagem;

        setTimeout(function() {
            window.location.href = url;
        }, 800);

    });

});
