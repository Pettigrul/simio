    var expanded = $(this).attr('aria-expanded') === 'true';
    $(this).attr('aria-expanded', !expanded);
  });

  $('.dropdown-toggle').on('keydown', function (e) {

    var menu = $(this).next('.dropdown-menu');
    var items = menu.find('a');

    switch (e.key) {

      case "ArrowDown":
        e.preventDefault();
        menu.addClass('open');
        items.first().focus();
        break;

      case "ArrowUp":
        e.preventDefault();
        menu.addClass('open');
        items.last().focus();
        break;

      case "Escape":
        menu.removeClass('open');
        $(this).attr('aria-expanded', false).focus();
        break;

    }

  });

  $('.dropdown-menu a').on('keydown', function (e) {

    var items = $(this).closest('.dropdown-menu').find('a');
    var index = items.index(this);

    switch (e.key) {

      case "ArrowDown":
        e.preventDefault();
        items.eq((index + 1) % items.length).focus();
        break;

      case "ArrowUp":
        e.preventDefault();
        items.eq((index - 1 + items.length) % items.length).focus();
        break;

      case "Escape":
        e.preventDefault();
        $(this).closest('.dropdown').find('.dropdown-toggle').focus();
        break;

    }

  });
