<script>
(function () {

  function abrirModalInscricao(e) {
    const link = e.target.closest('a[href="#inscricao"]');
    if (!link) return;

    e.preventDefault();

    const modalEl = document.getElementById('modal_ticket_registration');

    if (!modalEl) {
      console.warn('Modal não encontrado');
      return;
    }

    if (window.bootstrap && window.bootstrap.Modal) {
      const modal = window.bootstrap.Modal.getOrCreateInstance(modalEl);
      modal.show();
      return;
    }

    modalEl.classList.add('show');
    modalEl.style.display = 'block';
    modalEl.removeAttribute('aria-hidden');
    modalEl.setAttribute('aria-modal', 'true');

    let backdrop = document.createElement('div');
    backdrop.className = 'modal-backdrop fade show';
    document.body.appendChild(backdrop);

    document.body.classList.add('modal-open');

    function fecharModal() {
      modalEl.classList.remove('show');
      modalEl.style.display = 'none';
      modalEl.setAttribute('aria-hidden', 'true');
      modalEl.removeAttribute('aria-modal');

      if (backdrop) backdrop.remove();
      document.body.classList.remove('modal-open');

      modalEl.removeEventListener('click', handleClick);
      document.removeEventListener('keydown', handleEsc);
    }

    function handleClick(e) {
      if (
        e.target.closest('[data-bs-dismiss="modal"]') ||
        e.target.classList.contains('modal')
      ) {
        fecharModal();
      }
    }

    function handleEsc(e) {
      if (e.key === 'Escape') {
        fecharModal();
      }
    }

    modalEl.addEventListener('click', handleClick);
    document.addEventListener('keydown', handleEsc);
  }

  document.addEventListener('click', abrirModalInscricao);

})();
</script>



<script>
/* =============================================================
   META PIXEL - GZ Negócios & Investimentos
   Versão revisada para Odoo 19 / OWL SPA

   Objetivos:
   - Evitar fbq init duplicado
   - Evitar PageView duplicado
   - Evitar AddPaymentInfo duplicado
   - Evitar Purchase duplicado
   - Monitorar páginas principais do site
   - Monitorar funil do módulo de eventos / shop do Odoo
   - Monitorar Lead em formulários comuns
   - Monitorar Lead + Purchase em /shop/confirmation

   Observação:
   - Este script assume que NÃO existe outro snippet do Meta
     carregado em paralelo no site.
   ============================================================ */

(function () {
  "use strict";

  /* ============================================================
     1) CONFIGURAÇÃO BÁSICA
     ============================================================ */
  var PIXEL_ID = '1863585014481219';
  var DEBUG = true;

  function log() {
    if (!DEBUG) return;
    try {
      console.log.apply(console, ['[GZ Pixel]'].concat([].slice.call(arguments)));
    } catch (e) {}
  }

  /* ============================================================
     2) CARREGAMENTO E INIT DO PIXEL
     - Blindado para não inicializar múltiplas vezes
     ============================================================ */
  if (!window.fbq) {
    !function(f,b,e,v,n,t,s) {
      if (f.fbq) return;
      n = f.fbq = function() {
        n.callMethod ? n.callMethod.apply(n, arguments) : n.queue.push(arguments);
      };
      if (!f._fbq) f._fbq = n;
      n.push = n;
      n.loaded = true;
      n.version = '2.0';
      n.queue = [];
      t = b.createElement(e);
      t.async = true;
      t.src = v;
      s = b.getElementsByTagName(e)[0];
      s.parentNode.insertBefore(t, s);
    }(window, document, 'script', 'https://connect.facebook.net/en_US/fbevents.js');
  }

  if (!window._gzFbPixelInitialized) {
    fbq('init', PIXEL_ID);
    window._gzFbPixelInitialized = true;
    log('Pixel inicializado:', PIXEL_ID);
  } else {
    log('Pixel já havia sido inicializado anteriormente.');
  }

  /* ============================================================
     3) ESTADO GLOBAL
     - Mantém controle por rota para evitar duplicidade em SPA
     ============================================================ */
  if (!window._gzPixelState) {
    window._gzPixelState = {
      lastPath: null,
      tracked: {},
      timers: {}
    };
  }

  var state = window._gzPixelState;

  function routeKey(path, eventName) {
    return eventName + '::' + path;
  }

  function wasTracked(path, eventName) {
    return !!state.tracked[routeKey(path, eventName)];
  }

  function markTracked(path, eventName) {
    state.tracked[routeKey(path, eventName)] = true;
  }

  function clearRouteTimers() {
    Object.keys(state.timers).forEach(function (k) {
      try { clearTimeout(state.timers[k]); } catch (e) {}
    });
    state.timers = {};
  }

  /* ============================================================
     4) HELPERS
     ============================================================ */

  // Gera um identificador simples baseado na URL
  function getContentId() {
    var path = window.location.pathname || '/';
    var clean = path.replace(/^\/|\/$/g, '').replace(/\//g, '_');
    return clean || 'home';
  }

  // Extrai valor monetário do DOM
  // Busca em múltiplos seletores, pois o Odoo pode variar
  function getOrderValue() {
    var selectors = [
      '.oe_currency_value',
      '[data-field="amount_total"]',
      '.monetary_field',
      '#order_total .oe_currency_value',
      '.text-end .oe_currency_value',
      '.checkout_total .oe_currency_value'
    ];

    for (var i = 0; i < selectors.length; i++) {
      var el = document.querySelector(selectors[i]);

      if (el && el.textContent) {
        var raw = (el.textContent || '')
          .replace(/\s+/g, '')
          .replace(/\./g, '')      // remove milhares BR
          .replace(',', '.')       // decimal BR -> JS
          .replace(/[^\d.]/g, '');

        var val = parseFloat(raw);

        if (!isNaN(val) && val > 0) {
          return val;
        }
      }
    }

    return null;
  }

  // Wrapper para track com proteção de duplicidade por rota
  function trackOnce(path, eventName, params) {
    if (wasTracked(path, eventName)) {
      log('Ignorado duplicado:', eventName, 'em', path);
      return;
    }

    fbq('track', eventName, params || {});
    markTracked(path, eventName);
    log('Evento enviado:', eventName, 'em', path, params || {});
  }

  // Wrapper para custom events
  function trackCustomOnce(path, eventName, params) {
    if (wasTracked(path, eventName)) {
      log('Ignorado duplicado custom:', eventName, 'em', path);
      return;
    }

    fbq('trackCustom', eventName, params || {});
    markTracked(path, eventName);
    log('Evento custom enviado:', eventName, 'em', path, params || {});
  }

  /* ============================================================
     5) REGRAS DE TRACKING POR PÁGINA
     ============================================================ */
  function trackCurrentPage() {
    var path = window.location.pathname || '/';
    var contentId = getContentId();

    // Se a rota não mudou, não faz nada
    if (state.lastPath === path) {
      return;
    }

    // Mudou rota
    state.lastPath = path;
    clearRouteTimers();

    log('Nova rota detectada:', path);

    /* ------------------------------------------------------------
       5.1 PageView
       ------------------------------------------------------------ */
    trackOnce(path, 'PageView');

    /* ------------------------------------------------------------
       5.2 Páginas institucionais
       ------------------------------------------------------------ */
    if (path === '/consultoria360-investimentos') {
      trackOnce(path, 'ViewContent', {
        content_name: document.title || 'Consultoria 360 Investimentos',
        content_ids: [contentId],
        content_category: 'Investimentos',
        content_type: 'page'
      });
    }

    if (path === '/consultoria360-empresas') {
      trackOnce(path, 'ViewContent', {
        content_name: document.title || 'Consultoria 360 Empresas',
        content_ids: [contentId],
        content_category: 'Empresas',
        content_type: 'page'
      });
    }

    if (path === '/treinamentos') {
      trackOnce(path, 'ViewContent', {
        content_name: document.title || 'Treinamentos',
        content_ids: [contentId],
        content_category: 'Treinamentos',
        content_type: 'page'
      });
    }

    /* ------------------------------------------------------------
       5.3 Página de registro de evento
       Exemplo: /event/nomedoevento/register
       ------------------------------------------------------------ */
    if (path.indexOf('/event/') === 0 && path.indexOf('/register') > -1) {
      trackOnce(path, 'ViewContent', {
        content_name: document.title || 'Registro Evento',
        content_ids: [contentId],
        content_category: 'Evento',
        content_type: 'product'
      });
    }

    /* ------------------------------------------------------------
       5.4 Etapa de checkout
       O seu funil usa:
       - /shop/address
       - /shop/checkout
       ------------------------------------------------------------ */
    if (path === '/shop/address' || path === '/shop/checkout') {
      trackOnce(path, 'InitiateCheckout', {
        content_name: document.title || 'Checkout',
        content_ids: [contentId],
        content_category: 'Evento/Checkout',
        content_type: 'product'
      });
    }

    /* ------------------------------------------------------------
       5.5 Etapa de pagamento
       Dispara apenas uma vez por rota, com pequeno delay
       para o Odoo renderizar os valores no DOM
       ------------------------------------------------------------ */
    if (path === '/shop/payment') {
      state.timers.addPaymentInfo = setTimeout(function () {
        var orderValue = getOrderValue();
        var params = {
          currency: 'BRL',
          content_ids: [contentId],
          content_type: 'product'
        };

        if (orderValue && orderValue > 0) {
          params.value = orderValue;
        }

        trackOnce(path, 'AddPaymentInfo', params);
      }, 1200);
    }

    /* ------------------------------------------------------------
       5.6 Confirmação da compra / confirmação da inscrição
       Aqui você quer:
       - Lead (porque entra no CRM)
       - Purchase (porque houve a confirmação da venda)
       ------------------------------------------------------------ */
    if (path === '/shop/confirmation') {
      state.timers.confirmation = setTimeout(function () {
        var orderValue = getOrderValue();

        // LEAD
        trackOnce(path, 'Lead', {
          content_name: document.title || 'Confirmação de inscrição',
          content_category: 'Lead Evento',
          content_ids: [contentId]
        });

        // PURCHASE
        // O Meta exige value em Purchase.
        // Se não acharmos valor real no DOM, usamos fallback 1
        // para evitar o erro de missing value.
        var purchaseParams = {
          value: (orderValue && orderValue > 0) ? orderValue : 1,
          currency: 'BRL',
          content_ids: [contentId],
          content_type: 'product'
        };

        trackOnce(path, 'Purchase', purchaseParams);
      }, 1500);
    }

    /* ------------------------------------------------------------
       5.7 Página de status intermediária do pagamento
       Ex.: /payment/status
       Aqui normalmente só deixamos o PageView.
       ------------------------------------------------------------ */
  }

  /* ============================================================
     6) TRACKING DE LEAD EM FORMULÁRIOS COMUNS DO SITE
     - Qualquer envio de website_form / dataset_call_kw
     ============================================================ */
  if (!window._gzFormLeadListenerAdded) {
    window._gzFormLeadListenerAdded = true;

    document.addEventListener('submit', function (e) {
      var form = e.target;
      if (!form) return;

      var action = form.action || '';

      if (
        action.indexOf('/website_form/') > -1 ||
        action.indexOf('/web/dataset/call_kw') > -1
      ) {
        fbq('track', 'Lead', {
          content_name: document.title || 'Formulario Site',
          content_category: 'Lead Site',
          content_ids: [getContentId()]
        });

        log('Lead enviado por formulário comum.');
      }
    });

    log('Listener de formulários registrado.');
  }

  /* ============================================================
     7) CLIQUE EM WHATSAPP = LEAD
     ============================================================ */
  if (!window._gzWhatsappLeadListenerAdded) {
    window._gzWhatsappLeadListenerAdded = true;

    document.addEventListener('click', function (e) {
      var el = e.target && e.target.closest
        ? e.target.closest('a[href*="wa.me"], a[href*="whatsapp"]')
        : null;

      if (el) {
        fbq('track', 'Lead', {
          content_name: 'WhatsApp CTA',
          content_category: 'Lead WhatsApp',
          content_ids: [getContentId()]
        });

        log('Lead enviado por clique em WhatsApp.');
      }
    });

    log('Listener de WhatsApp registrado.');
  }

  /* ============================================================
     8) CLIQUE EM TELEFONE / API WHATSAPP = CONTACT
     ============================================================ */
  if (!window._gzContactListenerAdded) {
    window._gzContactListenerAdded = true;

    document.addEventListener('click', function (e) {
      var el = e.target && e.target.closest
        ? e.target.closest('a[href^="tel:"], a[href^="https://api.whatsapp"]')
        : null;

      if (el) {
        fbq('track', 'Contact', {
          content_name: el.href.indexOf('tel:') === 0 ? 'Telefone' : 'WhatsApp'
        });

        log('Contact enviado.');
      }
    });

    log('Listener de Contact registrado.');
  }

  /* ============================================================
     9) CLIQUES DE AGENDAMENTO = SCHEDULE
     ============================================================ */
  if (!window._gzScheduleListenerAdded) {
    window._gzScheduleListenerAdded = true;

    document.addEventListener('click', function (e) {
      var el = e.target && e.target.closest
        ? e.target.closest(
            'a[href*="calendly"], a[href*="agenda"], button[id*="agendar"], .btn[href*="demo"]'
          )
        : null;

      if (el) {
        fbq('track', 'Schedule', {
          content_name: 'Agendamento'
        });

        log('Schedule enviado.');
      }
    });

    log('Listener de Schedule registrado.');
  }

  /* ============================================================
     10) EVENTO CUSTOMIZADO DE ABANDONO DE PAGAMENTO
     - Dispara quando a pessoa sai da rota /shop/payment
       sem ir para /shop/confirmation
     ============================================================ */
  function fireAbandonedPaymentIfNeeded(oldPath, newPath) {
    if (oldPath === '/shop/payment' && newPath !== '/shop/confirmation') {
      var params = {
        currency: 'BRL',
        content_ids: ['shop_payment']
      };

      var orderValue = getOrderValue();
      if (orderValue && orderValue > 0) {
        params.value = orderValue;
      }

      // Não usa once por rota nova; usa chave do oldPath
      if (!wasTracked(oldPath, 'AbandonedPayment')) {
        fbq('trackCustom', 'AbandonedPayment', params);
        markTracked(oldPath, 'AbandonedPayment');
        log('AbandonedPayment enviado.', params);
      }
    }
  }

  /* ============================================================
     11) DETECÇÃO DE MUDANÇA DE ROTA EM SPA
     ============================================================ */
  if (!window._gzSpaHooksInstalled) {
    window._gzSpaHooksInstalled = true;

    var originalPushState = history.pushState;
    var originalReplaceState = history.replaceState;

    history.pushState = function () {
      var oldPath = window.location.pathname || '/';
      var result = originalPushState.apply(this, arguments);

      setTimeout(function () {
        var newPath = window.location.pathname || '/';
        fireAbandonedPaymentIfNeeded(oldPath, newPath);
        trackCurrentPage();
      }, 50);

      return result;
    };

    history.replaceState = function () {
      var oldPath = window.location.pathname || '/';
      var result = originalReplaceState.apply(this, arguments);

      setTimeout(function () {
        var newPath = window.location.pathname || '/';
        fireAbandonedPaymentIfNeeded(oldPath, newPath);
        trackCurrentPage();
      }, 50);

      return result;
    };

    window.addEventListener('popstate', function () {
      trackCurrentPage();
    });

    log('Hooks SPA instalados.');
  }

  /* ============================================================
     12) CARGA INICIAL
     ============================================================ */
  trackCurrentPage();
