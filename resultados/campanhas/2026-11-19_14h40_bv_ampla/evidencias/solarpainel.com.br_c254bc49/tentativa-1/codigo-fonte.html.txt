function renderMessage(params) {
    /*
    // params comes from server
    // The top format is for rendering the justification html
    // and the bottom format is for rendering the block html.
    @params = {
    msg: (string),
    justify: (htmlString)
    } or
    @params = {
    rawmsg: (htmlString)
    }
     */
    var notify_modal_container = document.createElement('div');
    if (document.body) {
        document.body.appendChild(notify_modal_container);
    }
    // Display justification pages
    doc = window.document;
    if (params.msg) {
        doc.title = params.msg;
    }

    var language = "en";
    if (language.indexOf("NS_LANGUAGE") > -1) {
        // no substitution done, set to default
        ns_debug("language is set to default language en");
        doc.documentElement.lang = 'en';
    } else {
        ns_debug("language is set to:" + language);
        doc.documentElement.lang = language;
    }

    notify_css = doc.createElement('style');
    notify_css.innerHTML = css;
    doc.getElementsByTagName("head")[0].appendChild(notify_css);

    doc.body.innerHTML = params.justify;
    var btnGroupLive = doc.querySelector('.ns-btn-group');
    if (btnGroupLive) { btnGroupLive.setAttribute('aria-live', 'polite'); }
    doc.querySelector('.msg').innerHTML = params.msg;

    formData = {};
    var justifyForm = doc.querySelector('.ns-template-inner-wrap form');
    var descriptionField = doc.querySelector('#ns-justify-description');
    var descriptionFieldBox = doc.querySelector('#ns-justify-description-box');
    var justifySubmitBtns = justifyForm.querySelectorAll('.ns-submit');
    var justifyRadio = justifyForm.querySelector('#justify-usage-radio');
    var falsePositiveRadio = justifyForm.querySelector('#false-positive-radio');

    var justification_req = 1;
    _ns_justify_message_ = 1;
    _ns_url_ = justifyForm.getAttribute('action');

    var justifyFields = justifyForm.querySelectorAll('[name]');
    for (var i=0; i<justifyFields.length; i++) {
        var field = justifyFields[i];
        var key = field.getAttribute('name');
        if(key == 'msuid') {
            _ns_msuid_ = field.value;
            break;
        }
    }

    function disableSubmitBtns() {
        if (justifySubmitBtns.length == 2) {
            justifySubmitBtns[1].setAttribute('disabled', 'disabled');
        } else if (justifySubmitBtns.length == 1) {
            justifySubmitBtns[0].setAttribute('disabled', 'disabled');
        }
    }

    function enableSubmitBtns() {
        if (justifySubmitBtns.length == 2) {
            justifySubmitBtns[1].removeAttribute('disabled');
        } else if (justifySubmitBtns.length == 1) {
            justifySubmitBtns[0].removeAttribute('disabled');
        }
    }

    function validateForm() {
        if(params.force_justify == "false")
            return;

        if (!descriptionFieldBox || !justifySubmitBtns) {
            return;
        }
        if (justification_req == 1 && !descriptionFieldBox.value.length) {
            disableSubmitBtns();
        } else {
            if (justification_req == 1 &&
                descriptionFieldBox.value.length &&
                !descriptionFieldBox.value.match(/([^\u0000-\u007F]|\w)+/)) {
                disableSubmitBtns();
            } else {
                enableSubmitBtns();
            }
        }
    }

    function submitModal() {
        var justifyFields = justifyForm.querySelectorAll('[name]');
        var url = justifyForm.getAttribute('action');

        var cancel_button = 0;
        if(this.value == 'cancel')
            cancel_button = 1;

        var xhr = new XMLHttpRequest();
        xhr.open("POST", url, true);
        xhr.setRequestHeader('Content-Type', 'application/json');
        xhr.onreadystatechange = function () {
            if (xhr.readyState == 2) {
                if(!cancel_button) {
                    document.location.reload();
                } else {
                    var referrer = "/";
                    if (referrer != "/")
                        window.location = "/";
                }
            }
        }

        formData["justification_type"] = "justification";
        formData["_uj_category_id"] = "511";
	    var appCategory = "";
	    if (appCategory.indexOf("NS_APP_CATEGORY") === -1) {
	        formData["_uj_app_category"] = appCategory;
	    }
        formData["_uj_other_categories"] = "Business;DLP-PII-Geral;DLP-PFI-Geral;DLP-Arquivos-Maiores-128MB;DLP-Arquivos-Protegidos-Senha;DLP-PCI-Geral;DLP-Dados-Analiticos;DLP-Apontamentos;DLP-Infra-Geral;DLP-Base-Contribuicao-Sindical;DLP-Informacoes-PEP;Bloqueio-Upload;DLP-Source-Code;Acesso-VIPs;DLP-Label-Publica-Info-Sensivel;DLP-Altamente-Confidencial-Senha;DLP-Normativas;DLP-Dados-Funcionarios;DLP-Info-Confidenciais-Gerais;DLP-Label-Interna;TodasCategorias;DLP-Transcricao-Reunioes;DLP-Base64;DLP-Dados-Juridicos;Todas_Categorias_Protection;DLP-Informacoes-Seguranca;DLP-Apresentacao-Maior-3MB";
        formData["web_category"] = "511";
        formData["policy"] = "[Firewall APP] Block Destination IP-cRM46L";
        formData["_uj_dest_ip"] = "192.99.35.71";
        formData["uj_response_file_name"] = "";
        formData["app"] = "solarpainel";
        formData["appsessiondata"] = "eyJhcHAiOiJzb2xhcnBhaW5lbCIsImNvbnRlbnRWZXJzaW9uIjoxNzk0OTc2NTI0fQ==";
        formData["from_user"] = "";
        formData["instance_id"] = "";
        formData["suppress_interval"] = "1800";
        formData["suppress_data"] = "";
        formData["no_alert"] = "false";
        for (var i=0; i<justifyFields.length; i++) {
            var field = justifyFields[i];
            var key = field.getAttribute('name');
            var value = field.value;
            if ((key == "justification_type") && (!field.checked)) {
                continue;
            }
            formData[key] = value;
        }

        if (this.value) {
            formData.submitId = this.value;
        } else {
            formData.submitId = 'cancel';
        }
        xhr.send(JSON.stringify(formData));

        // Modified logic for ok (cancel) button
        if(cancel_button) {
            handleSubmissionComplete();
        }
    }

    function handleSubmissionComplete() {
        // 1. Disable all submit buttons and make them grey
        for (var j=0; j<justifySubmitBtns.length; j++) {
            justifySubmitBtns[j].setAttribute('disabled', 'disabled');
            justifySubmitBtns[j].style.background = '#EEEEEE';
            justifySubmitBtns[j].style.borderColor = '#E3E3E3';
            justifySubmitBtns[j].style.color = '#ccc';
        }

        // 2. Disable the Text Box and make it grey
        var textBox = doc.querySelector('#ns-justify-description-box');
        if(textBox) {
            textBox.setAttribute('disabled', 'disabled');
            textBox.style.backgroundColor = '#EEEEEE';
            textBox.style.borderColor = '#E3E3E3';
            textBox.style.color = '#ccc';
        }

        // 3. Disable any other submit buttons found
        var proceedBtn = doc.querySelector('#ns-justify-submit');
        if(proceedBtn) {
            proceedBtn.setAttribute('disabled', 'disabled');
        }

        // 4. Disable radio buttons
        if(justifyRadio) {
            justifyRadio.setAttribute('disabled', 'disabled');
        }
        if(falsePositiveRadio) {
            falsePositiveRadio.setAttribute('disabled', 'disabled');
        }

        // 5. Show the message only if referrer is '/'
        var referrer = "/";
        if (referrer == "/") {
            var btnGroup = doc.querySelector('.ns-btn-group');
            if (btnGroup) {
                // Check if message already exists to prevent duplicate insertion
                if (!doc.querySelector('.ns-success-msg')) {
                    var msgSpan = doc.createElement('div');
                    msgSpan.className = 'ns-success-msg';

                    var currentLang = language.indexOf("NS_LANGUAGE") > -1 ? 'en' : language;
                    var langData = ackMessages[currentLang] || ackMessages['en'];

                    // Visibility detection for UI elements to determine appropriate message
                    // Check if description box is present and visible in DOM:
                    // - Element exists (descriptionFieldBox)
                    // - Element is rendered in layout (offsetParent !== null)
                    // - Element is not explicitly hidden (computed display !== 'none')
                    var hasDescriptionBox = Boolean(
                        descriptionFieldBox &&
                        descriptionFieldBox.offsetParent !== null &&
                        window.getComputedStyle(descriptionFieldBox).display !== 'none'
                    );
                    
                    // Check if either justification radio button is present and visible in DOM:
                    // - justifyRadio exists and is rendered (offsetParent !== null)
                    // - OR falsePositiveRadio exists and is rendered (offsetParent !== null)
                    var hasRadioButtons = Boolean((justifyRadio && justifyRadio.offsetParent !== null) ||
                                                   (falsePositiveRadio && falsePositiveRadio.offsetParent !== null));
                    
                    var message = (hasDescriptionBox || hasRadioButtons) ? langData.justification_message : langData.block_message;

                    msgSpan.innerHTML = message;
                    btnGroup.appendChild(msgSpan);
                }
            }
        }
    }

    validateForm();
    // add events
    if (descriptionField) {
        descriptionField.addEventListener('input', validateForm);
    }

    if(justifyRadio) {
        justifyRadio.addEventListener('click', function() {
            descriptionField.style.display = 'block';
            justification_req = 1;
            validateForm();
        });
    }

    if(falsePositiveRadio) {
        falsePositiveRadio.addEventListener('click', function() {
            descriptionField.style.display = 'none';
            descriptionField.value = '';
            justification_req = 0;
            validateForm();
        });
    }

    for (var i=0; i<justifySubmitBtns.length; i++) {
        justifySubmitBtns[i].addEventListener('click', submitModal);
    }

    if (doc.querySelector('#ns-justify-description-box')) {
        var keydownfunc = function(e) {
            e.stopPropagation();
        }
        doc.querySelector('#ns-justify-description-box').addEventListener("keydown", keydownfunc, false);

        var keypressfunc = function(e) {
            e.stopPropagation();
        }
        doc.querySelector('#ns-justify-description-box').addEventListener("keypress", keypressfunc, false);
    }
}
