(function () {
  var form   = document.getElementById('contactForm');
  var status = document.getElementById('contactStatus');
  var btn    = document.getElementById('contactSubmit');
  if (!form || !status || !btn) return;

  function setStatus(ok, msg) {
    status.hidden = false;
    status.className = 'contact-status ' + (ok ? 'is-ok' : 'is-error');
    status.textContent = msg;
  }

  form.addEventListener('submit', function (ev) {
    ev.preventDefault();
    btn.disabled = true;
    var btnHtml = btn.innerHTML;
    btn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i> Enviando…';
    status.hidden = true;

    var fd = new FormData(form);
    fetch(form.action, { method: 'POST', body: fd, credentials: 'same-origin' })
      .then(function (r) { return r.json().catch(function () { return { ok: false, msg: 'Resposta inválida do servidor.' }; }); })
      .then(function (j) {
        setStatus(!!j.ok, j.msg || (j.ok ? 'Mensagem enviada.' : 'Falha ao enviar.'));
        if (j.ok) form.reset();
      })
      .catch(function () { setStatus(false, 'Erro de rede. Tente novamente.'); })
      .finally(function () {
        btn.disabled = false;
        btn.innerHTML = btnHtml;
      });
  });
})();
