/* ===== VITRINE: a moto num fundo contínuo, sem quadro e sem borda =====

   altura em que ela aparece, e as bordas da foto se apagam nele: a moto
   parece solta, sem retângulo em volta. Texto à esquerda, moto à direita.

// O carrossel é a vitrine que a loja monta no sistema (Gerenciar site →
// Vitrine), com foto, cor e preço próprios — não depende do estoque nem das
// páginas dos modelos. Só antes de a primeira vitrine existir é que o site
// mostra, para não ficar vazio, uma moto de cada modelo do pátio.
function montarVitrine(vitrine, estoque) {
  if (Array.isArray(vitrine)) {
    return vitrine.filter(v => v.ativo && v.foto?.url).map((v, i) => ({
      id: `${v.slug}-${i}`, modelo: v.modelo, categoria: v.categoria, cor: v.cor || "",
      tipo: "Nova", precoVenda: Number(v.preco) || 0, fotos: [v.foto]
    }));
  }
  const vistos = new Set();
  return estoque.filter(m => m.fotos?.[0]?.url && !vistos.has(slug(m.modelo)) && vistos.add(slug(m.modelo)));
}

async function carregar() {

  } else {
    const [{ initializeApp }, fs] = await Promise.all([
      import("https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js"),
      import("https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js")
    ]);
    const db = fs.getFirestore(initializeApp(firebaseConfig));
    fb = { fs, db };
    const cfg = await fs.getDoc(fs.doc(db, "config", "loja"));
    if (cfg.exists()) config = { ...config, ...cfg.data() };
    // o pátio ("vitrine", a cópia pública do estoque, sem custo nem chassi)
    // só é lido enquanto a loja não montou a vitrine dela
    let estoque = [];
    if (!Array.isArray(config.vitrineSite)) {

// Lê a foto antes de mostrar: as cores do estúdio dela e se ela é recortada
// (PNG com fundo transparente). Lê a própria imagem da tela, já baixada — o
// Cloudinary libera essa leitura. Se der errado, fica o cinza padrão.
let trocaAtual = 0;
function analisarFoto(im) {
  try {
    const c = document.createElement("canvas");
    c.width = c.height = 3;
    const ctx = c.getContext("2d", { willReadFrequently: true });
    const w = im.naturalWidth, h = im.naturalHeight;
    const p = Math.max(4, Math.round(Math.min(w, h) * 0.04));
    const ler = (sx, sy) => {
      ctx.clearRect(0, 0, 3, 3);
      ctx.drawImage(im, sx, sy, p, p, 0, 0, 3, 3);
      const d = ctx.getImageData(0, 0, 3, 3).data;
      let r = 0, g = 0, b = 0, a = 0;
      for (let i = 0; i < d.length; i += 4) { r += d[i]; g += d[i + 1]; b += d[i + 2]; a += d[i + 3]; }
      const n = d.length / 4;
      return [r / n, g / n, b / n, a / n];
    };
    // canto transparente = moto recortada, pousa no estúdio claro
    const cantos = [[0, 0], [w - p, 0], [0, h - p], [w - p, h - p]].map(([x, y]) => ler(x, y));
    const recortada = cantos.reduce((t, q) => t + q[3], 0) / 4 < 250;
    // a foto de estúdio é um degradê de cima para baixo (o chão é quase
    // branco). Guarda a cor de seis faixas da foto: o fundo da seção repete
    // esse degradê na mesma altura, e a borda da foto encosta sempre numa
    // cor igual à dela. A cor vem da borda ESQUERDA, que é a que fica de
    // frente para o texto e mais aparece; se ali tiver uma roda encostada
    // (bem mais escuro que a borda direita), vale a direita
    const luz = c => c[0] * .3 + c[1] * .59 + c[2] * .11;
    const media = lista => [0, 1, 2].map(i => Math.round(lista.reduce((t, q) => t + q[i], 0) / lista.length));
    const linhas = [0, .2, .4, .6, .8, 1].map(f => {
      const y = Math.min(h - p, Math.round(f * (h - p)));
      const esq = media([ler(0, y), ler(Math.round(.03 * (w - p)), y)]);
      const dir = media([ler(w - p, y), ler(Math.round(.97 * (w - p)), y)]);
      return luz(esq) >= luz(dir) - 25 ? esq : dir;
    });
    return { recortada, linhas };
  } catch (e) {
    return null;
  }
}

// Pinta a camada com o degradê da foto, cada faixa na altura em que ela
// aparece na tela. Refaz quando a janela muda de tamanho (a foto muda de lugar).
function pintarCamada(camada, img, linhas) {
  if (!linhas || !img.isConnected) return;
  const r = img.getBoundingClientRect(), cr = camada.getBoundingClientRect();
  if (!r.height || !cr.height) return;
  const rgb = c => `rgb(${c.join(",")})`;
  const pos = f => (((r.top - cr.top) + r.height * f) / cr.height * 100).toFixed(1);
  const faixas = linhas.map((c, i) => `${rgb(c)} ${pos(i / (linhas.length - 1))}%`);
  camada.style.background = `linear-gradient(to bottom,${rgb(linhas[0])} 0%,${faixas.join(",")},${rgb(linhas[linhas.length - 1])} 100%)`;
}
let repintar = null;
window.addEventListener("resize", () => {
  clearTimeout(repintar);
  repintar = setTimeout(() => {
    const camada = [...document.querySelectorAll("#vt-foto .vt-camada")].pop();
    if (camada?._linhas) pintarCamada(camada, camada.querySelector("img"), camada._linhas);
  }, 150);
});

// Troca de moto: a nova entra numa camada própria — a foto junto com o fundo
// de estúdio dela —, por cima da antiga e só depois de carregada e pintada.
// Como cada camada combina consigo mesma, a fusão de uma na outra nunca
// mostra a emenda da foto, nem por um instante.
async function trocarFoto(url, alt, troca) {
  const fundo = document.getElementById("vt-foto");
  if (!url) {
    fundo.innerHTML = `<div class="sem"><i class="ti ti-motorbike"></i></div>`;
    return;
  }
  const nova = document.createElement("div");
  nova.className = "vt-camada";
  nova.style.opacity = "0";
  nova.innerHTML = `<div class="vt-area"><img class="vt-moto" crossorigin="anonymous" src="${esc(url)}" alt="${esc(alt)}"></div>`;
  fundo.querySelector(".sem")?.remove();
  fundo.appendChild(nova);

  // espera a foto carregar — mas com teto, pra nunca ficar presa.
  // (img.decode() trava quando a aba está em segundo plano)
  const img = nova.querySelector("img");
  await new Promise(pronto => {
    if (img.complete && img.naturalWidth) return pronto();
    img.addEventListener("load", pronto, { once: true });
    img.addEventListener("error", pronto, { once: true });
    setTimeout(pronto, 2000);
  });
  if (troca !== trocaAtual) { nova.remove(); return; }  // já pediram outra moto
  const info = img.naturalWidth ? analisarFoto(img) : null;

  // recortada: estúdio claro com sombra; foto normal: o fundo da camada
  // vira o degradê do estúdio dela e as bordas da foto se apagam nele
  const recortada = !!info?.recortada;
  nova.classList.toggle("recortada", recortada);
  nova._linhas = recortada ? null : info?.linhas || null;
  if (nova._linhas) pintarCamada(nova, img, nova._linhas);
  // foto que já estava baixada chega aqui no mesmo instante em que entrou
  // com opacidade 0; sem forçar o navegador a registrar esse 0, ele pula
  // direto pro 1 e a moto aparece de estalo
  void nova.offsetWidth;
  nova.style.opacity = "1";
  // as de baixo só saem depois que a nova terminou de cobrir tudo — todas,
  // não só a última, para nenhuma ficar esquecida se as trocas se atropelarem
  for (const velha of fundo.querySelectorAll(".vt-camada")) {
    if (velha !== nova) setTimeout(() => velha.remove(), 600);
  }
}

function renderCarousel() {
  const lista = listaFiltrada();
  const inner = document.getElementById("car-stage");
  const vazio = document.getElementById("car-vazio");
  if (!lista.length) { inner.style.display = "none"; vazio.style.display = "flex"; return; }
  inner.style.display = "block";
  vazio.style.display = "none";

  const n = lista.length;
  idx = ((idx % n) + n) % n;

  const montar = () => {
    trocaAtual++;
    const m = lista[idx];
    const url = m.fotos?.[0]?.url;
    const nomeMoto = [m.modelo, m.cor].filter(Boolean).join(" ");

    document.getElementById("vt-specs").textContent =
      [m.categoria, m.tipo === "Usada" ? "Usada" : "0 km", m.cor].filter(Boolean).join(" · ");
    document.getElementById("vt-nome").textContent = m.modelo;
    document.getElementById("vt-preco").innerHTML = Number(m.precoVenda) > 0
      ? `A partir de <b>${fmtBRL(m.precoVenda)}</b>` : "Consulte as condições";
    document.getElementById("vt-btn").dataset.moto = nomeMoto;
    // modelo novo (chegou na nota mas ainda não tem página) fica sem o link,
    // em vez de levar para uma página de "modelo não encontrado"
    const ficha = document.getElementById("vt-ficha");
    ficha.href = `modelo.html?m=${esc(slug(m.modelo))}`;
    ficha.style.display = acharModelo(slug(m.modelo)) ? "" : "none";

    trocarFoto(url, m.modelo, trocaAtual);

    // a próxima da fila: o nome dela ao lado das setas, e a foto já vai
    // sendo baixada para a troca sair sem espera
    const varias = n > 1;
    document.getElementById("vt-setas").style.display = varias ? "" : "none";
    document.getElementById("vt-rodape").style.visibility = varias ? "visible" : "hidden";
    const prox = lista[(idx + 1) % n];
    document.getElementById("vt-prox").innerHTML = varias ? `Próxima<b>${esc(prox.modelo)}</b>` : "";
    if (varias && prox.fotos?.[0]?.url) {
      const pre = new Image();
      pre.crossOrigin = "anonymous";
      pre.src = prox.fotos[0].url;
    }

    if (varias) {
      document.getElementById("car-count").textContent =
        `${String(idx + 1).padStart(2, "0")} / ${String(n).padStart(2, "0")}`;
      document.getElementById("vt-barra-fill").style.width = `${((idx + 1) / n) * 100}%`;
    }
  };

  montar();
}
