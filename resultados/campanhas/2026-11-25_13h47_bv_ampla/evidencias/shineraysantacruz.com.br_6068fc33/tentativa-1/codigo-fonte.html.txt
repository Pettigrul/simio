async function carregar() {
  if (!base) { semModelo(); return; }
  render(true);   // nome, categoria e botões aparecem de imediato
  if (DEMO) {
    try {
      const db = JSON.parse(localStorage.getItem("shineray_db_v1") || "{}");
      if (db.config) config = { ...config, ...db.config };
      todosModelos = db.modelos || [];
      conteudo = todosModelos.find(m => m.slug === slugAtual) || null;
      todoEstoque = (db.vitrine || db.motos || []).filter(m => m.status === "disponivel");
      estoque = todoEstoque.filter(m => slug(m.modelo) === slugAtual);
    } catch (e) { /* segue com o padrão */ }
  } else {
    // Tudo aqui é "se der, melhora a página" — nada pode impedir o render.
    // Antes, uma falha ao carregar o Firebase deixava a página só com traços
    // e o cliente tinha que recarregar na mão.
    try {
      const [{ initializeApp }, fs] = await Promise.all([
        import("https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js"),
        import("https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js")
      ]);
      const db = fs.getFirestore(initializeApp(firebaseConfig));
      fb = { fs, db };
      // as três buscas vão juntas: em série, a página esperava uma soma de idas
      // ao banco em vez da mais demorada delas
      const [cfg, snapM, snapE] = await Promise.all([
        fs.getDoc(fs.doc(db, "config", "loja")).catch(e => { console.warn("config", e.code || e.message); }),
        // só o documento deste modelo: antes lia os 25 de uma vez
        fs.getDoc(fs.doc(db, "modelos", slugAtual)).catch(e => { console.warn("modelo", e.code || e.message); }),
        // "vitrine": cópia pública do estoque, sem custo/chassi
        fs.getDocs(fs.query(fs.collection(db, "vitrine"), fs.where("status", "==", "disponivel")))
          .catch(e => { console.warn("estoque", e.code || e.message); })
      ]);
      if (cfg?.exists()) config = { ...config, ...cfg.data() };
      if (snapM?.exists()) conteudo = { slug: slugAtual, ...snapM.data() };
      if (snapE) {
        todoEstoque = snapE.docs.map(d => d.data());
        estoque = todoEstoque.filter(m => slug(m.modelo) === slugAtual);
      }
    } catch (e) {
      console.warn("Firebase indisponível — a página abre com o que já está no site", e.message);
    }
  }
  render();
}

// busca o documento de um modelo qualquer (usado pelas sugestões do rodapé)
async function buscarModelo(sl) {
  if (DEMO) return todosModelos.find(m => m.slug === sl) || null;
  if (!fb) return null;
  try {
    const d = await fb.fs.getDoc(fb.fs.doc(fb.db, "modelos", sl));
    return d.exists() ? { slug: sl, ...d.data() } : null;
  } catch (e) { return null; }
}

function precoDoModelo(sl, precoBase) {
  const noPatio = todoEstoque
    .filter(x => slug(x.modelo) === sl)
    .map(x => Number(x.precoVenda) || 0)
    .filter(v => v > 0);
  if (noPatio.length) return { valor: Math.min(...noPatio), noPatio: true };
  return { valor: Number(precoBase) || 0, noPatio: false };
}

const embaralhar = a => {
  const c = [...a];
  for (let i = c.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [c[i], c[j]] = [c[j], c[i]];
  }
  return c;
};

// Os quatro números da capa saem da ficha técnica que o Iago preenche.
// Cada um tem como achar o item e como encurtar o valor, porque na ficha
// eles vêm por extenso ("8 CV / 8000 RPM", "Hidráulico Disco (Ø = 220mm)").
const NUMEROS_DA_CAPA = [
  { procura: "cilindrada", rotulo: "cilindradas",
    encurta: v => /^[\d.,]+$/.test(v) ? v : v },
  { procura: "potencia",   rotulo: "potência",
    encurta: v => v.split("/")[0].trim() },
  { procura: "tanque",     rotulo: "tanque",
    encurta: v => v.replace(/\(.*?\)/g, "").trim() },
  { procura: "dianteiro",  rotulo: "freio dianteiro",
    encurta: v => {
      const p = (v.match(/disco|tambor/i) || [v])[0];
      return p.charAt(0).toUpperCase() + p.slice(1).toLowerCase();
    } }
];
const semAcento = t => String(t || "").normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase();

function montarNumerosDaCapa(ficha) {
  const todos = ficha.flatMap(g => g.itens);
  const achados = [];
  for (const n of NUMEROS_DA_CAPA) {
    const item = todos.find(i => semAcento(i.rotulo).includes(n.procura));
    if (!item) continue;
    const valor = n.encurta(item.valor.trim());
    if (valor) achados.push({ valor, rotulo: n.rotulo });
  }
  const capa = $("capa");
  if (achados.length < 2) { capa.classList.remove("com-numeros"); return; }
  capa.classList.add("com-numeros");
  const faixa = $("capa-numeros");
  faixa.style.gridTemplateColumns = `repeat(${achados.length}, 1fr)`;
  faixa.innerHTML = achados.map(a =>
    `<div class="cn-item"><b>${esc(a.valor)}</b><span>${esc(a.rotulo)}</span></div>`).join("");
}

async function salvarLead(nome, fone) {
  const hoje = new Date();
  const dataHoje = `${hoje.getFullYear()}-${String(hoje.getMonth()+1).padStart(2,"0")}-${String(hoje.getDate()).padStart(2,"0")}`;
  const lead = { nome, telefone: fone, motoInteresse: base ? base.nome : "", etapa: "novo", proximoContato: dataHoje, obs: "Veio pela página do modelo", ...origemAtual() };
  if (DEMO) {
    const db = JSON.parse(localStorage.getItem("shineray_db_v1") || "{}");
    db.leads = db.leads || [];
    db.leads.unshift({ id: crypto.randomUUID(), ...lead, criadoEm: Date.now() });
    localStorage.setItem("shineray_db_v1", JSON.stringify(db));
  } else {
    await fb.fs.addDoc(fb.fs.collection(fb.db, "leads"), { ...lead, criadoEm: fb.fs.serverTimestamp() });
  }
}
