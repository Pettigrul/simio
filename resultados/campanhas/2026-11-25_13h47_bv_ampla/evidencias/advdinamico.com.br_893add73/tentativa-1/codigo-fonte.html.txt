    document.addEventListener('DOMContentLoaded', function() {
        const fixedTopMenu = document.getElementById('fixedTopMenu');
        const socioContainer = document.getElementById('socioContainer');
        const entityCard = document.querySelector('.entity-name');
        
        let isMenuVisible = false;
        let lastScrollTop = 0;
        
        function toggleFixedMenu() {
            if (!entityCard) return;
            
            const scrollTop = window.pageYOffset || document.documentElement.scrollTop;
            const entityCardRect = entityCard.getBoundingClientRect();
            const entityCardBottom = entityCardRect.bottom;
            
            // Adiciona uma pequena tolerância para evitar flickering
            const tolerance = 20;
            const shouldShow = entityCardBottom < -tolerance;
            
            if (shouldShow && !isMenuVisible) {
                fixedTopMenu.classList.add('show');
                isMenuVisible = true;
            } else if (!shouldShow && isMenuVisible) {
                fixedTopMenu.classList.remove('show');
                isMenuVisible = false;
            }
            
            lastScrollTop = scrollTop;
        }
        
        // Usar requestAnimationFrame para suavizar o scroll
        let ticking = false;
        function requestTick() {
            if (!ticking) {
                requestAnimationFrame(function() {
                    toggleFixedMenu();
                    ticking = false;
                });
                ticking = true;
            }
        }
        
        // Adicionar event listener com passive para melhor performance
        window.addEventListener('scroll', requestTick, { passive: true });
        
        // Verificar estado inicial após um pequeno delay
        setTimeout(toggleFixedMenu, 100);
    });

  function copyPixCodeDirectly(text) {
    const input = document.getElementById('pixCodeInput');

    input.select();
    input.setSelectionRange(0, 99999); // Para dispositivos móveis
