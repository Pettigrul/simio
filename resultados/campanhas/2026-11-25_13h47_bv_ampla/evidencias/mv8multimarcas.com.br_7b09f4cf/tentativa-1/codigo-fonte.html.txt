(function() {
    var lb = document.getElementById('qs-lightbox');
    if (!lb) return;
    var lbImg    = lb.querySelector('img');
    var lbClose  = lb.querySelector('.qs-lb-close');
    var triggers = document.querySelectorAll('.qs-zoom');

    function abrir(src, alt) {
        lbImg.src = src;
        lbImg.alt = alt || '';
        lb.classList.add('show');
        document.body.style.overflow = 'hidden';
    }
    function fechar() {
        lb.classList.remove('show');
        document.body.style.overflow = '';
        setTimeout(function() { if (!lb.classList.contains('show')) lbImg.src = ''; }, 250);
    }

    for (var i = 0; i < triggers.length; i++) {
        (function(el) {
            el.addEventListener('click', function() { abrir(el.src, el.alt); });
        }(triggers[i]));
    }
    lbClose.addEventListener('click', fechar);
    lb.addEventListener('click', function(e) { if (e.target === lb) fechar(); });
    document.addEventListener('keydown', function(e) {
        if ((e.key === 'Escape' || e.keyCode === 27) && lb.classList.contains('show')) fechar();
    });
}());
