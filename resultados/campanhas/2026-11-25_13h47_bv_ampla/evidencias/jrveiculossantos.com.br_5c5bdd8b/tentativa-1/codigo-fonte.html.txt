    if(days.includes(day)){
      r.classList.add('open-now');
      const open = (day>=1&&day<=6&&h>=9&&h<19)||(day===0&&h>=9&&h<16);
      if(open){
        const t = r.querySelector('.time');
        const wrap = document.createElement('div');
        wrap.style.cssText = 'display:flex;flex-direction:column;align-items:flex-end;gap:.28rem';
        t.replaceWith(wrap);
        wrap.appendChild(t);
        wrap.insertAdjacentHTML('beforeend','<span class="dot"><i></i>Aberto agora</span>');
      } else {
        const top = document.querySelector('.hours-card .top h3');
        if(top) top.textContent='Fechado agora';
      }
    }
  });
}

// ===== PROOF COUNTER =====
(function(){
  const numEl=document.querySelector('.proof-num');
  if(!numEl)return;
  let ran=false;
  new IntersectionObserver(es=>{
    if(es[0].isIntersecting&&!ran){
      ran=true;
      (window.PRELOADER_DONE||Promise.resolve()).then(()=>{
        const target=parseInt(numEl.dataset.target);
        const dur=2000,t0=performance.now(),ease=p=>1-Math.pow(1-p,3);
        (function tick(now){
          const p=Math.min((now-t0)/dur,1);
          numEl.textContent=Math.round(ease(p)*target);
          if(p<1)requestAnimationFrame(tick);
        })(t0);
      });
    }
  },{threshold:.3}).observe(numEl.closest('.hero-toon'));
})();
