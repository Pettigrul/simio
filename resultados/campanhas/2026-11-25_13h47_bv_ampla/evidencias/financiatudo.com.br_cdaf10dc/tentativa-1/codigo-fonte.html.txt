                  Como funciona o financiamento de imóveis residenciais?
                </span>
                <span class="faq-icone"></span>
              </summary>
              <div class="faq-resposta">
                O financiamento de imóveis residenciais permite obter crédito para adquirir uma casa, apartamento ou outra propriedade destinada à moradia. A instituição financeira analisa o comprador, o imóvel, a capacidade de pagamento e a documentação antes de definir as condições da operação.
              </div>
            </details>

            <details class="faq-item">
              <summary>
                <span class="faq-pergunta">
                  Posso financiar uma casa ou apartamento novo?
                </span>
                <span class="faq-icone"></span>
              </summary>
              <div class="faq-resposta">
                Sim. Casas e apartamentos novos podem ser financiados, desde que o imóvel e o comprador atendam aos critérios da instituição financeira e a documentação esteja regularizada.
              </div>
            </details>

            <details class="faq-item">
              <summary>
                <span class="faq-pergunta">
                  Posso financiar um imóvel residencial usado?
                </span>
                <span class="faq-icone"></span>
              </summary>
              <div class="faq-resposta">
                Sim. Imóveis usados também podem ser financiados. A instituição poderá realizar uma avaliação da propriedade e analisar sua documentação, localização, situação jurídica e valor de mercado.
              </div>
            </details>

            <details class="faq-item">
              <summary>
                <span class="faq-pergunta">
                  Quanto posso financiar na compra de um imóvel residencial?
                </span>
                <span class="faq-icone"></span>
              </summary>
              <div class="faq-resposta">
                O valor depende da instituição financeira, do valor de avaliação do imóvel, preço de aquisição, capacidade de pagamento e demais critérios da operação. O percentual financiado pode variar conforme o perfil do comprador e as características da propriedade.
              </div>
            </details>

              <summary>

                  Preciso dar entrada para financiar um imóvel residencial?
                </span>
                <span class="faq-icone"></span>
              </summary>
              <div class="faq-resposta">
                A necessidade de entrada depende das condições da instituição financeira e do percentual que poderá ser financiado. Quando o financiamento não cobre todo o valor do imóvel, o comprador deverá complementar a diferença com recursos próprios.
              </div>
            </details>

                </span>
                <span class="faq-icone"></span>
              </summary>
              <div class="faq-resposta">

    const status = section.querySelector('#faq-busca-status');
    const items = Array.from(section.querySelectorAll('.faq-item'));
    const categories = Array.from(section.querySelectorAll('[data-faq-categoria]'));
    const initialCategoryVisibility = new Map(
      categories.map((category) => [category, !category.hidden])
    );
    const footer = section.querySelector('.faq-footer');
    const toggle = section.querySelector('#faq-ver-todas');
    let expanded = false;

    const normalizeText = (text) => text
      .normalize('NFD')
      .replace(/[&#768;-&#879;]/g, '')
      .toLowerCase();

    const updateFaq = () => {
      const query = normalizeText(input.value.trim());
      let matches = 0;

      items.forEach((item, index) => {
        const question = item.querySelector('.faq-pergunta');
        const matchesQuery = !query || normalizeText(question.textContent).includes(query);
        const visibleInList = expanded || index < 10 || Boolean(query);

        item.hidden = !(matchesQuery && visibleInList);

        if (query && matchesQuery) {
          matches += 1;
        }
      });

      categories.forEach((category) => {
        category.hidden = Boolean(query)
          || (!expanded && !initialCategoryVisibility.get(category));
      });

      footer.hidden = Boolean(query);

      if (query) {
        status.hidden = false;
        status.textContent = matches === 0
          ? 'Nenhuma pergunta encontrada.'
          : matches === 1
            ? '1 pergunta encontrada.'
            : `${matches} perguntas encontradas.`;
      } else {
        status.hidden = true;
        status.textContent = '';
      }

      toggle.setAttribute('aria-expanded', String(expanded));
      toggle.textContent = expanded
        ? 'Mostrar menos &uarr;'
        : 'Ver todas as perguntas &rarr;';
    };

    input.addEventListener('input', updateFaq);
    toggle.addEventListener('click', function (event) {
      event.preventDefault();
      expanded = !expanded;
      updateFaq();
    });

    updateFaq();
