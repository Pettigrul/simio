function updateRouteProgress() {
  document.querySelectorAll('.route-progress').forEach(progress => {
    const tabs = Array.from(progress.querySelectorAll('.route-tab'));
    const prevBtn = progress.querySelector('.route-nav-btn-prev');
    const nextBtn = progress.querySelector('.route-nav-btn-next');
    if (!tabs.length || !prevBtn || !nextBtn) return;
    const activePath = activeRouteForTabs();
    let activeIndex = ROUTE_TABS.findIndex(item => item.path === activePath);
    if (activeIndex < 0) activeIndex = 0;
    tabs.forEach((tab, index) => tab.classList.toggle('is-active', index === activeIndex));
    prevBtn.disabled = activeIndex <= 0;
    nextBtn.disabled = activeIndex >= ROUTE_TABS.length - 1;
    prevBtn.onclick = () => { if (activeIndex > 0) navigate(ROUTE_TABS[activeIndex - 1].path); };
    nextBtn.onclick = () => { if (activeIndex < ROUTE_TABS.length - 1) navigate(ROUTE_TABS[activeIndex + 1].path); };
  });
}

function renderCoverageUI(holder, states) {
  coverageStateMap = Object.fromEntries(states.map(state => [String(state.sigla || '').toUpperCase(), state]));
  holder.innerHTML = `
    <div class="coverage-experience">
      <div class="coverage-map-stage">
        <div class="coverage-map-copy">
          <h3>Mapa do Brasil</h3>
          <p>Clique em um estado para ver as cidades e unidades publicadas.</p>
        </div>
        <div class="coverage-map-shell"><brazil-component id="coverageBrazilMap"></brazil-component></div>
      </div>
      <div class="coverage-list-card">
        <div class="coverage-state-head">
          <span class="eyebrow">Estado selecionado</span>
          <h3 id="coverageStateTitle">Selecione um estado</h3>
          <p id="coverageStateIntro">A relação de cidades e unidades aparecerá aqui.</p>
        </div>
        <div id="coverageStateList" class="coverage-company-list"></div>
      </div>
    </div>`;
  const stateList = document.getElementById('coverageStateList');
  const stateTitle = document.getElementById('coverageStateTitle');
  const stateIntro = document.getElementById('coverageStateIntro');
  const renderState = (uf) => {
    const key = String(uf || '').toUpperCase();
    const state = coverageStateMap[key];
    stateTitle.textContent = state ? `${state.nome} (${state.sigla})` : `${BRAZIL_STATES[key] || key}`;
    if (!state) {
      stateIntro.textContent = 'Ainda não há cidades ou unidades publicadas neste estado.';
      stateList.innerHTML = '<div class="coverage-empty-state"><strong>Sem cadastro publicado</strong><p>Assim que a operação confirmar novas unidades, este estado passará a exibir a relação completa aqui.</p></div>';
      return;
    }
    const cities = Array.isArray(state.cidades) ? state.cidades : [];
    const units = cities.reduce((acc, city) => acc + ((city.enderecos || []).length), 0);
    stateIntro.textContent = `${cities.length} cidade(s) publicada(s) e ${units} unidade(s) disponível(is) para consulta.`;
    stateList.innerHTML = cities.map(city => {
      const addresses = Array.isArray(city.enderecos) ? city.enderecos : [];
      return addresses.map(addr => `<div class="coverage-company-card"><strong>${escapeHtml(addr.razao_social || 'Unidade YB')}</strong><span>${escapeHtml(city.nome || '')} · ${escapeHtml(addr.endereco || '')}</span><div class="coverage-company-meta"><span class="badge">${escapeHtml(state.sigla || key)}</span>${addr.telefone ? `<span class="badge">Tel: ${escapeHtml(addr.telefone)}</span>` : ''}${addr.email ? `<span class="badge">${escapeHtml(addr.email)}</span>` : ''}</div></div>`).join('');
    }).join('') || '<div class="coverage-empty-state"><strong>Sem endereço disponível</strong><p>O estado foi identificado, mas ainda não há unidades completas publicadas.</p></div>';
  };
  const mapEl = document.getElementById('coverageBrazilMap');
  if (mapEl) mapEl.addEventListener('onStateSelected', (event) => renderState(event.detail));
  renderState(states[0]?.sigla || 'RJ');
}

function bindCoverage() {
  const holder = document.getElementById('coverageContent');
  if (!holder) return;
  const preloadStates = Array.isArray(PRELOADED_COVERAGE) ? PRELOADED_COVERAGE : [];
  if (preloadStates.length) {
    renderCoverageUI(holder, preloadStates);
  } else {
    holder.innerHTML = '<div class="subtle">Carregando cobertura...</div>';
  }
  fetch(COVERAGE_ENDPOINT)
    .then(res => res.json())
    .then(data => {
      const states = Array.isArray(data?.results) ? data.results : [];
      if (!states.length && preloadStates.length) return;
      renderCoverageUI(holder, states);
    })
    .catch(() => {
      if (!preloadStates.length) {
        holder.innerHTML = '<div class="notice-card"><strong>Não foi possível carregar agora.</strong><p>A página de cobertura está publicada, mas a consulta à base pública falhou neste momento.</p></div>';
      }
    });
}

function bindForms() {
  const contactForm = document.getElementById('contactForm');
  if (contactForm) {
    contactForm.addEventListener('submit', async (e) => {
      e.preventDefault();
      const fd = new FormData(contactForm);
      const payload = Object.fromEntries(fd.entries());
      const feedback = document.getElementById('contactFeedback');
      feedback.textContent = 'Enviando mensagem...';
      try {
        const res = await fetch(CONTACT_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
        const data = await res.json();
        feedback.textContent = data.ok ? 'Mensagem enviada com sucesso.' : (data.message || 'Não foi possível enviar agora.');
        if (data.ok) contactForm.reset();
      } catch (err) {
        feedback.textContent = 'Falha ao enviar mensagem.';
      }
    });
  }
  const checkoutForm = document.getElementById('checkoutForm');
  if (checkoutForm) {
    checkoutForm.addEventListener('submit', async (e) => {
      e.preventDefault();
      const fd = new FormData(checkoutForm);
      const payload = Object.fromEntries(fd.entries());
      payload.mensagem += `\n\nItens selecionados: ${STATE.cart.map(i => `${i.id} (qtd ${i.qty})`).join(', ') || 'nenhum item'}`;
      const feedback = document.getElementById('checkoutFeedback');
      feedback.textContent = 'Enviando solicitação...';
      try {
        const res = await fetch(CONTACT_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
        const data = await res.json();
        if (data.ok) {
          feedback.textContent = 'Solicitação registrada com sucesso. O time comercial poderá prosseguir com o atendimento.';
          checkoutForm.reset();
          STATE.cart = [];
          saveCart();
        } else {
          feedback.textContent = data.message || 'Não foi possível registrar agora.';
        }
      } catch (err) {
        feedback.textContent = 'Falha ao registrar solicitação.';
      }
    });
  }
}

function renderRoute() {
  updateCartCount();
  const path = window.location.pathname.replace(/\/$/, '') || '/';
  document.body.classList.toggle('route-home', path === '/');
  document.body.classList.toggle('route-inner', path !== '/');
  if (path === '/') {
    app.innerHTML = '';
    app.appendChild(homeView());
  } else if (path in Object.fromEntries(Object.keys(CATEGORIES).map(k => [`/${k}`, k]))) {
    app.innerHTML = listingView(path.slice(1));
  } else if (path.startsWith('/produto/')) {
    app.innerHTML = productView(decodeURIComponent(path.split('/').pop()));
  } else if (path === '/quem-somos') {
    app.innerHTML = contentView('quem-somos', 'Quem Somos', 'A YB opera como rede autorizada voltada ao atendimento comercial, peças originais, suporte e presença institucional no Brasil.');
  } else if (path === '/contato') {
    app.innerHTML = contactView();
  } else if (path === '/solicitacao-de-garantia') {
    app.innerHTML = warrantyRequestView();
  } else if (path === '/cobertura') {
    app.innerHTML = coverageView();
  } else if (path === '/carrinho') {
    app.innerHTML = cartView();
  } else if (path === '/checkout') {
    app.innerHTML = checkoutView();
  } else {
    app.innerHTML = notFoundView();
  }
  bindForms();
  bindCoverage();
  syncFooterTimeline();
  updateRouteProgress();
  updateDynamicSiteText();
}

window.navigate = navigate;
window.addToCart = function(id, qty=1, toast=true) {
  const key = decodeURIComponent(id);
  const found = STATE.cart.find(item => String(item.id) === String(key));
  if (found) found.qty += qty; else STATE.cart.push({ id: key, qty });
  saveCart();
  if (toast) alert('Item adicionado ao carrinho.');
};
window.changeQty = function(id, delta) {
  const item = STATE.cart.find(p => String(p.id) === String(id));
  if (!item) return;
  item.qty += delta;
  if (item.qty <= 0) STATE.cart = STATE.cart.filter(p => String(p.id) !== String(id));
  saveCart();
  renderRoute();
};
window.removeFromCart = function(id) {
  STATE.cart = STATE.cart.filter(p => String(p.id) !== String(id));
  saveCart();
  renderRoute();
};
