            } else if (numbers.startsWith('5') && numbers.length === 1 && e.inputType === 'insertText') {
                 // Evita que digitar '5' inicie o 55 indesejado
                 numbers = '';
            }

            // Limita a 11 dígitos (2 do DDD + 9 do número)
            if (numbers.length > 11) {
                numbers = numbers.substr(0, 11);
            }

            let formattedValue = phonePrefix;

            // Aplica a máscara sequencialmente: +55 (DD) 9XXXX-XXXX
            if (numbers.length > 0) {
                // (DD
                formattedValue += '(' + numbers.substr(0, 2);
                if (numbers.length > 2) {
                    // (DD) 9XXXX
                    formattedValue += ') ' + numbers.substr(2, 5);
                    if (numbers.length > 7) {
                        // (DD) 9XXXX-XXXX
                        formattedValue += '-' + numbers.substr(7, 4);
                    }
                }
            }

            // Atualiza o valor do campo
            target.value = formattedValue;

            // Tenta manter o cursor no final, exceto em deleções
            if (e.inputType !== 'deleteContentBackward') {
                 target.setSelectionRange(target.value.length, target.value.length);
            }
        }

        // Eventos do campo de telefone
        phoneInput.addEventListener('input', handlePhoneInput);

        // Impede posicionar o cursor antes do "+55 "
        phoneInput.addEventListener('click', function(e) {
            if (this.selectionStart < phonePrefix.length) {
                this.setSelectionRange(phonePrefix.length, phonePrefix.length);
            }
        });

        // Impede usar Backspace para apagar o "+55 "
        phoneInput.addEventListener('keydown', function(e) {
            if (e.key === 'Backspace' && this.selectionStart <= phonePrefix.length) {
                e.preventDefault();
            }
        });
    }


    // --- 5. FUNÇÃO: UI - REMOVER CORES DE VALIDAÇÃO AO INTERAGIR ---

    const fieldsToWatch = document.querySelectorAll('#bloco-franquia input:not([type="hidden"]), #bloco-franquia select');
    fieldsToWatch.forEach(field => {
        field.addEventListener('input', () => {
            field.classList.remove('is-invalid', 'is-valid');
            infoLog.style.display = "none";
        });
        field.addEventListener('change', () => {
            field.classList.remove('is-invalid', 'is-valid');
            infoLog.style.display = "none";
        });
    });


    // --- 6. FUNÇÃO: ENVIO DO FORMULÁRIO E VALIDAÇÃO FINAL (Apenas Franquia) ---

    // Definimos a função send globalmente para que o onclick do botão funcione
    window.send = function() {
        let formData = new FormData();
        let inputsFranquia = document.querySelectorAll("#bloco-franquia input:not([type='hidden'])");
        let selectsFranquia = document.querySelectorAll("#bloco-franquia select");
        let hasError = false; 

        formData.append("tenant_id", document.querySelector("#tenant_id").value);
        formData.append("form_id", document.querySelector("#form_id").value);

        // Validação dos Inputs de Texto
        for (let i = 0; i < inputsFranquia.length; i++) {
            const element = inputsFranquia[i];
            let isValid = true;
            let name_field = element.name.replace("-preview", "");

            formData.append(name_field, element.value);

            // Verifica E-mail
            if (element.type === "text" && element.name.includes('email')) {
                const emailRegex = /^(([^<>()\[\]\\.,;:\s@"]+(\.[^<>()\[\]\\.,;:\s@"]+)*)|(".+"))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$/;
                if (!emailRegex.test(element.value) || element.value.trim() === "") {
                    isValid = false;
                }
            } 
            // Verifica Telefone. O formato final "+55 (DD) 9XXXX-XXXX" tem 19 caracteres.
            else if (element.type === "text" && element.name.includes('telefone')) {
                if (element.value.length < 19) {
                    isValid = false; 
                }
            } 
            // Verifica campos obrigatórios vazios
            else if (element.ariaRequired && element.value.trim() === "") {
                isValid = false;
            }

            // Aplica/Remove classes de validação
            if (isValid) {
                element.classList.remove('is-invalid');
                element.classList.add('is-valid');
            } else {
                element.classList.remove('is-valid');
                element.classList.add('is-invalid');
                hasError = true;
            }
        }

        // Validação dos Selects
        for (let i = 0; i < selectsFranquia.length; i++) {
            const element = selectsFranquia[i];
            let isValid = true;
            let name_field = element.name.replace("-preview", "");
            
            formData.append(name_field, element.value);

            if (element.ariaRequired && (element.value === "" || element.value === null)) {
                isValid = false;
            }

            if (isValid) {
                element.classList.remove('is-invalid');
                element.classList.add('is-valid');
            } else {
                element.classList.remove('is-valid');
                element.classList.add('is-invalid');
                hasError = true;
            }
        }

        // Se houver erro, para o envio e avisa
        if (hasError) {
            infoLog.style.color = "#e74c3c"; // Vermelho
            infoLog.style.display = "block";
            infoLog.innerText = "Por favor, preencha corretamente os campos destacados em vermelho.";
            return; 
        }

        // Se tudo estiver certo, envia
        infoLog.style.color = "#2ecc71"; // Verde
        infoLog.style.display = "block";
        infoLog.innerText = "Enviando dados...";

        axios
            .post("https://back.gdigital.com.br/form/register", formData)
            .then((resp) => {
                const query_params = resp.data && resp.data.query_params ? resp.data.query_params : '';
                let url = "https://veloxsolucoesfinanceiras.com.br/franquia/obrigado/" + query_params;
                window.location.href = url;
            })
            .catch((error) => {
                infoLog.style.color = "#e74c3c";
                infoLog.style.display = "block";
                infoLog.innerText = "Ocorreu um erro ao enviar. Tente novamente.";
            });
    }

});
