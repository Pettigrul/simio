    const carTitle = document.getElementById('simPage-car-title');
    const carCode = document.getElementById('simPage-car-code');

    const summaryPrice = document.getElementById('simPage-summary-price');
    const summaryEntry = document.getElementById('simPage-summary-entry');
    const summaryPrincipal = document.getElementById('simPage-summary-principal');
    const highlightInstallment = document.getElementById('simPage-highlight-installment');
    const tableAll = document.getElementById('simPage-all-installments');
    const btnWhatsapp = document.getElementById('simPage-btn-whatsapp');

    const monthlyRate = 0.0139; // Taxa de 1,39% ao mês

    function formatCurrency(val) {
        return val.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' });
    }

    function calculateInstallment(principal, months) {
        if (principal <= 0) return 0;
        return (principal * monthlyRate) / (1 - Math.pow(1 + monthlyRate, -months));
    }

    function updateSimulation() {
        let price = parseFloat(inputPrice.value) || 0;
        if (price < 5000) price = 5000;

        let entry = parseFloat(inputEntry.value) || 0;
        if (entry > price * 0.85) {
            entry = price * 0.85;
            inputEntry.value = Math.round(entry);
        }
        if (entry < 0) {
            entry = 0;
            inputEntry.value = 0;
        }

        const percent = Math.round((entry / price) * 100);
        rangeEntry.value = percent;
        entryPercentText.textContent = percent + '% de Entrada';

        const principal = Math.max(0, price - entry);
        const selectedMonths = parseInt(hiddenMonths.value) || 48;

        priceDisplay.textContent = formatCurrency(price);
        summaryPrice.textContent = formatCurrency(price);
        summaryEntry.textContent = '- ' + formatCurrency(entry);
        summaryPrincipal.textContent = formatCurrency(principal);

        const currentInst = calculateInstallment(principal, selectedMonths);
        highlightInstallment.textContent = selectedMonths + 'x de ' + formatCurrency(currentInst);

        // Preenche tabela comparativa de prazos
        tableAll.innerHTML = '';
        const prazos = [12, 24, 36, 48, 60];
        prazos.forEach(m => {
            const inst = calculateInstallment(principal, m);
            const isCur = m === selectedMonths;
            const div = document.createElement('div');
            div.style.display = 'flex';
            div.style.justifyContent = 'space-between';
            div.style.padding = '6px 10px';
            div.style.borderRadius = '6px';
            div.style.fontSize = '13px';
            div.style.background = isCur ? '#EFF6FF' : '#F8FAFC';
            div.style.border = isCur ? '1px solid #BFDBFE' : '1px solid #E2E8F0';
            div.style.color = isCur ? '#1E3A8A' : '#334155';
            div.style.fontWeight = isCur ? '700' : '500';
            div.innerHTML = `<span>${m} parcelas:</span><strong>${formatCurrency(inst)}</strong>`;
            tableAll.appendChild(div);
        });

        // Monta Mensagem do WhatsApp
        const opt = vSelect.options[vSelect.selectedIndex];
        const carName = opt ? (opt.dataset.title || 'Veículo do Estoque') : 'Veículo';
        const carCodeStr = opt ? (opt.dataset.code || '') : '';
        const tradeInfo = hasTrade.checked ? `\n- Veículo na Troca: Sim (${tradeModel.value || 'Carro usado'} ${tradeYear.value || ''})` : '';

        let msg = `Oi, eu gostaria de mais informações e aprovação de financiamento:\n\n`
                + `- Veículo: ${carName} ${carCodeStr ? '(Código: ' + carCodeStr + ')' : ''}\n`
                + `- Valor do Veículo: ${formatCurrency(price)}\n`
                + `- Entrada Simulada: ${formatCurrency(entry)} (${percent}%)\n`
                + `- Saldo Financiado: ${formatCurrency(principal)}\n`
                + `- Prazo Pretendido: ${selectedMonths}x de aproximadamente ${formatCurrency(currentInst)}${tradeInfo}\n\n`
                + `Poderia analisar meu crédito e me atender na Manancial 7 Veículos?`;

        btnWhatsapp.href = `https://wa.me/5513991569310?text=${encodeURIComponent(msg)}`;
    }

    // Eventos
    vSelect.addEventListener('change', function() {
        const opt = this.options[this.selectedIndex];
        if (opt && opt.value !== 'custom') {
            const p = parseFloat(opt.dataset.price) || 0;
            inputPrice.value = p;
            inputEntry.value = Math.round(p * 0.30);
            if (carPreview) {
                carPreview.style.display = 'flex';
                carImg.src = opt.dataset.cover;
                carTitle.textContent = opt.dataset.title;
                carCode.textContent = 'Código: ' + opt.dataset.code;
            }
        } else {
            if (carPreview) carPreview.style.display = 'none';
        }
        updateSimulation();
    });

    inputPrice.addEventListener('input', function() {
        const p = parseFloat(this.value) || 0;
        inputEntry.value = Math.round(p * 0.30);
        updateSimulation();
    });

    inputEntry.addEventListener('input', updateSimulation);

    rangeEntry.addEventListener('input', function() {
        const p = parseFloat(inputPrice.value) || 0;
        const perc = parseInt(this.value) || 0;
        inputEntry.value = Math.round(p * (perc / 100));
        updateSimulation();
    });

    document.querySelectorAll('.btn-entry-shortcut').forEach(btn => {
        btn.addEventListener('click', function() {
            const perc = parseInt(this.dataset.percent);
            const p = parseFloat(inputPrice.value) || 0;
            inputEntry.value = Math.round(p * (perc / 100));
            updateSimulation();
        });
    });

    hasTrade.addEventListener('change', function() {
        tradeBox.style.display = this.checked ? 'block' : 'none';
        updateSimulation();
    });

    if (tradeModel) tradeModel.addEventListener('input', updateSimulation);
    if (tradeYear) tradeYear.addEventListener('input', updateSimulation);

    document.querySelectorAll('.btn-month-choice').forEach(btn => {
        btn.addEventListener('click', function() {
            document.querySelectorAll('.btn-month-choice').forEach(b => {
                b.style.borderColor = 'var(--border-color)';
                b.style.background = 'white';
                b.style.color = 'var(--text-main)';
            });
            this.style.borderColor = 'var(--primary)';
            this.style.background = 'var(--primary)';
            this.style.color = 'white';
            hiddenMonths.value = this.dataset.months;
            updateSimulation();
        });
    });

    updateSimulation();
});
