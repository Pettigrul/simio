        (function () {
            var toggle = document.querySelector('.hn-nav-toggle');
            var menu = document.getElementById('hn-menu');
            function layerLayoutTop(el) {
                var main = document.querySelector('main');
                var y = main.offsetTop;
                var panes = main.querySelectorAll(':scope > .hn-layer');
                for (var i = 0; i < panes.length; i++) {
                    if (panes[i] === el) break;
                    y += panes[i].offsetHeight;
                }
                return y;
            }

            function scrollToLayer(id, fromHash) {
                var target = document.getElementById(id);
                if (!target || !target.classList.contains('hn-layer')) return;
                var header = document.querySelector('.hn-header');
                var headerH = header ? header.offsetHeight : 0;
                var top = layerLayoutTop(target) - headerH - 8;
                if (top < 0) top = 0;
                var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                window.scrollTo({ top: top, behavior: reduce ? 'auto' : 'smooth' });
                if (!fromHash && location.hash !== '#' + id) {
                    history.pushState(null, '', '#' + id);
                }
            }

            if (toggle && menu) {
                toggle.addEventListener('click', function () {
                    var open = menu.classList.toggle('is-open');
                    toggle.setAttribute('aria-expanded', open ? 'true' : 'false');
                });
                menu.addEventListener('click', function (event) {
                    var link = event.target.closest('a');
                    if (!link) return;
                    menu.classList.remove('is-open');
                    toggle.setAttribute('aria-expanded', 'false');
                    var href = link.getAttribute('href') || '';
                    if (href.charAt(0) !== '#') return;
                    event.preventDefault();
                    scrollToLayer(href.slice(1), false);
                });
            }

            window.addEventListener('hashchange', function () {
                scrollToLayer(location.hash.slice(1), true);
            });

            var layers = document.querySelectorAll('[data-parallax]');
            var reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
            function matchLayerHeight() {
                if (reduceMotion) return;
                var panes = document.querySelectorAll('main > .hn-layer');
                if (panes.length < 2) return;
                for (var i = 1; i < panes.length; i++) panes[i].style.minHeight = '';
                var height = Math.ceil(panes[0].getBoundingClientRect().height);
                for (var j = 1; j < panes.length; j++) panes[j].style.minHeight = height + 'px';
            }
            matchLayerHeight();
            window.addEventListener('load', matchLayerHeight);
            window.addEventListener('resize', matchLayerHeight);

            if (layers.length && !reduceMotion) {
                var ticking = false;
                function applyParallax() {
                    ticking = false;
                    var viewH = window.innerHeight || 1;
                    for (var i = 0; i < layers.length; i++) {
                        var img = layers[i];
                        var frame = img.parentElement;
                        if (!frame) continue;
                        var rect = frame.getBoundingClientRect();
                        if (rect.bottom < -200 || rect.top > viewH + 200) continue;
                        var speed = parseFloat(img.getAttribute('data-parallax')) || 0;
                        var delta = (rect.top + rect.height / 2 - viewH / 2) / viewH;
                        if (img.hasAttribute('data-parallax-fit')) {
                            var drift = delta * speed;
                            if (drift > 36) drift = 36;
                            if (drift < -36) drift = -36;
                            img.style.transform = 'translate3d(0,' + drift.toFixed(2) + 'px,0)';
                            continue;
                        }
                        var extra = frame.offsetHeight * 0.42;
                        var y = -extra * 0.5 + delta * speed;
                        if (y > -6) y = -6;
                        if (y < -extra + 6) y = -extra + 6;
                        img.style.transform = 'translate3d(0,' + y.toFixed(2) + 'px,0)';
                    }
                }
                function requestParallax() {
                    if (ticking) return;
                    ticking = true;
                    window.requestAnimationFrame(applyParallax);
                }
                window.addEventListener('scroll', requestParallax, { passive: true });
                window.addEventListener('resize', requestParallax);
                applyParallax();
            }
        })();
    </script>
