(function () {
    if (window.__meelionFeedbackWidgetAjaxBound) return;

    const isValidEmail = (value) => {
        const email = (value || '').trim();
        if (!email) return false;
        return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email);
    };

    const getCsrfToken = (form) => {
        const csrfInput = form?.querySelector?.('input[name="_csrfToken"]');
        if (csrfInput && csrfInput.value) return csrfInput.value;
        const meta = document.querySelector('meta[name="csrfToken"]');
        return meta?.content || '';
    };

    const setMessage = (form, type, text) => {
        const box = document.getElementById('feedbackWidgetMessage')
            || form?.closest?.('.feedback-widget')?.querySelector?.('#feedbackWidgetMessage');
        if (!box) return;
        box.classList.remove('feedback-widget__message--success', 'feedback-widget__message--error');
        box.classList.add(type === 'success'
            ? 'feedback-widget__message--success'
            : 'feedback-widget__message--error'
        );
        box.textContent = text || '';
        box.hidden = !text;
    };

    document.addEventListener('submit', async (e) => {
        const form = e.target;
        if (!form || form.id !== 'feedbackWidgetForm') return;

        e.preventDefault();

        const submitBtn = form.querySelector('button[type="submit"]');

        const nome = form.querySelector('#feedbackNome')?.value?.trim() || '';
        const email = form.querySelector('#feedbackEmail')?.value?.trim() || '';
        const tipoRadio = form.querySelector('input[name="tipo"]:checked');
        const tipo = tipoRadio?.value || '';
        const mensagem = form.querySelector('#feedbackMensagem')?.value?.trim() || '';

        if (!nome) {
            setMessage(form, 'error', 'Por favor, informe seu nome.');
            form.querySelector('#feedbackNome')?.focus();
            return;
        }

        if (!email) {
            setMessage(form, 'error', 'Por favor, informe seu email.');
            form.querySelector('#feedbackEmail')?.focus();
            return;
        }

        if (!isValidEmail(email)) {
            setMessage(form, 'error', 'Por favor, insira um email válido.');
            form.querySelector('#feedbackEmail')?.focus();
            return;
        }

        if (!tipo || (tipo !== 'duvida' && tipo !== 'sugestao')) {
            setMessage(form, 'error', 'Por favor, selecione se é Dúvida ou Sugestão.');
            form.querySelector('#feedbackTipoDuvida')?.focus();
            return;
        }

        if (!mensagem) {
            setMessage(form, 'error', 'Por favor, escreva sua mensagem.');
            form.querySelector('#feedbackMensagem')?.focus();
            return;
        }

        if (mensagem.length < 10) {
            setMessage(form, 'error', 'Escreva um pouco mais (mínimo de 10 caracteres).');
            form.querySelector('#feedbackMensagem')?.focus();
            return;
        }

        const csrfToken = getCsrfToken(form);

        const payload = new URLSearchParams({
            nome,
            email,
            tipo,
            mensagem
        });

        if (csrfToken) payload.append('_csrfToken', csrfToken);

        try {
            if (submitBtn) {
                submitBtn.disabled = true;
                submitBtn.textContent = 'Enviando...';
            }

            const headers = {
                'Content-Type': 'application/x-www-form-urlencoded;charset=UTF-8',
                'X-Requested-With': 'XMLHttpRequest'
            };
            if (csrfToken) headers['X-CSRF-Token'] = csrfToken;

            const response = await fetch('/feedback/', {
                method: 'POST',
                headers,
                credentials: 'same-origin',
                body: payload.toString()
            });

            let data = null;
            try {
                data = await response.json();
            } catch (parseErr) {
                data = null;
            }

            if (!response.ok || !data || data.success !== true) {
                const msg = data?.message || 'Não foi possível enviar seu feedback. Tente novamente.';
                setMessage(form, 'error', msg);
                return;
            }

            setMessage(form, 'success', data.message || 'Obrigado!');
        } catch (err) {
            setMessage(form, 'error', 'Erro de conexão. Tente novamente.');
        } finally {
            if (submitBtn) {
                submitBtn.disabled = false;
                submitBtn.textContent = 'Enviar';
            }
        }
    }, true);
})();
