(function () {
  // Hero video: seamless loop via the YouTube IFrame API (avoids the
  // rebuffer flash of the native loop=1&playlist trick).
  var heroFrame = document.getElementById('tf-hero-yt');
  function initHeroLoop() {
    if (!window.YT || !window.YT.Player) return;
    new YT.Player(heroFrame, {
      events: {
        onStateChange: function (e) {
          if (e.data === YT.PlayerState.ENDED) {
            e.target.seekTo(0);
            e.target.playVideo();
          }
        }
      }
    });
  }
  if (window.YT && window.YT.Player) {
    initHeroLoop();
  } else {
    window.onYouTubeIframeAPIReady = initHeroLoop;
    var ytTag = document.createElement('script');
    ytTag.src = 'https://www.youtube.com/iframe_api';
    document.head.appendChild(ytTag);
  }

  // Speaker carousel: drag-to-scroll + prev/next buttons.
  var track = document.getElementById('speakers-track');
  var isDown = false, startX = 0, startScroll = 0;

  track.addEventListener('mousedown', function (e) {
    isDown = true;
    startX = e.pageX;
    startScroll = track.scrollLeft;
    track.classList.add('dragging');
  });
  window.addEventListener('mouseup', function () {
    if (isDown) { isDown = false; track.classList.remove('dragging'); }
  });
  track.addEventListener('mouseleave', function () {
    if (isDown) { isDown = false; track.classList.remove('dragging'); }
  });
  track.addEventListener('mousemove', function (e) {
    if (!isDown) return;
    track.scrollLeft = startScroll - (e.pageX - startX);
  });

  document.querySelector('.carousel-prev').addEventListener('click', function () {
    track.scrollBy({ left: -260, behavior: 'smooth' });
  });
  document.querySelector('.carousel-next').addEventListener('click', function () {
    track.scrollBy({ left: 260, behavior: 'smooth' });
  });

  // Speaker cards: click to expand (accordion — one at a time).
  document.querySelectorAll('.speaker-card[data-toggle]').forEach(function (card) {
    card.addEventListener('click', function () {
      var wasExpanded = card.classList.contains('expanded');
      document.querySelectorAll('.speaker-card.expanded').forEach(function (c) {
        c.classList.remove('expanded');
      });
      if (!wasExpanded) card.classList.add('expanded');
    });
  });
  document.querySelectorAll('.speaker-bio a').forEach(function (a) {
    a.addEventListener('click', function (e) { e.stopPropagation(); });
  });

  // CTA form: client-side required-field validation before the Mailchimp POST.
  var form = document.getElementById('cta-form');
  var successMsg = document.getElementById('cta-success');
  form.addEventListener('submit', function (e) {
    var fields = ['FNAME', 'EMAIL', 'PHONE', 'COMPANY'];
    var hasError = false;
    fields.forEach(function (name) {
      var input = form.elements[name];
      var wrapper = input.closest('div');
      var errorEl = wrapper.querySelector('.form-error-msg');
      if (!input.value.trim()) {
        hasError = true;
        input.classList.add('field-error');
        if (errorEl) errorEl.classList.add('visible');
      } else {
        input.classList.remove('field-error');
        if (errorEl) errorEl.classList.remove('visible');
      }
    });
    if (hasError) {
      e.preventDefault();
      return;
    }
    form.style.display = 'none';
    successMsg.classList.add('visible');
  });

  // Programação: expand/collapse the full schedule.
  var programClip = document.getElementById('program-clip');
  var programToggle = document.getElementById('program-toggle');
  programToggle.addEventListener('click', function () {
    var open = programClip.classList.toggle('open');
    programToggle.setAttribute('aria-expanded', open ? 'true' : 'false');
    programToggle.textContent = open ? 'Ver menos' : 'Ver programação completa';
  });
})();
