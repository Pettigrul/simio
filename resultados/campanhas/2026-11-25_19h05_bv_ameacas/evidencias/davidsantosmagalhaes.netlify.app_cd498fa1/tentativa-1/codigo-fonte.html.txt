        (function () {
            var VALIDOS = ['inicio-manha', 'fim-manha', 'tarde', 'fim-tarde', 'noite', 'madrugada'];
            function detectar() {
                var m = new Date().getHours() * 60 + new Date().getMinutes();
                if (m >= 300 && m < 480)  return 'inicio-manha';
                if (m >= 480 && m < 720)  return 'fim-manha';
                if (m >= 720 && m < 1020) return 'tarde';
                if (m >= 1020 && m < 1110) return 'fim-tarde';
                if (m >= 1110 || m < 30)  return 'noite';
                return 'madrugada';
            }
            var salvo = sessionStorage.getItem('periodo_manual');
            var periodo = (salvo && VALIDOS.indexOf(salvo) !== -1) ? salvo : detectar();
            document.documentElement.setAttribute('data-periodo', periodo);
        })();
