    <script>
    (function() {
        'use strict';
        
        var UTM_PARAMS = ['utm_source', 'utm_medium', 'utm_campaign', 'utm_term', 'utm_content'];
        var CLICK_IDS = ['gclid', 'fbclid', 'msclkid', 'ttclid', 'li_fat_id', 'twclid', 'dclid'];
        var COOKIE_DAYS = 30;
        
        // ============================================
        // UTILIDADES
        // ============================================
        function getUrlParam(param) {
            var urlParams = new URLSearchParams(window.location.search);
            return urlParams.get(param) || '';
        }
        
        function setCookie(name, value, days) {
            if (!value) return;
            var expires = new Date();
            expires.setTime(expires.getTime() + (days * 24 * 60 * 60 * 1000));
            document.cookie = name + '=' + encodeURIComponent(value) + 
                              ';expires=' + expires.toUTCString() + 
                              ';path=/;SameSite=Lax';
        }
        
        function getCookie(name) {
            var match = document.cookie.match(new RegExp('(^| )' + name + '=([^;]+)'));
            return match ? decodeURIComponent(match[2]) : '';
        }
        
        function setStorage(key, value) {
            try {
                if (value) localStorage.setItem(key, value);
            } catch(e) {}
        }
        
        function getStorage(key) {
            try {
                return localStorage.getItem(key) || '';
            } catch(e) {
                return '';
            }
        }
        
        // Salva em cookie E localStorage (redundância)
        function saveData(key, value, days) {
            if (!value) return;
            setCookie(key, value, days);
            setStorage(key, value);
        }
        
        // Lê de cookie OU localStorage (fallback)
        function getData(key) {
            return getCookie(key) || getStorage(key);
        }
        
        // ============================================
        // DETECÇÃO DE FONTE VIA REFERRER
        // ============================================
        function parseReferrer() {
            var ref = document.referrer;
            if (!ref) return null;
            
            try {
                var refUrl = new URL(ref);
                var refHost = refUrl.hostname.toLowerCase();
                
                // Ignora referrer do próprio site
                if (refHost === window.location.hostname) return null;
                
                // Mapeamento de domínios para source/medium
                var sources = {
                    // Google
                    'google': { source: 'google', medium: 'organic' },
                    'google.com': { source: 'google', medium: 'organic' },
                    'google.com.br': { source: 'google', medium: 'organic' },
                    // Facebook/Meta
                    'facebook.com': { source: 'facebook', medium: 'social' },
                    'fb.com': { source: 'facebook', medium: 'social' },
                    'l.facebook.com': { source: 'facebook', medium: 'social' },
                    'lm.facebook.com': { source: 'facebook', medium: 'social' },
                    'm.facebook.com': { source: 'facebook', medium: 'social' },
                    'web.facebook.com': { source: 'facebook', medium: 'social' },
                    // Instagram
                    'instagram.com': { source: 'instagram', medium: 'social' },
                    'l.instagram.com': { source: 'instagram', medium: 'social' },
                    // LinkedIn
                    'linkedin.com': { source: 'linkedin', medium: 'social' },
                    'lnkd.in': { source: 'linkedin', medium: 'social' },
                    // Twitter/X
                    'twitter.com': { source: 'twitter', medium: 'social' },
                    'x.com': { source: 'twitter', medium: 'social' },
                    't.co': { source: 'twitter', medium: 'social' },
                    // YouTube
                    'youtube.com': { source: 'youtube', medium: 'social' },
                    'youtu.be': { source: 'youtube', medium: 'social' },
                    // TikTok
                    'tiktok.com': { source: 'tiktok', medium: 'social' },
                    // Bing
                    'bing.com': { source: 'bing', medium: 'organic' },
                    // Yahoo
                    'yahoo.com': { source: 'yahoo', medium: 'organic' },
                    'search.yahoo.com': { source: 'yahoo', medium: 'organic' },
                    // DuckDuckGo
                    'duckduckgo.com': { source: 'duckduckgo', medium: 'organic' },
                    // Pinterest
                    'pinterest.com': { source: 'pinterest', medium: 'social' },
                    'pin.it': { source: 'pinterest', medium: 'social' },
                    // Reddit
                    'reddit.com': { source: 'reddit', medium: 'social' },
                    // WhatsApp
                    'whatsapp.com': { source: 'whatsapp', medium: 'social' },
                    'api.whatsapp.com': { source: 'whatsapp', medium: 'social' },
                    // Telegram
                    'telegram.org': { source: 'telegram', medium: 'social' },
                    't.me': { source: 'telegram', medium: 'social' },
                    // Email providers (webmail)
                    'mail.google.com': { source: 'gmail', medium: 'email' },
                    'outlook.live.com': { source: 'outlook', medium: 'email' },
                    'outlook.office.com': { source: 'outlook', medium: 'email' },
                    'mail.yahoo.com': { source: 'yahoo_mail', medium: 'email' }
                };
                
                // Busca correspondência exata primeiro
                if (sources[refHost]) {
                    return sources[refHost];
                }
                
                // Busca parcial (para subdomínios)
                for (var domain in sources) {
                    if (refHost.indexOf(domain) !== -1) {
                        return sources[domain];
                    }
                }
                
                // Referrer desconhecido - usa o domínio como source
                return {
                    source: refHost.replace('www.', ''),
                    medium: 'referral'
                };
                
            } catch(e) {
                return null;
            }
        }
        
        // ============================================
        // DETECÇÃO DE CLICK IDS (gclid, fbclid, etc)
        // ============================================
        function detectClickId() {
            for (var i = 0; i < CLICK_IDS.length; i++) {
                var clickId = getUrlParam(CLICK_IDS[i]);
                if (clickId) {
                    var mapping = {
                        'gclid': { source: 'google', medium: 'cpc' },
                        'fbclid': { source: 'facebook', medium: 'cpc' },
                        'msclkid': { source: 'bing', medium: 'cpc' },
                        'ttclid': { source: 'tiktok', medium: 'cpc' },
                        'li_fat_id': { source: 'linkedin', medium: 'cpc' },
                        'twclid': { source: 'twitter', medium: 'cpc' },
                        'dclid': { source: 'google', medium: 'display' }
                    };
                    return {
                        clickIdName: CLICK_IDS[i],
                        clickIdValue: clickId,
                        inferred: mapping[CLICK_IDS[i]] || null
                    };
                }
            }
            return null;
        }
        
        // ============================================
        // CAPTURA PRINCIPAL
        // ============================================
        function captureAttribution() {
            var now = new Date().toISOString();
            
            // 1. Verifica se há UTMs na URL
            var hasUtm = false;
            var utmData = {};
            
            for (var i = 0; i < UTM_PARAMS.length; i++) {
                var param = UTM_PARAMS[i];
                var value = getUrlParam(param);
                if (value) {
                    hasUtm = true;
                    utmData[param] = value;
                }
            }
            
            // 2. Se tem UTMs, salva normalmente
            if (hasUtm) {
                // Entry: só salva se não existir
                for (var param in utmData) {
                    var entryKey = param + '_entry';
                    if (!getData(entryKey)) {
                        saveData(entryKey, utmData[param], COOKIE_DAYS);
                    }
                }
                
                // Last: sempre atualiza
                for (var param in utmData) {
                    var lastKey = param + '_last';
                    saveData(lastKey, utmData[param], COOKIE_DAYS);
                }
                
                // Salva timestamp e landing page
                if (!getData('first_visit_at')) {
                    saveData('first_visit_at', now, COOKIE_DAYS);
                    saveData('landing_page_entry', window.location.href, COOKIE_DAYS);
                }
                saveData('last_visit_at', now, COOKIE_DAYS);
                saveData('landing_page_last', window.location.href, COOKIE_DAYS);
                
                return; // UTMs encontradas, não precisa de fallback
            }
            
            // 3. Sem UTMs - tenta detectar via click ID (gclid, fbclid, etc)
            var clickIdData = detectClickId();
            if (clickIdData && clickIdData.inferred) {
                // Salva o click ID
                saveData(clickIdData.clickIdName + '_entry', clickIdData.clickIdValue, COOKIE_DAYS);
                saveData(clickIdData.clickIdName + '_last', clickIdData.clickIdValue, COOKIE_DAYS);
                
                // Infere UTMs do click ID
                var inferredUtm = {
                    'utm_source': clickIdData.inferred.source,
                    'utm_medium': clickIdData.inferred.medium,
                    'utm_campaign': '(auto-detected)',
                    'utm_term': '',
                    'utm_content': ''
                };
                
                // Entry: só salva se não existir
                for (var param in inferredUtm) {
                    var entryKey = param + '_entry';
                    if (!getData(entryKey) && inferredUtm[param]) {
                        saveData(entryKey, inferredUtm[param], COOKIE_DAYS);
                    }
                }
                
                // Last: sempre atualiza
                for (var param in inferredUtm) {
                    var lastKey = param + '_last';
                    if (inferredUtm[param]) {
                        saveData(lastKey, inferredUtm[param], COOKIE_DAYS);
                    }
                }
                
                if (!getData('first_visit_at')) {
                    saveData('first_visit_at', now, COOKIE_DAYS);
                    saveData('landing_page_entry', window.location.href, COOKIE_DAYS);
                }
                saveData('last_visit_at', now, COOKIE_DAYS);
                saveData('landing_page_last', window.location.href, COOKIE_DAYS);
                
                return;
            }
            
            // 4. Sem UTMs e sem click ID - tenta detectar via referrer
            var referrerData = parseReferrer();
            if (referrerData) {
                var inferredUtm = {
                    'utm_source': referrerData.source,
                    'utm_medium': referrerData.medium,
                    'utm_campaign': '(auto-detected)',
                    'utm_term': '',
                    'utm_content': ''
                };
                
                // Entry: só salva se não existir
                for (var param in inferredUtm) {
                    var entryKey = param + '_entry';
                    if (!getData(entryKey) && inferredUtm[param]) {
                        saveData(entryKey, inferredUtm[param], COOKIE_DAYS);
                    }
                }
                
                // Last: sempre atualiza (mas NÃO sobrescreve com vazio se já tem valor)
                for (var param in inferredUtm) {
                    var lastKey = param + '_last';
                    if (inferredUtm[param]) {
                        saveData(lastKey, inferredUtm[param], COOKIE_DAYS);
                    }
                }
                
                // Salva referrer original também
                saveData('referrer_entry', document.referrer, COOKIE_DAYS);
                saveData('referrer_last', document.referrer, COOKIE_DAYS);
            }
            
            // 5. Salva timestamps e landing page mesmo sem atribuição
            if (!getData('first_visit_at')) {
                saveData('first_visit_at', now, COOKIE_DAYS);
                saveData('landing_page_entry', window.location.href, COOKIE_DAYS);
                
                // Se é primeira visita sem nenhuma fonte, marca como direto
                if (!getData('utm_source_entry')) {
                    saveData('utm_source_entry', '(direct)', COOKIE_DAYS);
                    saveData('utm_medium_entry', '(none)', COOKIE_DAYS);
                }
            }
            
            saveData('last_visit_at', now, COOKIE_DAYS);
            saveData('landing_page_last', window.location.href, COOKIE_DAYS);
        }
        
        // ============================================
        // PUSH PARA DATALAYER
        // ============================================
        function pushToDataLayer() {
            var data = {};
            var allKeys = [
                'utm_source_entry', 'utm_medium_entry', 'utm_campaign_entry', 'utm_term_entry', 'utm_content_entry',
                'utm_source_last', 'utm_medium_last', 'utm_campaign_last', 'utm_term_last', 'utm_content_last',
                'gclid_entry', 'gclid_last', 'fbclid_entry', 'fbclid_last',
                'referrer_entry', 'referrer_last',
                'landing_page_entry', 'landing_page_last',
                'first_visit_at', 'last_visit_at'
            ];
            
            for (var i = 0; i < allKeys.length; i++) {
                var value = getData(allKeys[i]);
                if (value) {
                    data[allKeys[i]] = value;
                }
            }
            
            if (Object.keys(data).length > 0) {
                window.dataLayerPYS = window.dataLayerPYS || [];
                window.dataLayer.push(data);
            }
        }
        
        // ============================================
        // EXECUTA
        // ============================================
        captureAttribution();
        pushToDataLayer();
        
    })();

				$(this).parent().parent().parent().parent().next().slideDown();
