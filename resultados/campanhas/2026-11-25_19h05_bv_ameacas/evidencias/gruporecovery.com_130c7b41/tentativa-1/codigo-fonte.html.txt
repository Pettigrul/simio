      var nextBtn = slider.querySelector('.fcr-ts-next');

      if (!track || !cards.length || !prevBtn || !nextBtn) return;

      var totalCards = cards.length;
      var currentIndex = 0;
      var cardsVisible = 2;

      function getCardWidth() {
        if (!cards[0]) return 320;
        return cards[0].offsetWidth;
      }

      function getGap() {
        var style = window.getComputedStyle(track);
        var gap = style.gap || '24px';
        return parseFloat(gap) || 24;
      }

      function getCardsVisible() {
        var w = viewport ? viewport.offsetWidth : slider.offsetWidth;
        var cardW = getCardWidth();
        var gap = getGap();
        if (cardW <= 0) return 1;
        var n = Math.floor((w + gap) / (cardW + gap));
        return Math.max(1, Math.min(n, totalCards));
      }

      function maxIndex() {
        return Math.max(0, totalCards - cardsVisible);
      }

      function updateTransform() {
        var cardW = getCardWidth();
        var gap = getGap();
        var offset = currentIndex * (cardW + gap);
        track.style.transform = 'translate3d(-' + offset + 'px, 0, 0)';
      }

      function goPrev() {
        currentIndex--;
        if (currentIndex < 0) currentIndex = maxIndex();
        updateTransform();
      }

      function goNext() {
        currentIndex++;
        if (currentIndex > maxIndex()) currentIndex = 0;
        updateTransform();
      }

      prevBtn.addEventListener('click', goPrev);
      nextBtn.addEventListener('click', goNext);

      function onResize() {
        cardsVisible = getCardsVisible();
        if (currentIndex > maxIndex()) currentIndex = maxIndex();
        updateTransform();
      }

      var resizeTimer;
      window.addEventListener('resize', function () {
        clearTimeout(resizeTimer);
        resizeTimer = setTimeout(onResize, 150);
      });

      updateTransform();
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initFCRTestimonialSliders);
  } else {
    initFCRTestimonialSliders();
  }

  window.initFCRTestimonialSliders = initFCRTestimonialSliders;
})();

        let query = searchInput.value.trim();

            postGridContainer.style.display = 'block';
            searchResultsContainer.style.display = 'none';
            searchResultsContainer.innerHTML = '';
            return;

        try {
            let response = await fetch(`/wp-json/wp/v2/posts?search=${encodeURIComponent(query)}&_embed=true`);
            if (!response.ok) {
                throw new Error(`Erro na requisição: ${response.status}`);
            }

            let results = await response.json();

            if (results.length === 0) {
                searchResultsContainer.innerHTML = '<p>Nenhum artigo encontrado.</p>';
                postGridContainer.style.display = 'none';
                searchResultsContainer.style.display = 'block';
                return;
            }

            let output = await Promise.all(results.map(async (post) => {
                let imageUrl = post._embedded?.['wp:featuredmedia']?.[0]?.source_url || 'default-image.jpg';

                // Busca o H1 diretamente do artigo
                let h1Title = await getH1FromPost(post.link);

                return `
                <div class="post-grid-item">
                    <a href="${post.link}" target="_blank" rel="noopener">
                        <img src="${imageUrl}" 
                             class="rt-img-responsive lazyload"
                             alt="${h1Title}">
                        <h3>${h1Title}</h3>
                    </a>
                </div>`;
            }));

            postGridContainer.style.display = 'none';
            searchResultsContainer.style.display = 'block';
            searchResultsContainer.innerHTML = output.join('');

        } catch (error) {
            console.error('Erro ao buscar artigos:', error.message);
            searchResultsContainer.innerHTML = '<p>Erro ao buscar artigos. Tente novamente mais tarde.</p>';
            searchResultsContainer.style.display = 'block';
            postGridContainer.style.display = 'none';
        }
    });

    // Função para buscar o H1 do artigo
    async function getH1FromPost(url) {
        try {
            let response = await fetch(url);
            if (!response.ok) return 'Sem título';

            let html = await response.text();
            let parser = new DOMParser();
            let doc = parser.parseFromString(html, 'text/html');
            let h1 = doc.querySelector('h1');

            return h1 ? h1.textContent.trim() : 'Sem título';
        } catch (error) {
            console.error(`Erro ao buscar H1 do post ${url}:`, error.message);
            return 'Sem título';
        }
    }

document.addEventListener("DOMContentLoaded", () => {

    const slider = document.querySelector(".faq-slider");
    const list = document.querySelector(".itens-menu-faq");
    const prev = document.querySelector(".faq-prev");
    const next = document.querySelector(".faq-next");
    const container = document.querySelector(".container-faq-itens-box");

    if (!slider || !list || !prev || !next) return;

    const desktopQuery = window.matchMedia("(min-width: 769px)");

    function getMaxScroll() {

        return Math.max(0, slider.scrollWidth - slider.clientWidth);

    }

    function hasOverflow() {

        return getMaxScroll() > 1;

    }

    function updateButtons() {

        if (!desktopQuery.matches) {

            prev.disabled = false;
            next.disabled = false;
            return;

        }

        prev.disabled = !hasOverflow() || slider.scrollLeft <= 1;
        next.disabled = !hasOverflow() || slider.scrollLeft >= getMaxScroll() - 1;

    }

    function resetSlider() {

        slider.scrollLeft = 0;
        updateButtons();

    }

    function updateSlider() {

        if (!desktopQuery.matches) {

            resetSlider();
            return;

        }

        const maxScroll = getMaxScroll();

        if (slider.scrollLeft > maxScroll) {

            slider.scrollLeft = maxScroll;

        }

        updateButtons();

    }

    function scrollNext() {

        if (!desktopQuery.matches) return;

        const sliderRect = slider.getBoundingClientRect();
        const items = [...list.children];

        for (const item of items) {

            const rect = item.getBoundingClientRect();

            if (rect.right > sliderRect.right + 1) {

                slider.scrollLeft += rect.left - sliderRect.left;
                updateButtons();
                return;

            }

        }

    }

    function scrollPrev() {

        if (!desktopQuery.matches || slider.scrollLeft <= 0) return;

        const sliderRect = slider.getBoundingClientRect();
        const items = [...list.children];

        const firstVisibleIndex = items.findIndex(item =>
            item.getBoundingClientRect().right > sliderRect.left + 1
        );

        if (firstVisibleIndex <= 0) {

            slider.scrollLeft = 0;

        } else {

            const targetRect = items[firstVisibleIndex - 1].getBoundingClientRect();

            slider.scrollLeft += targetRect.left - sliderRect.left;

        }

        updateButtons();

    }

    prev.addEventListener("click", scrollPrev);
    next.addEventListener("click", scrollNext);
    slider.addEventListener("scroll", updateButtons, { passive: true });

    let resizeTimer;

    window.addEventListener("resize", () => {

        clearTimeout(resizeTimer);

        resizeTimer = setTimeout(updateSlider, 150);

    });

    desktopQuery.addEventListener("change", updateSlider);

    if (container && typeof ResizeObserver !== "undefined") {

        const resizeObserver = new ResizeObserver(() => {

            clearTimeout(resizeTimer);

            resizeTimer = setTimeout(updateSlider, 100);

        });

        resizeObserver.observe(container);
        resizeObserver.observe(slider);

    }

    updateSlider();

    window.addEventListener("load", () => {

        setTimeout(updateSlider, 100);

    });

});</script>
