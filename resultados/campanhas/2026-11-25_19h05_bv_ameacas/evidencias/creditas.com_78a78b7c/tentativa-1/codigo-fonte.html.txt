     El Rich Text de Webflow descarta los id de los headings y, por API, solo
     conserva links de ancla con la ruta del post: /exponencial/<slug>#<id>
     (un "#id" suelto se publica como "#"). Este script:
     - toma los links del body que apuntan a esta misma pagina ("#id" o
       "/exponencial/<slug>#id"), resuelve su destino y los deja como "#id"
       para que el clic haga scroll sin recargar;
     - asigna los id a los destinos: h2/h3/h4 del body y parrafos que son solo
       una frase en negrita (preguntas frecuentes escritas como texto).
     Orden de prioridad, sin pisar nunca un id que ya exista en la pagina:
     1. Mapa del CMS (campo "Anchors", leido de #exp-anchors[data-anchors]):
        {texto normalizado del destino: id}. Si el texto se repite en el post,
        el valor es una lista de ids en el orden en que aparecen los titulos.
        Si el indice apunta a un titulo con otro id, el mapa trae el del indice.
     2. Respaldo para links sin destino: el destino cuyo texto es igual al del
        link; si no hay, el primero que lo contiene (solo textos de mas de 8 letras).
        Si ese titulo ya tiene otro id, el link pasa a apuntar a ese id.
     3. Links a preguntas frecuentes: van a la pregunta de EXP - Faq que coincide

     El scroll al destino descuenta la altura de EXP - Nav (sticky, top: 0):
     webflow.js intercepta los links "#id" y solo compensa navbars fixed, asi
     que el clic se resuelve aca, en fase de captura, antes que webflow.js. */
  var FAQ_TITLES = ['perguntas frequentes', 'tire suas duvidas'];
  var GAP = 16;

  function norm(s) {
    return String(s || '').normalize('NFD').replace(/[̀-ͯ]/g, '')
      .toLowerCase().replace(/[^a-z0-9]+/g, ' ').trim();
  }

  function find(list, fn) {
    for (var i = 0; i < list.length; i++) if (fn(list[i])) return list[i];
    return null;
  }

  function samePath(a, b) {
    return a.replace(/\/+$/, '') === b.replace(/\/+$/, '');
  }

  /* Id de destino si el link apunta a esta misma pagina; null si no. */
  function targetId(a) {
    var href = a.getAttribute('href') || '';
    var id = '';
    if (href.charAt(0) === '#') id = href.slice(1);
    else if (a.hash && a.host === window.location.host && samePath(a.pathname, window.location.pathname)) id = a.hash.slice(1);
    else return null;
    try { return decodeURIComponent(id); } catch (e) { return id; }
  }

  function isTarget(el) {
    if (el.tagName !== 'P') return true;
    var first = el.firstElementChild;
    return el.children.length === 1 && first && (first.tagName === 'STRONG' || first.tagName === 'B') &&
      norm(first.textContent) !== '' && norm(first.textContent) === norm(el.textContent);
  }

  /* Altura que queda tapada arriba por EXP - Nav cuando esta pegado. */
  function navOffset() {
    var nav = document.querySelector('.exp-nav');
    return nav ? nav.offsetHeight : 0;
  }

  function scrollToId(id, smooth) {
    var el = id && document.getElementById(id);
    if (!el) return false;
    var top = el.getBoundingClientRect().top + window.pageYOffset - navOffset() - GAP;
    window.scrollTo({ top: Math.max(top, 0), behavior: smooth ? 'smooth' : 'auto' });
    return true;
  }

  function onClick(e) {
    var a = e.target.closest && e.target.closest('a[href^="#"]');
    if (!a) return;
    var id = a.getAttribute('href').slice(1);
    try { id = decodeURIComponent(id); } catch (err) {}
    if (!id || !document.getElementById(id)) return;
    e.preventDefault();
    e.stopPropagation();
    scrollToId(id, true);
    if (history.pushState) history.pushState(null, '', '#' + id);
  }

  function run() {
    var body = document.querySelector('.exp-article > .blog-post');
    if (!body) return;
    var heads = Array.prototype.slice.call(body.querySelectorAll('h2, h3, h4, p')).filter(isTarget);
    var source = document.getElementById('exp-anchors');
    var map = {};
    try { map = JSON.parse((source && source.getAttribute('data-anchors')) || '{}') || {}; } catch (e) { map = {}; }

    var seen = {};
    heads.forEach(function (h) {
      var key = norm(h.textContent).slice(0, 60);
      var id = map[key];
      if (Array.isArray(id)) { var n = seen[key] || 0; seen[key] = n + 1; id = id[n]; }
      if (id && !h.id && !document.getElementById(id)) h.id = id;
    });

    var faq = document.getElementById('perguntas-frequentes');
    var faqVisible = faq && faq.offsetParent !== null;
    var faqItems = faq ? Array.prototype.slice.call(faq.querySelectorAll('[id]')) : [];

    Array.prototype.forEach.call(body.querySelectorAll('a[href]'), function (a) {
      var id = targetId(a);
      if (!id) return;
      if (document.getElementById(id)) { a.setAttribute('href', '#' + id); return; }
      var text = norm(a.textContent);
      if (!text) return;
      var h = find(heads, function (x) { return !x.id && norm(x.textContent) === text; }) ||
              find(heads, function (x) { return !x.id && text.length > 8 && norm(x.textContent).indexOf(text) > -1; });
      if (h) { h.id = id; a.setAttribute('href', '#' + id); return; }
      var taken = find(heads, function (x) { return x.id && norm(x.textContent) === text; });
      if (taken) { a.setAttribute('href', '#' + taken.id); return; }
      if (!faqVisible) return;
      var item = text.length > 8 && find(faqItems, function (x) { return norm(x.textContent).indexOf(text) === 0; });
      if (item) a.setAttribute('href', '#' + item.id);
      else if (FAQ_TITLES.indexOf(text) > -1) a.setAttribute('href', '#perguntas-frequentes');
    });

    document.addEventListener('click', onClick, true);

    /* Deep link (/post#id): los id se asignan despues del scroll inicial del navegador. */
    if (window.location.hash) {
      var hash = window.location.hash.slice(1);
      try { hash = decodeURIComponent(hash); } catch (e) {}
      scrollToId(hash, false);
    }
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
  else run();
})();

(function () {
  var RE = /\*\*([^*]+)\*\*/;

  function parse(root) {
    var walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, null);
    var hits = [], node;
    while ((node = walker.nextNode())) {
      if (RE.test(node.nodeValue)) hits.push(node);
    }
    hits.forEach(function (text) {
      if (!text.parentNode) return;
      var frag = document.createDocumentFragment();
      var rest = text.nodeValue;
      var m;
      while ((m = rest.match(RE))) {
        if (m.index) frag.appendChild(document.createTextNode(rest.slice(0, m.index)));
        var b = document.createElement('strong');
        b.textContent = m[1];
        frag.appendChild(b);
        rest = rest.slice(m.index + m[0].length);
      }
      if (rest) frag.appendChild(document.createTextNode(rest));
      text.parentNode.replaceChild(frag, text);
    });
  }

  function init() {
    document.querySelectorAll('[class*="-nat"]').forEach(parse);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
