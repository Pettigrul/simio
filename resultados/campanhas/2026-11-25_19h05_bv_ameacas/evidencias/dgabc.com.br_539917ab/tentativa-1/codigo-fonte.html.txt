    function getCookie(k) {
        var cookies = " " + document.cookie;
        var key = "id=" + k;
        var start = cookies.indexOf(key);

        if (start === -1) return null;

        var pos = start + key.length;
        var last = cookies.indexOf(";", pos);

        if (last !== -1) return cookies.substring(pos, last);

        return cookies.substring(pos);

    function montarGolsMobile(gols, jogoId) {
        if (!gols || gols.length === 0)
            return "";

        const golsMandante = gols.filter(g => g.lado === "mandante");
        const golsVisitante = gols.filter(g => g.lado === "visitante");

        const total = gols.length;
        const temMais = total > 2;

        function montarLista(lista, classeExtra) {
            return lista.map((g, index) => `
            <div class="copa-mobile-gol ${index >= 2 ? "gol-extra" : ""}" ${index >= 2 ? "style='display:none;'" : ""}>
                ${g.atleta || ""} ${g.minuto || ""}' ⚽
            </div>
        `).join("");
        }

        return `
        <div class="copa-mobile-gols" id="gols-${jogoId}">
            <div class="copa-mobile-golsMandante">
                ${montarLista(golsMandante, "mandante")}
            </div>

            <div class="copa-mobile-golsVisitante">
                ${montarLista(golsVisitante, "visitante")}
            </div>

            ${temMais ? `
                <button type="button"
                        class="copa-mobile-expandir"
                        onclick="toggleGolsMobile('${jogoId}')">
                    ⌄
                </button>
            ` : ""}
        </div>
    `;
    }

    function toggleGolsMobile(jogoId) {
        const box = document.getElementById(`gols-${jogoId}`);
        if (!box) return;

        const extras = box.querySelectorAll(".gol-extra");
        const btn = box.querySelector(".copa-mobile-expandir");

        if (!extras.length || !btn) return;

        const aberto = box.classList.contains("aberto");

        extras.forEach(item => {
            item.style.display = aberto ? "none" : "block";
        });

        box.classList.toggle("aberto", !aberto);
        btn.innerHTML = aberto ? "⌄" : "⌃";
    }
