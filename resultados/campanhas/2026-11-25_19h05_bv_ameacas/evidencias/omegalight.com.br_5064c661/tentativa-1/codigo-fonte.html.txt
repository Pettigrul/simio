  <script>
    jQuery(document).ready(function ($) {
      var bannerPrincipal = new Splide('.banner-principal', {
        type: 'fade',
        rewind: true,
        cover: true,
        arrows: false,
        height: '100vh',
        autoplay: true,
        interval: 8000,
        pauseOnHover: false,
        pauseOnFocus: false
      });

      // Banners em vídeo: sempre que o slide fica ativo o vídeo recomeça do zero e toca
      // sozinho (sem som). Enquanto ele roda, o autoplay do carrossel fica em espera e o
      // banner avança assim que o vídeo termina (com um limite de segurança caso o vídeo
      // trave). Os banners de vídeo não param com mouse ou foco, igual aos de imagem.
      var bannerSlides = Array.prototype.slice.call(document.querySelectorAll('.banner-principal .splide__list > .splide__slide'));
      var bannerVideos = Array.prototype.slice.call(document.querySelectorAll('.banner-principal .splide__slide video'));
      var reduzMovimento = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
      var timerVideo = null;

      function indiceDoSlide(el) {
        return bannerSlides.indexOf(el ? el.closest('.splide__slide') : null);
      }

      function bannerAutoplay(ligar) {
        var ap = bannerPrincipal.Components && bannerPrincipal.Components.Autoplay;
        if (!ap || bannerPrincipal.length < 2) return;
        if (ligar) { ap.play(); } else { ap.pause(); }
      }

      function limparTimerVideo() {
        if (timerVideo) { clearTimeout(timerVideo); timerVideo = null; }
      }

      function avancarBanner() {
        limparTimerVideo();
        if (bannerPrincipal.length > 1) { bannerPrincipal.go('>'); }
      }

      // Limite de segurança: duração do vídeo + 1,5 s (ou 20 s se a duração ainda não é conhecida)
      function agendarAvanco(v) {
        limparTimerVideo();
        if (bannerPrincipal.length < 2) return;
        var dur = (isFinite(v.duration) && v.duration > 0) ? v.duration : 0;
        var ms = dur ? Math.max(dur * 1000 + 1500, 3000) : 20000;
        timerVideo = setTimeout(avancarBanner, ms);
      }

      function sincronizarBannerVideos(indiceAtivo) {
        if (typeof indiceAtivo !== 'number') indiceAtivo = bannerPrincipal.index;
        limparTimerVideo();
        var tocando = false;

        bannerVideos.forEach(function (v) {
          var ativo = indiceDoSlide(v) === indiceAtivo;
          if (!ativo || reduzMovimento) {
            if (!v.paused) v.pause();
            return;
          }
          v.muted = true;
          v.loop = bannerPrincipal.length < 2;
          try { v.currentTime = 0; } catch (e) {}
          var p = v.play();
          tocando = true;
          agendarAvanco(v);
          if (p && typeof p.catch === 'function') {
