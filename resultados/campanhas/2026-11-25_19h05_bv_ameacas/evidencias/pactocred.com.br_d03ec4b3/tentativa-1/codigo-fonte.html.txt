    el.innerHTML = '<div class="pc-chart-vazio">Sem dados dispon&iacute;veis para o per&iacute;odo.</div>';
    return;
  }
  var max = 0; dados.forEach(function(d){ if (d.v > max) max = d.v; });
  var h = (el.clientHeight || 92) - 18;
  el.innerHTML = dados.map(function(d, i){
    var alt = Math.max(4, Math.round((d.v / (max || 1)) * h));
    var cor = i === dados.length - 1 ? 'var(--o)' : 'var(--vm)';
    var rot = String(d.v).replace('.', ',');
    return '<div class="bc" title="' + d.m + ': R$ ' + rot + ' mi"><div class="bc-bar" style="height:' + alt + 'px;background:' + cor + '"></div><div class="bc-lbl">' + d.m + '</div></div>';
  }).join('');
}

function toggleTheme() {
  var cur  = document.documentElement.getAttribute('data-theme') || 'dark';
  var next = cur === 'dark' ? 'light' : 'dark';
  document.documentElement.setAttribute('data-theme', next);
  try { localStorage.setItem('pc-theme', next); } catch(e) {}
  /* atualizar todos os botões de tema (site + dash) */
  document.querySelectorAll('.theme-btn').forEach(function(b){
    b.setAttribute('data-mode', next);
    b.title = next === 'dark' ? 'Alternar para modo claro' : 'Alternar para modo escuro';
  });
}

/* ─── ENVIOS DE FORMULÁRIOS DE OVERLAY ──────────────────── */
function enviarContato() {
  var nome = pcValor('ctc-nome'), email = pcValor('ctc-email'), msg = pcValor('ctc-msg'), assunto = pcValor('ctc-assunto');
  if (!nome || !email || !msg) { showToast('Preencha todos os campos obrigatórios.', 'err'); return; }
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email)) { showToast('Informe um e-mail válido.', 'err'); return; }
  if (PC_CONFIG.leadsEndpoint) {
    pcEnviarLead({ tipo: 'contato', assunto: assunto, nome: nome, email: email, mensagem: msg, pagina: location.pathname, host: location.host, data: new Date().toISOString() })
      .then(function(){ gaEvento('generate_lead', { origem: 'contato' }); showToast('Mensagem recebida. Responderemos pelo e-mail informado.', 'ok'); setTimeout(closePage, 1400); })
      .catch(function(){ showToast('Não foi possível enviar agora. Nada foi enviado; tente novamente ou use o WhatsApp.', 'err'); });
    return;
  }
  pcAbrirWhatsApp(['Olá, Pacto Cred. Contato pelo site.', 'Assunto: ' + assunto, 'Nome: ' + nome, 'E-mail: ' + email, 'Mensagem: ' + msg], 'contato');
  showToast('Mensagem preparada no WhatsApp. Ela só é enviada quando você confirmar no aplicativo.', 'info');
}
function enviarLicenciado() {
  var nome = pcValor('lic-nome'), email = pcValor('lic-email'), tel = pcValor('lic-tel'), cidade = pcValor('lic-cidade');
  if (!nome || !email || !tel || !cidade) { showToast('Preencha nome, e-mail, WhatsApp e cidade.', 'err'); return; }
  if (tel.replace(/\D/g, '').length < 10) { showToast('Informe o WhatsApp com DDD.', 'err'); return; }
  var dados = { tipo: 'licenciado', nome: nome, email: email, whatsapp: tel, cidade: cidade, uf: pcValor('lic-uf'), modelo: pcValor('lic-modelo'), experiencia: pcValor('lic-exp') };
  if (PC_CONFIG.leadsEndpoint) {
    dados.pagina = location.pathname; dados.host = location.host; dados.data = new Date().toISOString();
    pcEnviarLead(dados)
      .then(function(){ gaEvento('generate_lead', { origem: 'licenciado' }); showToast('Solicitação recebida. A equipe de licenciamento fará contato.', 'ok'); setTimeout(closePage, 1400); })
      .catch(function(){ showToast('Não foi possível enviar agora. Nada foi enviado; tente novamente ou use o WhatsApp.', 'err'); });
    return;
  }
  pcAbrirWhatsApp(['Olá, Pacto Cred. Quero ser licenciado.', 'Nome: ' + nome, 'E-mail: ' + email, 'WhatsApp: ' + tel, 'Cidade/UF: ' + cidade + ' - ' + dados.uf, 'Modelo: ' + dados.modelo, dados.experiencia ? 'Experiência: ' + dados.experiencia : ''].filter(Boolean), 'licenciado');
  showToast('Solicitação preparada no WhatsApp. Ela só é enviada quando você confirmar no aplicativo.', 'info');
}

  if (tc) { !isE ? tc.classList.add('on') : tc.classList.remove('on'); }
  /* blocos login */
  var be = document.getElementById('blk-entrar');
  if (be) be.style.display = isE ? 'block' : 'none';
  /* reset COMPLETO do cadastro ao trocar de aba */
  var bc   = document.getElementById('blk-cadastro');
  var bc2  = document.getElementById('blk-cad2');
  var bsuc = document.getElementById('blk-sucesso');
  var bk   = document.getElementById('lc1-back');
  if (bc)   bc.style.display   = !isE ? 'block' : 'none';
  if (bc2)  bc2.style.display  = 'none';
  if (bsuc) bsuc.style.display = 'none';
  if (bk)   bk.style.display   = 'block';
  /* reset tipo selecionado */
  if (!isE) {
    selCadTipo = null;
    document.querySelectorAll('.cad-tipo').forEach(function(c){ c.classList.remove('on'); });
    var btn = document.getElementById('btn-cad-continuar');
    if (btn){ btn.setAttribute('disabled',''); btn.style.opacity='.4'; }
  }
}

/* ─── LOGIN FLOW ─────────────────────────────────────── */
function loginStep2() {
  var email = (document.getElementById('liEmail') || {}).value || '';
  var pass  = (document.getElementById('liPass')  || {}).value || '';
  var errEl = document.getElementById('login-cred-err');
  if (!email.trim() || !pass) {
    if (errEl) { errEl.textContent = 'Preencha e-mail e senha.'; errEl.style.display = 'block'; }
    return;
  }
  if (errEl) errEl.style.display = 'none';
  var aviso = document.getElementById('pc-login-aviso');
  if (!PC_CONFIG.authLoginEndpoint) {
    /* sem backend de autenticação: nenhuma credencial é aceita */
    if (aviso) {
      aviso.innerHTML = 'O acesso à plataforma ainda não está liberado por este site. Para abrir ou acessar sua conta, fale com o atendimento: <a href="https://wa.me/' + PC_CONFIG.whatsapp + '" target="_blank" rel="noopener">WhatsApp</a> ou <a href="mailto:sac@pactocred.com.br">sac@pactocred.com.br</a>. <br><button type="button" onclick="pcAbrirDemonstracao()">Ver demonstração do painel</button>';
      aviso.classList.add('show');
    }
    return;
  }
  var btn = document.querySelector('#blk-entrar .btn-blk');
  if (btn) btn.disabled = true;
  fetch(PC_CONFIG.authLoginEndpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, credentials: 'same-origin', body: JSON.stringify({ login: email.trim(), senha: pass }) })
    .then(function(r){ if (!r.ok) throw new Error('auth'); return r.json().catch(function(){ return {}; }); })
    .then(function(res){
      gaEvento('login', { method: 'email_cpf' });
      if (document.getElementById('liPass')) document.getElementById('liPass').value = '';
      if (res && res.mfa) {
        var lc1 = document.getElementById('lc1'), lc2 = document.getElementById('lc2');
        if (lc1) lc1.style.display = 'none';
        if (lc2) lc2.style.display = 'flex';
        var m0 = document.getElementById('m0'); if (m0) m0.focus();
      } else if (res && res.redirect) {
        window.location.href = res.redirect;
      } else {
        pcSessao.auth = true; go('dash');
      }
    })
    .catch(function(){ if (errEl) { errEl.textContent = 'E-mail/CPF ou senha inválidos.'; errEl.style.display = 'block'; } })
    .then(function(){ if (btn) btn.disabled = false; });
}
function loginStep3() {
  var code = '';
  for (var i = 0; i < 6; i++) {
    var f = document.getElementById('m' + i);
    code += f ? (f.value || '') : '';
  }
  if (!PC_CONFIG.authMfaEndpoint || code.length !== 6) {
    showToast(code.length !== 6 ? 'Informe os 6 dígitos do código.' : 'Verificação indisponível no momento.', 'err');
    return;
  }
  fetch(PC_CONFIG.authMfaEndpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, credentials: 'same-origin', body: JSON.stringify({ codigo: code }) })
    .then(function(r){ if (!r.ok) throw new Error('mfa'); return r.json().catch(function(){ return {}; }); })
    .then(function(res){
      for (var i = 0; i < 6; i++) { var f = document.getElementById('m' + i); if (f) f.value = ''; }
      if (res && res.redirect) { window.location.href = res.redirect; return; }
      pcSessao.auth = true;
      var lc2 = document.getElementById('lc2');
      var lc3 = document.getElementById('lc3');
      if (lc2) lc2.style.display = 'none';
      if (lc3) lc3.style.display = 'flex';
    })
    .catch(function(){
      for (var i = 0; i < 6; i++) { var f = document.getElementById('m' + i); if (f) f.value = ''; }
      var m0 = document.getElementById('m0'); if (m0) m0.focus();
      showToast('Código inválido ou expirado.', 'err');
    });
}

  inp.value = inp.value.replace(/[^0-9]/g,'').slice(0,1);
  if (inp.value.length === 1 && idx < 5) {

/* Máscara telefone */
function cadMaskTel(inp) {
  var v = inp.value.replace(/\D/g,'').slice(0,11);
  if (v.length>6) v = '('+v.slice(0,2)+') '+v.slice(2,7)+'-'+v.slice(7);
  else if (v.length>2) v = '('+v.slice(0,2)+') '+v.slice(2);
  inp.value = v;
}

/* Máscara CPF */
function cadMaskCPF(inp) {
  var v = inp.value.replace(/\D/g,'').slice(0,11);
  if (v.length>9) v = v.slice(0,3)+'.'+v.slice(3,6)+'.'+v.slice(6,9)+'-'+v.slice(9);
  else if (v.length>6) v = v.slice(0,3)+'.'+v.slice(3,6)+'.'+v.slice(6);
  else if (v.length>3) v = v.slice(0,3)+'.'+v.slice(3);
  inp.value = v;
}

/* Enviar lead / criar conta */
function cadEnviar() {
  var nome  = (document.getElementById('c2-nome')  ||{}).value||'';
  var tel   = (document.getElementById('c2-tel')   ||{}).value||'';
  var email = (document.getElementById('c2-email') ||{}).value||'';
  var lgpd  = document.getElementById('c2-lgpd');
  /* validação final */
  if (!nome.trim())  { cadErrField('c2-nome','Nome obrigatório'); return; }
  if (tel.replace(/\D/g,'').length<10) { cadErrField('c2-tel','WhatsApp inválido'); return; }
  if (email.indexOf('@')<1)            { cadErrField('c2-email','E-mail inválido'); return; }
  if (!lgpd||!lgpd.checked)           { showToast('Aceite os Termos e a LGPD.','warn'); return; }
  var tipoTxt = tipoLabels[selCadTipo] || selCadTipo;
  var cidade = pcValor('c2-cidade'), empresa = pcValor('c2-empresa');
  function mostrarResultado(titulo, texto, passo1, mostrarBotaoWa) {
    document.getElementById('blk-cad2').style.display    = 'none';
    document.getElementById('lc1-back').style.display    = 'none';
    document.getElementById('blk-sucesso').style.display = 'block';
    var t = document.getElementById('sucesso-tit'); if (t) t.textContent = titulo;
    var el = document.getElementById('sucesso-msg'); if (el) el.textContent = texto;
    var p1 = document.getElementById('sucesso-p1'); if (p1) p1.textContent = '\u25CF ' + passo1;
    var bw = document.getElementById('btn-sucesso-wa'); if (bw) bw.style.display = mostrarBotaoWa ? '' : 'none';
  }
  if (PC_CONFIG.leadsEndpoint) {
    var btnE = document.getElementById('btn-cad2-enviar'); if (btnE) btnE.disabled = true;
    pcEnviarLead({ tipo: 'cadastro', perfil: selCadTipo, nome: nome.trim(), whatsapp: tel, email: email.trim(), cpf: pcValor('c2-cpf'), cidade: cidade, empresa: empresa, aceite_lgpd: true, pagina: location.pathname, host: location.host, data: new Date().toISOString() })
      .then(function(){
        gaEvento('cadastro_realizado', { tipo: selCadTipo, method: 'form' });
        gaEvento('generate_lead', { value: 1, currency: 'BRL' });
        mostrarResultado('Cadastro recebido', 'Recebemos seus dados. A equipe Pacto Cred fará contato para validar o cadastro.', 'Aguarde o contato da equipe pelo WhatsApp ou e-mail informados', false);
      })
      .catch(function(){ showToast('Não foi possível enviar o cadastro agora. Nada foi registrado; tente novamente.', 'err'); })
      .then(function(){ if (btnE) btnE.disabled = false; });
    return;
  }
  /* sem backend: prepara a solicitação no WhatsApp (CPF não é enviado por mensagem) */
  pcAbrirWhatsApp(['Olá, Pacto Cred. Quero abrir minha conta.', 'Perfil: ' + tipoTxt, 'Nome: ' + nome.trim(), 'WhatsApp: ' + tel, 'E-mail: ' + email.trim(), cidade ? 'Cidade: ' + cidade : '', empresa ? 'Empresa: ' + empresa : '', 'Li e aceito os Termos e a LGPD.'].filter(Boolean), 'cadastro');
  mostrarResultado('Solicitação preparada', 'Abrimos o WhatsApp com seus dados. O cadastro só chega à Pacto Cred quando você enviar a mensagem no aplicativo.', 'Envie a mensagem que abrimos no WhatsApp', true);
}
