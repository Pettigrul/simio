<div id="otz_adunit_paragraph__1_container" class="box-otz-adunit box-otz-adunit-paragraph otz-adunit code-block div-gpt-ad-otz_adunit_paragraph__1 otzads-paragraph-loaded"><div class="otz-paragraph-sticky"><div class="otz-paragraph-separator"><span>Links Patrocinados</span></div><div id="otz_adunit_paragraph__1" class="otz_adunit_paragraph" style="min-height: 280px" data-google-query-id="CKz4h8H_o5gDFRLPoAYd3jscDw"><div id="google_ads_iframe_/22899355185/ideiadehoje.com/Ideiadehoje_Content1_20250404_0__container__" style="border: 0pt none; width: 250px; height: 250px;"></div></div></div><div class="otz-paragraph-after-spacing" style="height: 200px;width: 100%;"></div></div><p class="wp-block-paragraph paragraph-item" state="prepared" paragraph-position="1">Realize uma simulação de financiamento para veículos (Carros, Motos, Caminhão) aproveitando as taxas de juros para compra de veículos fornecidas pela BV Financeira, atualizadas diariamente em conformidade com o Banco Central. </p>



<p class="wp-block-paragraph paragraph-item" state="prepared" paragraph-position="2">Insira o montante desejado para financiamento e proceda com a simulação.</p>



<p class="wp-block-paragraph paragraph-item" state="prepared" paragraph-position="3">Esta ferramenta foi criada para combinar dados de taxas de juros de instituições financeiras com a facilidade de simular diferentes opções de parcelamento em apenas um clique. </p>



<p class="wp-block-paragraph paragraph-item" state="prepared" paragraph-position="4">É importante notar que os resultados da simulação são estimativas para referência e podem diferir do valor final devido a taxas adicionais específicas de cada banco.</p>



<p class="wp-block-paragraph paragraph-item" state="prepared" paragraph-position="5">Para uma proposta de financiamento formal, é recomendado consultar o banco que ofereça a taxa mais vantajosa. Para comentários ou sugestões, por favor, entre em contato.</p>

    function calcularParcelas() {
        var valor = parseFloat(document.getElementById('valor').value.replace(/\D/g, '')); // Remove tudo que não é dígito
        var taxaJuros = parseFloat(document.getElementById('juros').value);
        
        var prazos = [12, 24, 36, 48, 60, 72, 80];
        
        var table = '<h3>Tabela de Parcelas</h3><table><tr><th>Prazo (meses)</th><th>Valor da Parcela (R$)</th><th>Total de Juros (R$)</th><th>Total Geral (R$)</th></tr>';
        
        for (var i = 0; i < prazos.length; i++) {
            var parcela = calcularParcela(valor, taxaJuros, prazos[i]);
            var totalJuros = (parcela * prazos[i]) - valor;
            var totalGeral = valor + totalJuros;
            parcela = parcela.toLocaleString('pt-BR', {style: 'currency', currency: 'BRL'});
            totalJuros = totalJuros.toLocaleString('pt-BR', {style: 'currency', currency: 'BRL'});
            totalGeral = totalGeral.toLocaleString('pt-BR', {style: 'currency', currency: 'BRL'});
            table += '<tr><td>' + prazos[i] + '</td><td>' + parcela + '</td><td>' + totalJuros + '</td><td>' + totalGeral + '</td></tr>';
        }
        
        table += '</table>';
        
        document.getElementById('resultado').innerHTML = table;
    }
    
    function calcularParcela(valor, taxaJuros, meses) {
        var taxaJurosMensal = taxaJuros / 100;
        var parcela = valor * (taxaJurosMensal / (1 - Math.pow(1 + taxaJurosMensal, -meses)));
        return parcela;
    }
