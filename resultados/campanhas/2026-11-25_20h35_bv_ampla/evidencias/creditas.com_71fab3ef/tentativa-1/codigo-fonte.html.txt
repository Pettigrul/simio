(function () {
  var RE = /\*\*([^*]+)\*\*/;

  function parse(root) {
    var walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, null);
    var hits = [], node;
    while ((node = walker.nextNode())) {
      if (RE.test(node.nodeValue)) hits.push(node);
    }
    hits.forEach(function (text) {
      if (!text.parentNode) return;
      var frag = document.createDocumentFragment();
      var rest = text.nodeValue;
      var m;
      while ((m = rest.match(RE))) {
        if (m.index) frag.appendChild(document.createTextNode(rest.slice(0, m.index)));
        var b = document.createElement('strong');
        b.textContent = m[1];
        frag.appendChild(b);
        rest = rest.slice(m.index + m[0].length);
      }
      if (rest) frag.appendChild(document.createTextNode(rest));
      text.parentNode.replaceChild(frag, text);
    });
  }

  function init() {
    document.querySelectorAll('[class*="-nat"]').forEach(parse);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
