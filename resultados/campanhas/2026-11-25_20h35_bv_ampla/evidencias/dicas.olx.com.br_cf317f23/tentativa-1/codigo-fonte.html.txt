        function adjustFinalPosition() {
            const segmentAngle = 360 / theWheel.numSegments;
            const currentAngle = theWheel.getRotationPosition();

            // Calculate the distance to the closest segment line
            let distanceToSegmentLine = currentAngle % segmentAngle;

            if (distanceToSegmentLine < segmentAngle / 20 || distanceToSegmentLine > segmentAngle - (segmentAngle / 20)) {
                if (distanceToSegmentLine < segmentAngle / 20) {
                    theWheel.rotationAngle += (segmentAngle / 30);
                } else {
