		function clearPerformanceCookies() {
			var cookies = document.cookie.split(";");
			var domains = [".mpmg.mp.br", "www.mpmg.mp.br", window.location.hostname];

			for (var i = 0; i < cookies.length; i++) {
				var cookie = cookies[i].trim();
				var name = cookie.split("=")[0];

				// Apaga qualquer cookie do Google (GA4, UA, Ads, GTM)
				if (name.startsWith('_ga') || name.startsWith('_gid') || name.startsWith('_gat') || name.startsWith('_gcl')) {
					for (var j = 0; j < domains.length; j++) {
						document.cookie = name + "=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/; domain=" + domains[j];
						document.cookie = name + "=; Max-Age=0; path=/; domain=" + domains[j];
					}
					// Fallback para o domínio atual sem o ponto
					document.cookie = name + "=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/";
					document.cookie = name + "=; Max-Age=0; path=/";
				}
			}
		}
