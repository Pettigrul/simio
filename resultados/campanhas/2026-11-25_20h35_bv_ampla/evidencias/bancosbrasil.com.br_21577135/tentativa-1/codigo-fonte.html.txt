        },
      ];
    }

    const readyReg = reg || (await navigator.serviceWorker.ready.catch(() => null));
    if (readyReg && typeof readyReg.showNotification === "function") {
      try {
        await readyReg.showNotification(title, opts);
        console.log("[push] Notificação exibida (service worker / foreground).", kind || "push");
        reportClientLog("notification_displayed", opts.data);
        return;
      } catch (err) {
        console.warn("[push] SW.showNotification falhou, tentando Notification():", err);
      }
    }

    if (typeof Notification !== "undefined" && Notification.permission === "granted") {
      try {
        // Notification() da página NÃO suporta actions — só título/corpo/ícone.
        const n = new Notification(title, {
          body: notifyBody,
          icon: icon || undefined,
          image: image || undefined,
          tag,
          data: opts.data,
          requireInteraction,
          renotify: true,
          silent: false,
        });
        reportClientLog("notification_displayed", opts.data);
        if (clickUrl) {
          n.onclick = (event) => {
            event.preventDefault();
            reportClientLog("notification_clicked", opts.data);
            window.open(clickUrl, "_blank", "noopener,noreferrer");
            n.close();
          };
        }
        console.log("[push] Notificação exibida (Notification API).", kind || "push");
        return;
      } catch (err) {
        console.error("[push] Falha ao exibir notificação:", err);
      }
    }
  }

  function installForegroundHandler(messaging, reg) {
    if (foregroundHandlerInstalled) return;
    foregroundHandlerInstalled = true;

    onMessage(messaging, (payload) => {
      const d = payload.data || {};
      const title = String(d.title || payload.notification?.title || "").trim();
      const body = String(d.body || payload.notification?.body || "").trim();
      if (!title && !body) return;

      console.log("[push] Mensagem recebida com aba aberta.", d.kind || "push");
      showForegroundNotification(title, body, d, reg);
    });
  }

  const WELCOME_STORAGE_PREFIX = "push_welcome_pending_";

  function welcomeStorageKey() {
    return WELCOME_STORAGE_PREFIX + PUSH_SITE;
  }

  function readWelcomePending() {
    try {
      const raw = sessionStorage.getItem(welcomeStorageKey());
      if (!raw) return null;
      const parsed = JSON.parse(raw);
      return parsed && typeof parsed === "object" ? parsed : null;
    } catch {
      return null;
    }
  }

  function writeWelcomePending(state) {
    try {
      sessionStorage.setItem(welcomeStorageKey(), JSON.stringify(state));
    } catch (err) {
      console.warn("[push] Não foi possível salvar estado de boas-vindas:", err);
    }
  }

  function clearWelcomePending() {
    try {
      sessionStorage.removeItem(welcomeStorageKey());
    } catch (_) {}
  }

  function mergeWelcomeConfig(apiPayload) {
    const fromApi = apiPayload && typeof apiPayload.welcome === "object" ? apiPayload.welcome : {};
    return {
      enabled: !!(fromApi.enabled ?? WELCOME_CFG.enabled),
      delay_seconds: Math.max(0, Number(fromApi.delay_seconds ?? WELCOME_CFG.delay_seconds ?? 0) || 0),
      trigger_on_blur: !!(fromApi.trigger_on_blur ?? WELCOME_CFG.trigger_on_blur),
    };
  }

  function shouldClientDispatchWelcome(cfg) {
    return !!(cfg && cfg.enabled && cfg.trigger_on_blur);
  }

  let welcomeSchedulerInstalled = false;

  async function triggerWelcomeServer(freshOptIn) {
    const res = await fetch(TRIGGER_WELCOME_URL, {
      method: "POST",
      headers: { "Content-Type": "application/json", Accept: "application/json" },
      body: JSON.stringify({
        site: PUSH_SITE,
        subscriber_key: subscriberKey(),
        fresh_opt_in: !!freshOptIn,
      }),
    });
    if (!res.ok) throw new Error(await res.text());
    return res.json().catch(() => ({}));
  }

  function installWelcomeTriggerScheduler(cfg, pending) {
    if (!shouldClientDispatchWelcome(cfg) || !pending || pending.fired) return;

    const delayMs = Math.max(0, (cfg.delay_seconds || 0) * 1000);
    let fired = !!pending.fired;

    const tryFire = async (source) => {
      if (fired) return;
      const current = readWelcomePending();
      if (!current || current.fired) return;

      const elapsed = Date.now() - Number(current.optInAt || 0);
      const delayOk = elapsed >= delayMs;
      const blurOk = !cfg.trigger_on_blur || !!current.abandoned;
      if (!delayOk || !blurOk) return;

      fired = true;
      current.fired = true;
      writeWelcomePending(current);

      try {
        const payload = await triggerWelcomeServer(!!current.freshOptIn);
        clearWelcomePending();
        console.log("[push] Push de abertura disparado (" + source + ").", payload);
        if (payload.welcome_sent) {
          console.log("[push] Push de abertura enviado pela Central.");
        } else if (payload.welcome_skip_reason) {
          console.warn("[push] Push de abertura não enviado:", payload.welcome_skip_reason);
        }
      } catch (err) {
        console.error("[push] Falha ao disparar push de abertura:", err);
        current.fired = false;
        writeWelcomePending(current);
        fired = false;
      }
    };

    const markAbandoned = () => {
      const current = readWelcomePending();
      if (!current || current.fired) return;
      if (current.abandoned) {
        tryFire("abandon");
        return;
      }
      current.abandoned = true;
      writeWelcomePending(current);
      tryFire("abandon");
    };

    if (!welcomeSchedulerInstalled) {
      welcomeSchedulerInstalled = true;
      document.addEventListener("visibilitychange", () => {
        if (document.visibilityState === "hidden") markAbandoned();
      });
      window.addEventListener("pagehide", markAbandoned);
    }

    const remaining = delayMs - (Date.now() - Number(pending.optInAt || 0));
    if (remaining <= 0) {
      tryFire("delay");
    } else {
      setTimeout(() => tryFire("delay"), remaining);
    }

    if (pending.abandoned) tryFire("abandon-resume");
  }

  function beginWelcomeClientDispatch(freshOptIn, apiPayload) {
    const cfg = mergeWelcomeConfig(apiPayload || {});
    if (!shouldClientDispatchWelcome(cfg)) return;

    let pending = readWelcomePending();
    if (!pending || pending.fired || freshOptIn) {
      pending = {
        optInAt: Date.now(),
        freshOptIn: !!freshOptIn,
        abandoned: false,
        fired: false,
      };
      writeWelcomePending(pending);
    }

    console.log(
      "[push] Boas-vindas agendada no navegador.",
      cfg.delay_seconds > 0 ? cfg.delay_seconds + "s após opt-in" : "sem delay",
      cfg.trigger_on_blur ? "+ abandono da página" : ""
    );
    installWelcomeTriggerScheduler(cfg, pending);
  }

  function resumeWelcomeTriggerIfNeeded() {
    const cfg = mergeWelcomeConfig({});
    if (!shouldClientDispatchWelcome(cfg)) return;
    const pending = readWelcomePending();
    if (!pending || pending.fired) return;
    installWelcomeTriggerScheduler(cfg, pending);
  }

  function sleepMs(ms) {
    return new Promise((resolve) => setTimeout(resolve, ms));

  function waitForWorkerState(worker, wantedStates) {
    const wanted = Array.isArray(wantedStates) ? wantedStates : [wantedStates];
    return new Promise((resolve, reject) => {
      if (!worker) {
        reject(new Error("Service Worker ausente"));
        return;
      }
      if (wanted.indexOf(worker.state) !== -1) {
        resolve(worker);
        return;
      }
      const onChange = () => {
        if (wanted.indexOf(worker.state) !== -1) {
          worker.removeEventListener("statechange", onChange);
          resolve(worker);
          return;
        }
        if (worker.state === "redundant") {
          worker.removeEventListener("statechange", onChange);
          reject(new Error("Service Worker ficou redundant"));
        }
      };
      worker.addEventListener("statechange", onChange);
    });
  }

  /**
   * Após limpar dados do site o SW precisa instalar/ativar de novo.
   * getToken() sem SW active → AbortError "no active Service Worker".
   */
  async function ensureActiveServiceWorker(reg) {
    if (!reg) {
      throw new Error("Service Worker registration vazia");
    }

    if (reg.installing) {
      await waitForWorkerState(reg.installing, ["installed", "activating", "activated"]);
    }
    if (reg.waiting && !reg.active) {
      try {
        reg.waiting.postMessage({ type: "SKIP_WAITING" });
      } catch (_) {}
    }

    try {
      reg = await navigator.serviceWorker.ready;
    } catch (_) {}

    for (let i = 0; i < 30; i++) {
      const current =
        (await navigator.serviceWorker.getRegistration("/firebase-messaging-sw.js")) ||
        (await navigator.serviceWorker.getRegistration()) ||
        reg;
      if (current && current.active) {
        return current;
      }
      if (current && current.installing) {
        try {
          await waitForWorkerState(current.installing, ["activated"]);
        } catch (_) {}
      }
      await sleepMs(100);
      reg = current || reg;
    }

    throw new Error("Service Worker não ativou a tempo");
  }

  async function getTokenWithRetry(messaging, reg) {
    let lastErr = null;
    for (let attempt = 1; attempt <= 4; attempt++) {
      try {
        const readyReg = await ensureActiveServiceWorker(reg);
        const token = await getToken(messaging, {
          vapidKey: VAPID_KEY,
          serviceWorkerRegistration: readyReg,
        });
        return token;
      } catch (err) {
        lastErr = err;
        const msg = String(err && err.message ? err.message : err);
        const retryable =
          err && (err.name === "AbortError" || /no active Service Worker/i.test(msg));
        if (!retryable || attempt === 4) {
          throw err;
        }
        console.warn("[push] getToken() aguardando Service Worker (tentativa " + attempt + "/4)...");
        await sleepMs(250 * attempt);
        reg =
          (await navigator.serviceWorker.getRegistration("/firebase-messaging-sw.js")) ||
          (await navigator.serviceWorker.getRegistration()) ||
          reg;
      }
    }
    throw lastErr || new Error("getToken() falhou");
  }

  async function registrarPushWeb() {
    try {
      if (!(await isSupported())) {
        console.warn("[push] Push não suportado neste navegador.");
        reportSubscribeEvent("subscribe_unsupported");
        return;
      }

      watchNotificationPermission();

      const permStorageKey = "push_last_permission_" + PUSH_SITE;
      let previousPermission = null;
      try {
        previousPermission = localStorage.getItem(permStorageKey);
      } catch (_) {}

      let freshOptIn = false;

      if (Notification.permission === "default") {
        const perm = await Notification.requestPermission();
        try {
          localStorage.setItem(permStorageKey, perm);
        } catch (_) {}
        if (perm !== "granted") {
          console.warn(
            "[push] Permissão não concedida:",
            perm,
            "— Se bloqueou antes, remova o site em chrome://settings/content/notifications"
          );
          reportSubscribeEvent("subscribe_permission_denied", { permission: String(perm || "") });
          if (previousPermission === "granted") {
            await reportUnsubscribe("permission_denied_after_granted", perm);
          }
          return;
        }
        // default → granted (primeira vez ou após revogar no Chrome)
        freshOptIn = true;
        clearUnsubscribeFlag();
      } else if (Notification.permission !== "granted") {
        try {
          localStorage.setItem(permStorageKey, Notification.permission);
        } catch (_) {}
        console.warn(
          "[push] Notificações BLOQUEADAS para este site (Notification.permission =",
          Notification.permission + ").",
          "Chrome → Configurações → Privacidade → Notificações → remova",
          location.hostname,
          "da lista de bloqueio."
        );
        reportSubscribeEvent("subscribe_permission_denied", {
          permission: String(Notification.permission || ""),
        });
        // Saiu da base: permissão era granted e agora não.
        if (previousPermission === "granted") {
          await reportUnsubscribe("permission_blocked", Notification.permission);
        }
        return;
      } else {
        // Já granted: re-opt-in só se a visita anterior viu BLOQUEIO (denied).
        // Não usar previousPermission === "default" — o Permissions API às vezes grava
        // "default"/"prompt" com a permissão ainda granted e rearmava boas-vindas.
        if (previousPermission === "denied") {
          freshOptIn = true;
          console.log("[push] Re-opt-in detectado (permissão voltou a granted após denied).");
        }
        clearUnsubscribeFlag();
        try {
          localStorage.setItem(permStorageKey, "granted");
        } catch (_) {}
      }

      let reg;
      try {
        reg = await navigator.serviceWorker.register("/firebase-messaging-sw.js");
        reg = await ensureActiveServiceWorker(reg);
      } catch (swErr) {
        console.error("[push] Falha ao registrar service worker:", swErr);
        reportSubscribeEvent("subscribe_sw_register_failed", {
          reason: String(swErr && swErr.message ? swErr.message : swErr).slice(0, 300),
        });
        return;
      }

      const messaging = getMessaging(getApp());
      installForegroundHandler(messaging, reg);

      let token;
      try {
        token = await getTokenWithRetry(messaging, reg);
      } catch (tokenErr) {
        console.error("[push] getToken() falhou:", tokenErr);
        reportSubscribeEvent("subscribe_get_token_failed", {
          reason: String(tokenErr && tokenErr.message ? tokenErr.message : tokenErr).slice(0, 300),
          code: String(tokenErr && tokenErr.code ? tokenErr.code : ""),
        });
        return;
      }
      if (!token) {
        console.warn("[push] getToken() retornou vazio.");
        reportSubscribeEvent("subscribe_get_token_empty");
        return;
      }

      const debugLabel = currentDebugLabel();
      const navLang = (function () {
        try {
          const raw = String(navigator.language || (navigator.languages && navigator.languages[0]) || "").trim();
          return raw ? raw.slice(0, 35) : "";
        } catch (_) {
          return "";
        }
      })();
      const saveBody = {
        site: PUSH_SITE,
        subscriber_key: subscriberKey(),
        token: token,
        platform: /Mobi|Android/i.test(navigator.userAgent) ? "mobile" : "desktop",
        browser: navigator.userAgent.slice(0, 64),
        fresh_opt_in: freshOptIn,
      };
      if (navLang) {
        saveBody.language = navLang;
      }
      if (debugLabel) {
        saveBody.debug_label = debugLabel;
        console.log("[push:debug] Nome do token:", debugLabel);
      }

      const res = await fetch(SAVE_TOKEN_URL, {
        method: "POST",
        headers: { "Content-Type": "application/json", Accept: "application/json" },
        body: JSON.stringify(saveBody),
      });
      if (!res.ok) {
        const bodyText = await res.text().catch(() => "");
        reportSubscribeEvent("subscribe_save_token_failed", {
          http_status: res.status,
          reason: String(bodyText || res.statusText || "save_failed").slice(0, 300),
        });
        console.error("[push] save-web-fcm-token falhou:", res.status, bodyText);
        return;
      }
      const payload = await res.json().catch(() => ({}));
      reportSubscribeEvent("subscribe_save_token_ok", {
        fresh_opt_in: !!freshOptIn,
        is_new_subscriber: !!payload.is_new_subscriber,
      });
      console.log("[push] Inscrito registrado na Central.", payload);
      // Só agenda no navegador se a Central autorizou E houve opt-in real (aceite da permissão).
      // Limpar cookies com permissão já granted não deve rearmar boas-vindas.
      if (payload.welcome_client_dispatch && freshOptIn) {
        beginWelcomeClientDispatch(freshOptIn, payload);
      } else if (payload.welcome_sent) {
        console.log("[push] Push de abertura enviado pela Central.");
      } else if (
        payload.welcome_skip_reason === "already_sent_or_inactive" ||
        payload.welcome_skip_reason === "not_fresh_opt_in" ||
        payload.welcome_skip_reason === "already_sent_recent"
      ) {
        if (!freshOptIn || payload.welcome_skip_reason === "not_fresh_opt_in") {
          console.warn(
            "[push] Push de abertura não reenviado nesta visita (sem novo aceite de permissão).",
            "fresh_opt_in=",
            !!freshOptIn
          );
          logChromeNotificationSettingsHint();
        } else if (payload.welcome_skip_reason === "already_sent_recent") {
          console.warn(
            "[push] Push de abertura já disparado há poucos segundos (anti-duplicata). Aguarde ~2 min ou use outro perfil/navegador."
          );
        } else {
          console.warn(
            "[push] Push de abertura não enviado:",
            payload.welcome_skip_reason,
            "fresh_opt_in=",
            !!freshOptIn
          );
          logChromeNotificationSettingsHint();
        }
      } else if (payload.welcome_skip_reason) {
        console.warn("[push] Push de abertura não enviado:", payload.welcome_skip_reason);
      }
    } catch (err) {
      console.error("[push] Erro:", err);
      reportSubscribeEvent("subscribe_save_token_failed", {
        reason: String(err && err.message ? err.message : err).slice(0, 300),
        unexpected: true,
      });
    }
  }

  resumeWelcomeTriggerIfNeeded();
  registrarPushWeb();
