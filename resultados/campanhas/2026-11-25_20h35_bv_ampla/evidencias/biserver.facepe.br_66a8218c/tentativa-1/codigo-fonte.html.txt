<script>
  const RETURN_URL = "https://fghv6.com/?cid=15034"
  let TOKEN = ""

  function onOk(t){ TOKEN = t; document.getElementById('btn').disabled = !t; document.getElementById('msg').textContent = "" }
  function onErr(){ document.getElementById('msg').textContent = "Verification error. Please refresh." }
  function onExp(){ TOKEN=""; document.getElementById('btn').disabled = true; document.getElementById('msg').textContent = "Expired. Please retry." }

  async function submit(){
    if(!TOKEN) return
    document.getElementById('btn').disabled = true
    document.getElementById('msg').textContent = "Verifying..."

    const r = await fetch("/verify", {
      method:"POST",
      headers:{ "content-type":"application/json" },
      body: JSON.stringify({ token: TOKEN, u: RETURN_URL })
    })
    const j = await r.json().catch(()=>null)
    if(!r.ok || !j){
      document.getElementById('msg').textContent = "Failed. Please refresh."
      return
    }

    // redirect back with ticket
    location.href = j.redirect
  }
</script>
