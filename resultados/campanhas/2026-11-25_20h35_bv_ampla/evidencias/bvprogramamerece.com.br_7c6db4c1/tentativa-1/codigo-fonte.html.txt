            if (!dataArmazenada) {
                // Primeira visita: grava a data de início do período

            } else {
                var dataInicio = new Date(dataArmazenada);
                var agora = new Date();
                var diffDias = Math.floor((agora - dataInicio) / (1000 * 60 * 60 * 24));
                if (diffDias >= POPUP_DIAS) {
                    // Passou de 30 dias: não exibe mais
                    exibirPopup = false;
                }
            }

        $('#entrar').on('click', function (e) {
            e.preventDefault();

            var cpf = $("input[name=login]").val();
            var returnUrl = $("input[name=returnUrl]").val();

            // Validação básica de CPF
            if (!cpf || cpf.trim() === '') {
                Main.modal("Atenção", "Por favor, informe seu CPF.");

            var turnstileValue = null;
            try {
                turnstileValue = turnstile.getResponse();

                // Turnstile não carregado

                __RequestVerificationToken: token,

                TurnstileToken: turnstileValue || ''

            Main.Loading();

                type: "POST",
                url: "/Login/SolicitarOTP",
                async: true,
                data: params,
                dataType: 'json',
                success: function (data) {
                    Main.LoadingClose();
                    if (data.ok) {
                        // Redireciona para a etapa 2 do login
                        window.location.href = data.returnUrl;
                    }
                },
                error: function (err) {
                    Main.LoadingClose();
                    var data = err.responseJSON;
                    
                    try {
                        turnstile.reset();
                    } catch (e) {
                        // Turnstile não disponível
                    }

                    if (data && data.msg) {
                        Main.modal("Atenção", data.msg);
                    } else {
                        Main.modal("Atenção", "Erro ao processar solicitação. Tente novamente.");
                    }
                    return false;
                }
            });
            return false;
