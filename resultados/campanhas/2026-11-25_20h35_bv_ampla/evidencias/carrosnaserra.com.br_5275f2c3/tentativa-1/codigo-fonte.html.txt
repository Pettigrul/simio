    (function () {
        var carousels = document.querySelectorAll('[data-patrocinado-foto-carousel]');

        for (var i = 0; i < carousels.length; i++) {
            (function (carousel) {
                var image = carousel.querySelector('[data-patrocinado-foto]');
                var counter = carousel.querySelector('[data-patrocinado-foto-contador]');
                var previous = carousel.querySelector('[data-patrocinado-foto-prev]');
                var next = carousel.querySelector('[data-patrocinado-foto-next]');
                var cta = carousel.querySelector('[data-patrocinado-foto-cta]');
                var photos = [];
                var totalPhotos = parseInt(carousel.getAttribute('data-total-photos') || '0', 10);
                var hasCta = carousel.getAttribute('data-has-gallery-cta') === '1';

                try {
                    photos = JSON.parse(carousel.getAttribute('data-photos') || '[]');
                } catch (error) {
                    photos = [];
                }

                if (!image || !photos.length) {
                    return;
                }

                var current = 0;
                var slidesTotal = photos.length + (hasCta ? 1 : 0);
                var render = function () {
                    if (hasCta && current === photos.length) {
                        image.style.display = 'none';
                        if (previous) {
                            previous.hidden = true;
                        }
                        if (next) {
                            next.hidden = true;
                        }
                        if (cta) {
                            cta.removeAttribute('hidden');
                        }
                        if (counter) {
                            counter.innerHTML = '+ fotos · ' + totalPhotos;
                        }
                        return;
                    }

                    image.style.display = 'block';
                    if (previous) {
                        previous.hidden = false;
                    }
                    if (next) {
                        next.hidden = false;
                    }
                    if (cta) {
                        cta.setAttribute('hidden', 'hidden');
                    }
                    image.src = photos[current];
                    if (counter) {
                        counter.innerHTML = (current + 1) + ' / ' + totalPhotos;
                    }
                };
                var move = function (direction, event) {
                    event.preventDefault();
                    event.stopPropagation();
                    current = (current + direction + slidesTotal) % slidesTotal;
                    render();
                };

                if (previous) {
                    previous.addEventListener('click', function (event) { move(-1, event); });
                }
                if (next) {
                    next.addEventListener('click', function (event) { move(1, event); });
                }
            })(carousels[i]);
        }
    })();
