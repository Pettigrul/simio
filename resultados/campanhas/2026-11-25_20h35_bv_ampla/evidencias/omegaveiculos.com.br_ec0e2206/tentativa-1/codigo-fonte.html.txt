      // Mantém a opção vazia e só oferece modelos da marca escolhida.
      model.replaceChildren(new Option('Todos os modelos', ''));
      models.forEach(function (entry) {
        if (selectedBrand && entry.brand !== selectedBrand) return;
        if (seen[entry.name]) return;
        seen[entry.name] = true;
        model.add(new Option(entry.name, entry.name));
      });
      if (Array.prototype.some.call(model.options, function (option) { return option.value === previouslySelected; })) {
        model.value = previouslySelected;
      }
      // Sincroniza também o seletor visual personalizado do site.
      model.dispatchEvent(new Event('change', { bubbles: true }));
    }
    brand.addEventListener('change', updateModels);
    updateModels();
  })();
