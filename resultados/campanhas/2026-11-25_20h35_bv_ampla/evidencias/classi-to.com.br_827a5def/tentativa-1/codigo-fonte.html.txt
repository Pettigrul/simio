    function searchAuto(initial) {
        return {
            q: initial || '',
            open: false,
            results: [],
            _timer: null,
            fetchResults() {
                clearTimeout(this._timer);
                const term = this.q.trim();
                if (term.length < 2) { this.results = []; this.open = false; return; }
                this._timer = setTimeout(async () => {
                    try {
                        const res = await fetch(`/api/ads/suggestions?q=${encodeURIComponent(term)}`);
                        this.results = res.ok ? await res.json() : [];
                        this.open = this.results.length > 0;
                    } catch (e) {
                        this.results = [];
                    }
                }, 250);
            },
            pick(title) {
                this.q = title;
                this.open = false;
            }
        }
    }

    function cityAuto(initial) {
        return {
            q: initial || '',
            val: initial || '',
            open: false,
            get results() {
                const norm = s => s.normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase();
                if (!this.q) return TO_CITIES.slice(0, 8);
                return TO_CITIES.filter(c => norm(c).includes(norm(this.q))).slice(0, 12);
            },
            pick(city) {
                this.q = city;
                this.val = city;
                this.open = false;
                // carrega bairros se houver #neighborhood-wrap na página
                const wrap = document.getElementById('neighborhood-wrap');
                if (wrap) {
                    fetch(`/partials/neighborhoods?city=${encodeURIComponent(city)}`)
                        .then(r => r.text())
                        .then(html => { wrap.innerHTML = html; });
                }
            }
        }

    // Também na linha 3 dos filtros da home (dono, 23/11/2026: "mesmo movimento
    // dos Stories, fica padrão"), com `area` = a própria linha.
    function nudgeScrollRow(el, area) {
        if (!el) return;
        let stopped = false;
        const stop = () => { stopped = true; };
        // A área toda (fileira + setas ‹ › do computador): clicar numa seta
        // também para o balanço — antes ele puxava a fileira de volta e as
        // setas pareciam não funcionar (achado em 15/11/2026).
        area = area || el.closest('section') || el;
        area.addEventListener('pointerdown', stop, { once: true });
        area.addEventListener('touchstart', stop, { once: true, passive: true });
        area.addEventListener('wheel', stop, { once: true, passive: true });

        const amplitude = 14;
        const periodMs = 3400;
        let start = null;
        let base = null;  // balança a partir de onde a fileira abriu (no computador, o meio)
        function tick(now) {
            if (stopped || !el.isConnected) return;  // a grade trocou (htmx): para
            if (start === null) { start = now; base = el.scrollLeft; }
            const phase = ((now - start) % periodMs) / periodMs * Math.PI * 2;
            el.scrollLeft = base + (Math.sin(phase - Math.PI / 2) + 1) / 2 * amplitude;
            requestAnimationFrame(tick);
        }
        setTimeout(() => { if (!stopped) requestAnimationFrame(tick); }, 700);
    }

    // Fileira dos Stories infinita para os dois lados (dono, 15/11/2026). O
    // template desenha a fileira 3 vezes (data-copia 0, 1, 2); ela abre na cópia
    // do meio — no celular com o bloco principal no começo da tela, no
    // computador com ele no centro — e, quando a rolagem chega perto de uma
    // ponta, pula uma cópia inteira para o lado de lá, no mesmo ponto: para
    // quem usa, é uma fita sem fim.
    function infiniteStoryRow(el) {
        if (!el) return;
        const inicioDa = copia => Math.min(...[...el.querySelectorAll(`.story-item[data-copia="${copia}"]`)].map(a => a.offsetLeft));
        let largura = 0, meio = 0;
        function mede() {
            const c0 = inicioDa(0), c1 = inicioDa(1);
            if (!isFinite(c0) || !isFinite(c1)) return false;
            largura = c1 - c0;
            meio = c1;
            return largura > 0;
        }
        requestAnimationFrame(() => {
            if (!mede()) return;
            const desktop = window.innerWidth >= 640;
            // As duas maiores categorias (data-centro) abrem no centro da tela,
            // no computador e no celular (dono, 20/11/2026).
            const centro = [...el.querySelectorAll('.story-item[data-copia="1"][data-centro]')];
            if (centro.length) {
                const caixa = el.getBoundingClientRect();
                const rs = centro.map(a => a.getBoundingClientRect());
                const meioPar = (Math.min(...rs.map(r => r.left)) + Math.max(...rs.map(r => r.right))) / 2;
                // - 7: o balanço de convite (amplitude 14) oscila em volta do centro.
                el.scrollLeft += meioPar - (caixa.left + el.clientWidth / 2) - 7;
            } else {
                el.scrollLeft = desktop ? meio + largura / 2 - el.clientWidth / 2 : meio - 8;
            }
            let pendente = false;
            el.addEventListener('scroll', () => {
                if (pendente) return;
                pendente = true;
                requestAnimationFrame(() => {
                    pendente = false;
                    let pulo = 0;
                    if (el.scrollLeft < meio - largura / 2) pulo = largura;
                    else if (el.scrollLeft > meio + largura / 2) pulo = -largura;
                    if (!pulo) return;
                    el.scrollLeft += pulo;
                    if (arrasto) arrasto.inicio += pulo;  // arrastando com o mouse: segue do ponto novo
                });
            }, { passive: true });
            window.addEventListener('resize', mede, { passive: true });
        });

        // Computador: arrastar com o mouse (clicar e puxar), como o dedo no
        // celular. Se arrastou, o clique que termina o arraste não abre o Stories.
        let arrasto = null, arrastou = false;
        el.addEventListener('pointerdown', e => {
            if (e.pointerType !== 'mouse' || e.button !== 0) return;
            arrasto = { x: e.clientX, inicio: el.scrollLeft };
            arrastou = false;
        });
        window.addEventListener('pointermove', e => {
            if (!arrasto) return;
            const dx = e.clientX - arrasto.x;
            if (Math.abs(dx) > 5) arrastou = true;
            if (arrastou) { el.scrollLeft = arrasto.inicio - dx; e.preventDefault(); }
        });
        window.addEventListener('pointerup', () => { arrasto = null; });
        el.addEventListener('click', e => {
            if (arrastou) { e.preventDefault(); e.stopPropagation(); arrastou = false; }
        }, true);
        el.addEventListener('dragstart', e => e.preventDefault());
    }

    // Visualizador de stories por categoria (estilo WhatsApp Status /
    // Instagram Stories): abre no carretel de uma categoria e, ao acabar,
    // avança sozinho pra próxima categoria da lista — sessão contínua, só
    // fecha no fim da última ou quando a pessoa fecha manualmente.
    // ── Preferências de quem visita (12/11/2026, ver app/services/user_prefs.py) ──
    // O celular aprende com 3 frentes: Stories (tocar em "Ver anúncio" e assistir),
    // abrir anúncio (mais se ficar 20 s ou tocar no WhatsApp) e busca ("moto
    // amarela" = Motos + "amarela"). O peso cai pela metade a cada 7 dias. Só com
    // os cookies aceitos. Um resumo vai no cookie classito_pref para a página
    // principal; logado, também fica na conta (vale em outro aparelho).
    const CLASSITO_LOGGED = false;
    const PREF_KEY = 'classito_pref_v1';
    const PREF_HALF_LIFE = 7 * 86400000;
    function prefsConsent() {
        try { return localStorage.getItem('cookies_decision') === 'accepted'; } catch (e) { return false; }
    }
    function prefsLoad() {
        try { const d = JSON.parse(localStorage.getItem(PREF_KEY) || '{}'); return { cats: d.cats || {}, terms: d.terms || {} }; }
        catch (e) { return { cats: {}, terms: {} }; }
    }
    function prefsWeight(entry, now) {
        return entry ? entry.w * Math.pow(0.5, (now - entry.t) / PREF_HALF_LIFE) : 0;
    }
    function _prefsTop(map, now, n, min) {
        return Object.entries(map).map(([k, e]) => [k, prefsWeight(e, now)])
            .filter(([, w]) => w >= min).sort((a, b) => b[1] - a[1]).slice(0, n);
    }
    function prefsSave(p, sync = true) {
        const now = Date.now();
        for (const key of ['cats', 'terms']) {
            p[key] = Object.fromEntries(_prefsTop(p[key], now, 30, 0.05).map(([k]) => [k, p[key][k]]));
        }
        try { localStorage.setItem(PREF_KEY, JSON.stringify(p)); } catch (e) {}
        const summary = {
            c: Object.fromEntries(_prefsTop(p.cats, now, 4, 0.3).map(([k, w]) => [k, Math.round(w * 10) / 10])),
            t: Object.fromEntries(_prefsTop(p.terms, now, 6, 0.3).map(([k, w]) => [k, Math.round(w * 10) / 10])),
        };
        document.cookie = 'classito_pref=' + encodeURIComponent(JSON.stringify(summary)) + '; path=/; max-age=2592000; samesite=lax';
        if (sync && CLASSITO_LOGGED) {
            clearTimeout(window._prefsSyncTimer);
            window._prefsSyncTimer = setTimeout(() => {
                fetch('/api/users/me/prefs', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(p) }).catch(() => {});
            }, 2000);
        }
    }
    function prefsAdd(kind, key, amount) {
        if (!key || !prefsConsent()) return;
        key = String(key).normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim();
        if (!key) return;
        const p = prefsLoad();
        const map = kind === 'cat' ? p.cats : p.terms;
        const now = Date.now();
        map[key] = { w: Math.min(prefsWeight(map[key], now) + amount, 50), t: now };
        prefsSave(p);
    }
    function prefsFromSearch(sp) {
        if (!sp) return;
        prefsAdd('cat', sp.cat, 3);
        (sp.terms || []).forEach(t => prefsAdd('term', t, 3));
    }
    function prefsClear(reload) {
        try { localStorage.removeItem(PREF_KEY); } catch (e) {}
        document.cookie = 'classito_pref=; path=/; max-age=0';
        const done = () => { if (reload) location.reload(); };
        if (CLASSITO_LOGGED) fetch('/api/users/me/prefs', { method: 'DELETE' }).catch(() => {}).finally(done);
        else done();
    }
    // Logado: junta uma vez por sessão o que está na conta com o que o celular aprendeu.
    (function prefsMergeAccount() {
        if (!CLASSITO_LOGGED || !prefsConsent()) return;
        try { if (sessionStorage.getItem('classito_pref_sync')) return; sessionStorage.setItem('classito_pref_sync', '1'); } catch (e) {}
        fetch('/api/users/me/prefs').then(r => r.ok ? r.json() : {}).then(srv => {
            const p = prefsLoad();
            const now = Date.now();
            for (const key of ['cats', 'terms']) {
                for (const [k, e] of Object.entries((srv && srv[key]) || {})) {
                    if (!p[key][k] || prefsWeight(e, now) > prefsWeight(p[key][k], now)) p[key][k] = e;
                }
            }
            prefsSave(p);
        }).catch(() => {});
    })();

    // Vitrine: conta quando um cartão de anúncio APARECE na tela (pedido do dono
    // em 15/11/2026 — o anunciante compara com o Instagram, onde "visualização" é
    // aparecer, não abrir). Na grade a pessoa vê vários de uma vez ("no atacado"),
    // então só conta quando ela PAROU para olhar: a tela fica parada 3 segundos
    // com o cartão INTEIRO visível (90%: o dono viu que o celular dele mostra 6
    // inteiros; os cortados embaixo não contam); qualquer rolagem zera. Não conta
    // com os Stories abertos por cima nem com a aba escondida.
    // Cada anúncio vai uma vez por dia neste celular; o servidor também confere
    // (ver AdImpression e /api/impressions em app/main.py). Envia em lotes.
    (function () {
        if (!('IntersectionObserver' in window)) return;
        const KEY = 'classito_vitrine_' + new Date(Date.now() - 3 * 3600e3).toISOString().slice(0, 10);
        let sent;
        try { sent = new Set(JSON.parse(localStorage.getItem(KEY) || '[]')); } catch (e) { sent = new Set(); }
        try {
            Object.keys(localStorage).forEach(k => { if (k.startsWith('classito_vitrine_') && k !== KEY) localStorage.removeItem(k); });
        } catch (e) {}
        const timers = new Map();
        // Fila por origem: o cartão pode dizer de onde é (data-origem="chat" nas
        // respostas do chat, 20/11/2026); sem isso, vale a página.
        const filas = {};
        function origem() {
            const p = location.pathname, q = new URLSearchParams(location.search);
            if (p.startsWith('/u/') || p.startsWith('/perfil') || p.startsWith('/loja')) return 'perfil';
            if (q.get('q') || p.startsWith('/busca')) return 'busca';
            if (q.get('category')) return 'categoria';
            return p === '/' ? 'inicio' : 'outro';
        }
        function envia() {
            Object.keys(filas).forEach(org => {
                const ids = filas[org].splice(0);
                if (!ids.length) return;
                const body = JSON.stringify({ ad_ids: ids, origem: org });
                if (navigator.sendBeacon) navigator.sendBeacon('/api/impressions', new Blob([body], { type: 'application/json' }));
                else fetch('/api/impressions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body, keepalive: true }).catch(() => {});
            });
        }
        function marca(id, org) {
            if (sent.has(id)) return;
            sent.add(id);
            const o = org || origem();
            (filas[o] = filas[o] || []).push(id);
            try { localStorage.setItem(KEY, JSON.stringify([...sent].slice(-3000))); } catch (e) {}
            if (filas[o].length >= 30) envia();
        }
        const PARADO_MS = 3000;
        const visiveis = new Set();   // cartões com metade ou mais na tela agora
        let paradoTimer = null;
        function conta() {
            paradoTimer = null;
            if (document.visibilityState !== 'visible') return;
            if (document.body.style.overflow === 'hidden') return;  // Stories (ou outra tela) por cima
            visiveis.forEach(el => {
                const id = +el.dataset.vitrine;
                if (id) marca(id, el.dataset.origem);
                io.unobserve(el);
                visiveis.delete(el);
            });
        }
        function recomeca() {
            clearTimeout(paradoTimer);
            paradoTimer = visiveis.size ? setTimeout(conta, PARADO_MS) : null;
        }
        // Rolar a página zera os 3 segundos. (Só a página: a fileira dos Stories
        // balança sozinha e não pode impedir a contagem da grade.)
        window.addEventListener('scroll', recomeca, { passive: true });
        const io = new IntersectionObserver(entries => {
            entries.forEach(e => {
                const id = +e.target.dataset.vitrine;
                if (!id || sent.has(id)) { io.unobserve(e.target); visiveis.delete(e.target); return; }
                if (e.isIntersecting && e.intersectionRatio >= 0.9) visiveis.add(e.target);
                else visiveis.delete(e.target);
            });
            if (!paradoTimer) recomeca();
        }, { threshold: [0, 0.9] });
        function observa() {
            document.querySelectorAll('[data-vitrine]').forEach(el => {
                if (!el._vitrine) { el._vitrine = 1; io.observe(el); }
            });
        }
        // Cartões chegam depois também ("carregar mais", busca, troca de filtro).
        let espera = null;
        new MutationObserver(() => { clearTimeout(espera); espera = setTimeout(observa, 300); })
            .observe(document.documentElement, { childList: true, subtree: true });
        if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', observa); else observa();
        setInterval(envia, 5000);
        document.addEventListener('visibilitychange', () => { if (document.visibilityState === 'hidden') envia(); });
        window.addEventListener('pagehide', envia);
    })();

    // Intenção (dono, 15/11/2026): cliques que mostram o que a pessoa quer —
    // tocar numa bolinha dos Stories, "Ver anúncio", trocar de categoria
    // arrastando, WhatsApp. Deixar rodar, pular ou fechar NÃO contam. A última
    // intenção filtra a grade de anúncios (só nesta visita: sessionStorage) e
    // também reforça a preferência guardada (prefsAdd, peso maior).
    const INTENCAO_KEY = 'classito_intencao';
    function intencoes() {
        try { return JSON.parse(sessionStorage.getItem(INTENCAO_KEY) || '[]'); } catch (e) { return []; }
    }
    function registrarIntencao(slug, nome, peso) {
        if (!slug) return;
        const antes = intencoes();
        const mesma = antes[0] && antes[0].slug === slug ? antes[0] : null;
        const lista = antes.filter(i => i.slug !== slug);
        // Mesma categoria de novo: mantém o refinamento (Aluga, Casa...); outra: começa limpa.
        lista.unshift(mesma ? { ...mesma, nome: nome || mesma.nome } : { slug, nome: nome || slug });
        try { sessionStorage.setItem(INTENCAO_KEY, JSON.stringify(lista.slice(0, 4))); } catch (e) {}
        try { prefsAdd('cat', slug, peso || 2); } catch (e) {}
        window.dispatchEvent(new CustomEvent('intencao-mudou', { detail: { slug } }));
    }
    // Recarrega a grade da tela inicial pela última intenção (90% da categoria +
    // 10% convidados, ver home() em app/main.py). Só na tela inicial sem
    // filtro/busca; a URL não muda (o filtro vale só nesta visita).
    function aplicarIntencao(rolar) {
        const alvo = document.getElementById('results-section');
        const params = new URLSearchParams(location.search);
        params.delete('intencao');
        if (!alvo || !window.htmx || location.pathname !== '/' || [...params].length) return;
        if (location.search) history.replaceState(history.state, '', '/');
        const it = intencoes()[0];
        let url = '/';
        if (it) {
            url = '/?intencao=' + encodeURIComponent(it.slug);
            if (it.natureza) url += '&natureza=' + encodeURIComponent(it.natureza);
            if (it.tipo) url += '&tipo=' + encodeURIComponent(it.tipo);
            if (it.ordem) url += '&sort=' + encodeURIComponent(it.ordem);
        }
        htmx.ajax('GET', url, { target: '#results-section', swap: 'innerHTML' }).then(() => {
            if (rolar) rolarAteFiltros();
        });
    }
    // Parada da tela ao escolher categoria (dono, 23/11/2026): a busca fixa em

    function rolarAteFiltros() {
        const alvo = document.getElementById('results-section');
        const barra = document.querySelector('section.sticky');
        const nome = [...document.querySelectorAll('.story-nome')].find(n => n.offsetParent);
        const ref = nome || alvo;
        if (!ref) return;
        const destino = () => {
            const fundo = barra ? barra.getBoundingClientRect().bottom : 140;
            return Math.max(0, Math.round(ref.getBoundingClientRect().top + window.scrollY - fundo - 6));
        };
        window.scrollTo({ top: destino(), behavior: 'smooth' });
        setTimeout(() => {
            const t = destino();
            if (Math.abs(t - window.scrollY) > 4) window.scrollTo({ top: t, behavior: 'smooth' });
        }, 550);
    }
    // Botões de refinar (Vende/Aluga, Casa/Apartamento...): ajusta a intenção
    // atual e recarrega a grade. '' = "Todos". Também é clique de intenção.
    function refinarIntencao(campos, recarregar = true) {

        if (!lista.length) return;
        lista[0] = { ...lista[0], ...campos };
        try { sessionStorage.setItem(INTENCAO_KEY, JSON.stringify(lista)); } catch (e) {}
        try { prefsAdd('cat', lista[0].slug, 1); } catch (e) {}
        if (recarregar) aplicarIntencao(false);
    }
    function limparIntencao() {

    const STORY_SEEN_KEY = 'classito_story_vistos';
    function storySeenIds() {

    function rememberStorySeen(id) {

        } catch (e) {}
    }
    function storyPhoto(ad) {
        return ad.story_photo || (ad.images && ad.images.length ? ad.images[0].url : null);
    }

    function categoryStories(categoryList) {
        return {
            categories: categoryList || [],
            open: false,
            catIndex: 0,
            storyAds: [],
            storyIndex: 0,
            progress: 0,
            loading: false,
            paused: false,
            _timer: null,
            _touchStartY: 0,
            _touchStartX: 0,
            _historyPushed: false,
            _storySeen: new Set(),
            _storyFlushBound: false,
            voltaCompleta: false,   // aviso "Você viu todos" ao recomeçar a categoria
            _pool: [],              // todos os anúncios da categoria, já na ordem do Stories
            filtroNatureza: '',     // botões de refinar dentro do Stories (Vende/Aluga...)
            filtroTipo: '',         // (Casa/Apartamento, Hatch/SUV...) — ver app/services/refino.py
            _swiped: false,         // arrastou para o lado: o toque não avança anúncio

            get currentAd() {
                return this.storyAds[this.storyIndex] || null;
            },
            get currentCategory() {
                return this.categories[this.catIndex] || null;
            },

            async openSession(slug) {
                const idx = this.categories.findIndex(c => c.slug === slug);
                if (idx === -1) return;
                this.catIndex = idx;
                registrarIntencao(slug, this.categories[idx].name);
                this.open = true;
                document.body.style.overflow = 'hidden';
                if (!this._storyFlushBound) {
                    // "Ver anúncio" sai da página sem passar por close().
                    this._storyFlushBound = true;
                    window.addEventListener('pagehide', () => this._flushStoryViews());
                    document.addEventListener('visibilitychange', () => {
                        if (document.visibilityState === 'hidden') this._flushStoryViews();
                    });
                }
                // Espera a story (tela preta) pintar DE VERDADE antes de
                // empurrar a entrada de histórico — confirmado com o usuário
                // (app instalado/PWA, modo standalone): o Android tira uma
                // "foto" da tela pro gesto nativo de voltar (Predictive Back,
                // Android 13+) no momento em que a entrada de histórico é
                // criada. Sem esse atraso, open=true e pushState rodavam no
                // mesmo tick, antes do navegador pintar o primeiro frame da
                // story — a foto capturada era da página de baixo (rolada),
                // e era ELA que aparecia rapidinho durante o gesto de voltar.
                // Dois requestAnimationFrame = espera o navegador de fato
                // pintar (1º agenda o frame, 2º confirma que o anterior já
                // foi desenhado), não só o Alpine aplicar a mudança no DOM.
                await new Promise(resolve => requestAnimationFrame(() => requestAnimationFrame(resolve)));
                // Empurra uma entrada de histórico só pra story abrir — sem
                // isso, o botão/gesto de "voltar" do celular não tinha
                // NENHUMA entrada própria da story pra consumir, então saía
                // direto pra qualquer página que já estivesse antes na aba
                // em vez de só fechar a story e voltar pra home (achado
                // real do usuário). _onPopState() fecha quando "voltar" é
                // apertado com a story aberta.
                history.pushState({ classitoStory: true }, '', location.href);
                this._historyPushed = true;
                await this._loadCategoryReel();
            },

            close() {
                this.open = false;
                this._stopTimer();
                this._flushStoryViews();
                document.body.style.overflow = '';
                // A grade embaixo segue o que ela escolheu. Depois do history.back()
                // logo abaixo terminar — senão o "voltar" desfaz a troca da grade.
                setTimeout(() => aplicarIntencao(true), 400);
                if (this.$refs.storyVideo) this.$refs.storyVideo.pause();
                // Fechado pelo X/swipe/fim da última categoria (não pelo
                // "voltar" do celular) — a entrada de histórico empurrada em
                // openSession() ainda está lá, "cancela" ela com history.back()
                // pra não sobrar um passo de "voltar" fantasma depois.
                if (this._historyPushed) {
                    this._historyPushed = false;
                    history.back();
                }
            },

            // Chamado quando o "voltar" do celular (ou do navegador) dispara
            // popstate — o próprio gesto JÁ consumiu a entrada de histórico,
            // então não chama history.back() de novo (senão voltaria um
            // passo extra, pulando a home).
            //
            // ACHADO REAL (voltou a piscar mesmo depois do atraso via
            // requestAnimationFrame em openSession()): a story SEMPRE abre
            // por cima da própria home, no MESMO documento — não existe
            // "página de baixo" diferente pra recarregar. O recarregamento
            // completo ("location.href = '/'") que tínhamos aqui é, sozinho,
            // o motivo do pisca: todo reload de página tem um instante de
            // tela branca/sem estilo antes de pintar de novo — e pior ainda
            // com a internet lenta da região. Não tem gesto nativo que
            // resolva isso, porque o problema nem é o gesto: é recarregar
            // quando não precisava. Se a home por baixo já está limpa (sem
            // busca/filtro na URL — o caso relatado), fechar é só esconder a
            // story e subir o scroll, sem nenhuma navegação. Só quando tem
            // filtro/busca ativo (conteúdo de baixo é outro de verdade) que
            // ainda recarrega "/" do zero.
            _onPopState() {
                if (this.open) {
                    this._historyPushed = false;
                    this._stopTimer();
                    if (this.$refs.storyVideo) this.$refs.storyVideo.pause();
                    if (location.pathname === '/' && !location.search) {
                        this.open = false;
                        document.body.style.overflow = '';
                        if (intencoes().length) aplicarIntencao(true); else window.scrollTo(0, 0);
                        return;
                    }
                    try { sessionStorage.setItem('classito_scroll_top', '1'); } catch (e) {}
                    window.location.href = '/';
                }
            },

            // Mantém o <video> da story em sincronia com currentAd — troca o
            // src só quando muda de fato (evita reiniciar o vídeo em todo
            // tick reativo do Alpine) e tenta tocar; autoplay com som bloqueado
            // pelo navegador cai no catch, silencioso (o <video> já é muted).
            _syncStoryVideo() {
                const ad = this.currentAd;
                this.$nextTick(() => {
                    const v = this.$refs.storyVideo;
                    if (!v) return;
                    const url = ad && ad.video_file_url;
                    if (!url) {
                        v.pause();
                        v.removeAttribute('src');
                        return;
                    }
                    if (v.getAttribute('src') !== url) {
                        v.src = url;
                        v.load();
                    }
                    if (!this.paused) v.play().catch(() => {});
                });
                this._prefetchNextVideo();
            },

            // Sem preload nenhum, o vídeo só começava a baixar no exato
            // instante em que o slide virava vídeo — dava um engasgo visível
            // antes de tocar (photos já tinham isso via _preloadImages, vídeo
            // não). Só adianta o PRÓXIMO (não o pool inteiro, como as fotos —
            // vídeo é pesado demais pra isso) e passa pelo mesmo service
            // worker que já cacheia /static/* (ver sw.js), então quando o
            // <video> real pedir essa URL já sai do cache, instantâneo.
            _prefetchNextVideo() {
                if (!window.shouldAutoplayVideo()) return;
                const next = this.storyAds[this.storyIndex + 1];
                if (next && next.video_file_url) {
                    fetch(next.video_file_url).catch(() => {});
                }
            },

            async _loadCategoryReel() {
                this.storyIndex = 0;
                this.progress = 0;
                this.storyAds = [];
                const cat = this.currentCategory;
                if (!cat) { this.close(); return; }
                // Reusa o pool da PRÓXIMA categoria se já tiver sido adiantado
                // (ver _prefetchNextCategory) — sem isso, toda troca de
                // categoria mostrava o spinner de carregando de novo, mesmo a
                // lista de categorias sendo conhecida o tempo todo e dando
                // tempo de sobra (a duração da categoria atual) pra já ter
                // buscado. Guarda a Promise, não só o resultado, pra conseguir
                // reusar mesmo se a pessoa avançar rápido demais e a busca
                // ainda estiver em andamento (ainda economiza tempo).
                const prefetch = (this._nextCatPrefetch && this._nextCatPrefetch.slug === cat.slug) ? this._nextCatPrefetch : null;
                this._nextCatPrefetch = null;
                if (!prefetch || !prefetch.ready) this.loading = true;
                try {
                    const pool = prefetch ? await prefetch.promise : await this._fetchCategoryPool(cat.slug);
                    // Todos os anúncios da categoria (antes eram só 10 e o Stories
                    // pulava sozinho para outra categoria — dono, 15/11/2026).
                    this._pool = this._orderStory(pool, cat.ad_id);
                    // Refinamento que a pessoa já escolheu nesta categoria (grade ou Stories).
                    const it = intencoes()[0];
                    this.filtroNatureza = it && it.slug === cat.slug ? (it.natureza || '') : '';
                    this.filtroTipo = it && it.slug === cat.slug ? (it.tipo || '') : '';
                    this.storyAds = this._filtraPool();
                } catch (e) {
                    this.storyAds = [];
                }
                this.loading = false;
                if (this.storyAds.length === 0) {
                    this._advanceCategory();
                    return;
                }
                this._preloadImages();
                this._startTimer();
                this._prefetchNextCategory();
            },

            // Busca um pool maior ordenado por mais visualizado (proxy de
            // "anúncio mais atraente", já que não existe pontuação de
            // qualidade de foto) e embaralha — assim a sequência muda a cada
            // vez que a story é aberta, sempre puxando dos anúncios com mais
            // engajamento da categoria.
            _fetchCategoryPool(slug) {
                return fetch(`/api/ads/stories?category=${encodeURIComponent(slug)}`)
                    .then(res => res.ok ? res.json() : [])
                    .catch(() => []);
            },

            _prefetchNextCategory() {
                const next = this.categories[this.catIndex + 1];
                if (!next) return;
                const entry = { slug: next.slug, ready: false, promise: null };
                entry.promise = this._fetchCategoryPool(next.slug).then(pool => { entry.ready = true; return pool; });
                this._nextCatPrefetch = entry;
            },

            // Pré-carrega as fotos do pool inteiro assim que a story abre —
            // sem isso, cada troca de slide só começava a baixar a foto
            // naquele instante, travando visivelmente numa conexão mais lenta.
            // Vídeo não entra aqui: preload="auto" no próprio <video> já
            // cuida disso, e baixar vídeo antecipado desperdiçaria dado móvel.
            // Em economia de dados, pula o preload adiantado — carrega só o
            // slide atual mesmo, do jeito normal (ver window.shouldAutoplayVideo).
            // Só as PRÓXIMAS 3 fotos (12/11/2026): antes baixava as 8 de uma vez ao
            // abrir, pesado na internet do Tocantins. Chamado de novo a cada avanço.
            _preloadImages() {
                if (!window.shouldAutoplayVideo()) return;
                this.storyAds.slice(this.storyIndex + 1, this.storyIndex + 4).forEach(ad => {
                    if (ad.video_file_url) return;
                    const url = storyPhoto(ad);
                    if (url) { const img = new Image(); img.src = url; }
                });
            },

            // Regra dos Stories (usuário, 12/11/2026): o 1º slide é o anúncio da
            // CAPA (tocou no carro, abre no carro); depois metade novidades (até
            // 7 dias no ar) e metade sorteio da categoria, intercalados
            // (nova, sorteada, nova...). Nas duas metades vem antes o que a
            // pessoa ainda não viu; novidades com foto de nota maior antes;
            // cartaz/print (nota < 3) no fim do sorteio. Faltou de um lado,
            // o outro completa. Viu todos da categoria: o rodízio recomeça.
            _orderStory(pool, coverAdId) {
                let seen = storySeenIds();
                if (pool.some(ad => !ad.vendido) && pool.every(ad => ad.vendido || seen.has(ad.id))) {
                    forgetStorySeen(pool.map(ad => ad.id));
                    seen = new Set();
                }
                const cover = pool.find(ad => ad.id === coverAdId);
                // Vendido/alugado (carimbo, prova de venda — dono, 20/11/2026): fora
                // da ordem normal; entra depois, 1 a cada 10 e nunca nos 4 primeiros.
                // Sorteio com peso: o vendido mais recente tem mais chance de vir antes.
                const vendidos = pool.filter(ad => ad.vendido && ad !== cover)
                    .map(ad => [Math.pow(Math.random(), 1 / (ad.vendido_peso || 1)), ad])
                    .sort((a, b) => b[0] - a[0]).map(par => par[1]);
                const rest = pool.filter(ad => ad !== cover && !ad.vendido);
                const nota = ad => (ad.nota === null || ad.nota === undefined) ? 5 : ad.nota;
                const unseenFirst = list => {
                    const shuffled = this._shuffle(list);
                    return shuffled.filter(ad => !seen.has(ad.id)).concat(shuffled.filter(ad => seen.has(ad.id)));
                };
                const fresh = unseenFirst(rest.filter(ad => ad.is_new))
                    .sort((a, b) => (seen.has(a.id) - seen.has(b.id)) || (nota(b) - nota(a)));
                // No sorteio, quem está parado (terço menos visto da categoria, ver
                // home_mix.stuck_ids) vem antes — pedido do dono em 15/11/2026.
                const others = unseenFirst(rest.filter(ad => !ad.is_new))
                    .sort((a, b) => (seen.has(a.id) - seen.has(b.id)) || ((nota(a) < 3) - (nota(b) < 3))
                                    || (!!b.parado - !!a.parado));
                const out = cover ? [cover] : [];
                while (fresh.length || others.length) {
                    if (fresh.length) out.push(fresh.shift());
                    if (others.length) out.push(others.shift());
                }
                if (!vendidos.length) return out;
                const comVendidos = [];
                for (let i = 0; i < out.length; i += 9) {
                    const bloco = out.slice(i, i + 9);
                    if (vendidos.length && bloco.length >= 4) {
                        const pos = 4 + Math.floor(Math.random() * (bloco.length - 3));
                        bloco.splice(pos, 0, vendidos.shift());
                    }
                    comVendidos.push(...bloco);
                }
                return comVendidos;
            },

            // Trocar de categoria é escolha da pessoa (arrastar para o lado ou os
            // botões ‹ › no computador) e conta como intenção. As categorias
            // também giram em volta: depois da última vem a primeira.
            changeCategory(passo) {
                const cat = this.currentCategory;
                if (cat) markCategoryStorySeen(cat.id, cat.ad_id);
                this._stopTimer();
                if (!this.categories.length) { this.close(); return; }
                this.catIndex = (this.catIndex + passo + this.categories.length) % this.categories.length;
                const nova = this.currentCategory;
                registrarIntencao(nova.slug, nova.name);
                this._loadCategoryReel();
            },

            _advanceCategory() {
                // Categoria sem anúncio para mostrar: segue para a próxima.
                this.changeCategory(1);
            },

            next() {
                this._stopTimer();
                if (this.storyIndex < this.storyAds.length - 1) {
                    this.storyIndex++;
                    this.progress = 0;
                    this._preloadImages();
                    this._startTimer();
                } else {
                    // Viu todos da categoria: recomeça nela mesma (infinito).
                    const cat = this.currentCategory;
                    if (cat) markCategoryStorySeen(cat.id, cat.ad_id);
                    this.voltaCompleta = true;
                    setTimeout(() => { this.voltaCompleta = false; }, 2200);
                    this.storyIndex = 0;
                    this.progress = 0;
                    this._preloadImages();
                    this._startTimer();
                }
            },

            // ── Botões de refinar dentro do Stories ──
            _filtraPool() {
                const lista = this._pool.filter(ad =>
                    (!this.filtroNatureza || ad.ad_type === this.filtroNatureza) &&
                    (!this.filtroTipo || ad.tipo === this.filtroTipo));
                return lista.length ? lista : this._pool;
            },
            // Mesmas regras da grade: conta cada botão respeitando o outro filtro,
            // some com zero, e uma linha com uma opção só não aparece.
            get botoesStory() {
                const cfg = this.currentCategory && this.currentCategory.refino;
                if (!cfg) return { grupo: [], naturezas: [], tipos: [] };
                const nat = {}, tip = {};
                for (const ad of this._pool) {
                    if (ad.vendido) continue;  // contagem é do que ainda está à venda/para alugar
                    if (!this.filtroTipo || ad.tipo === this.filtroTipo) nat[ad.ad_type] = (nat[ad.ad_type] || 0) + 1;
                    if (ad.tipo && cfg.tipos[ad.tipo] && (!this.filtroNatureza || ad.ad_type === this.filtroNatureza)) tip[ad.tipo] = (tip[ad.tipo] || 0) + 1;
                }
                const monta = (cont, rotulos, ativo) => {
                    const out = Object.entries(cont).filter(([, n]) => n > 0).sort((a, b) => b[1] - a[1])
                        .map(([id, n]) => ({ id, n, label: rotulos[id] || id, ativo: id === ativo }));
                    return out.length >= 2 || ativo ? out : [];
                };
                return {
                    grupo: cfg.grupo || [],
                    naturezas: monta(nat, cfg.naturezas || {}, this.filtroNatureza),
                    tipos: cfg.campo ? monta(tip, cfg.tipos || {}, this.filtroTipo) : [],
                };
            },
            refinaStory(campo, valor) {
                if (campo === 'natureza') this.filtroNatureza = this.filtroNatureza === valor ? '' : valor;
                else this.filtroTipo = this.filtroTipo === valor ? '' : valor;
                const cat = this.currentCategory;
                if (cat) {
                    registrarIntencao(cat.slug, cat.name);
                    refinarIntencao({ natureza: this.filtroNatureza, tipo: this.filtroTipo }, false);
                }
                this.storyAds = this._filtraPool();
                this.storyIndex = 0;
                this.progress = 0;
                this._preloadImages();
                this._startTimer();
            },
            irParaCategoria(slug) {
                const i = this.categories.findIndex(c => c.slug === slug);
                if (i === -1 || i === this.catIndex) return;
                this.changeCategory(i - this.catIndex);
            },

            verAnuncio() {
                const ad = this.currentAd;
                if (ad && ad.category) registrarIntencao(ad.category.slug, ad.category.name, 3);
                this._flushStoryViews();
            },

            // Barrinhas do topo: no máximo 10 de cada vez (a categoria pode ter
            // 80 anúncios); mostra o bloco de 10 em que a pessoa está.
            get barras() {
                const inicio = Math.floor(this.storyIndex / 10) * 10;
                return this.storyAds.slice(inicio, inicio + 10).map((ad, k) => ({ ad, i: inicio + k }));
            },

            prev() {
                if (this.storyIndex > 0) {
                    this._stopTimer();
                    this.storyIndex--;
                    this.progress = 0;
                    this._startTimer();
                }
            },

            pause() {
                this.paused = true;
                if (this.$refs.storyVideo) this.$refs.storyVideo.pause();
            },

            resume() {
                this.paused = false;
                if (this.$refs.storyVideo && this.currentAd && this.currentAd.video_file_url) {
                    this.$refs.storyVideo.play().catch(() => {});
                }
            },

            _startTimer() {
                this._stopTimer();
                const stepMs = 50;
                const durationMs = 4500;
                const viewAfterMs = 3000;  // 3 s na tela = visualização, igual ao Instagram (dono, 15/11/2026)
                this._timer = setInterval(() => {
                    if (this.paused) return;
                    this.progress += (stepMs / durationMs) * 100;
                    if (this.currentAd && !this.currentAd.vendido && this.progress >= (viewAfterMs / durationMs) * 100
                            && !this._storySeen.has(this.currentAd.id)) {
                        this._storySeen.add(this.currentAd.id);
                        rememberStorySeen(this.currentAd.id);
                        prefsAdd('cat', this.currentAd.category && this.currentAd.category.slug, 0.5);
                    }
                    if (this.progress >= 100) this.next();
                }, stepMs);
            },

            _stopTimer() {
                if (this._timer) { clearInterval(this._timer); this._timer = null; }
            },

            // Um envio só, ao sair dos Stories: a internet da região é lenta.
            _flushStoryViews() {
                if (!this._storySeen.size) return;
                const body = new Blob([JSON.stringify({ ad_ids: [...this._storySeen] })], { type: 'application/json' });
                this._storySeen.clear();
                try { navigator.sendBeacon('/api/stories/views', body); } catch (e) {}
            },

            onTouchStart(e) {
                this._touchStartY = e.touches[0].clientY;
                this._touchStartX = e.touches[0].clientX;
                this.pause();
            },

            onTouchEnd(e) {
                const dy = e.changedTouches[0].clientY - this._touchStartY;
                const dx = Math.abs(e.changedTouches[0].clientX - this._touchStartX);
                if (dy > 80 && dx < 60) {
                    this.close();
                    return;
                }
                const dxs = e.changedTouches[0].clientX - this._touchStartX;
                if (Math.abs(dxs) > 70 && Math.abs(dy) < 60) {
                    this._swiped = true;  // o toque que termina o arraste não avança anúncio
                    setTimeout(() => { this._swiped = false; }, 350);
                    this.changeCategory(dxs < 0 ? 1 : -1);
                    this.paused = false;
                    return;
                }
                this.resume();
            },

            formatPrice(ad) {
                if (!ad || ad.price == null) return 'Gratuito';
                return window.formatCardPrice(ad.price);
            },

            _shuffle(arr) {
                const a = arr.slice();
                for (let i = a.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [a[i], a[j]] = [a[j], a[i]];
                }
                return a;
            },
        };
    }

    function aiSearch() {
        return {
            chatHistory: [],
            chatMessages: [],
            chatLoading: false,
            // Última pergunta mandada pra IA (digitada ou de um chip) — usada
            // só pra personalizar o indicador de "buscando..." (ver
            // index.html), em vez de um "alta demanda"/genérico sem relação
            // com o que a pessoa pediu.
            lastQuery: '',
            chatInput: '',
            started: false,
            uiConfig: null,
            detectedCity: null,
            isDesktop: window.matchMedia('(min-width: 640px)').matches,
            installNudgeShown: false,
            animatedPlaceholder: 'O que você procura hoje?',
            _placeholderPhrases: [
                'O que você procura hoje?',
                'Carro branco em Palmas...',
                'Imóvel pra alugar...',
                'Vaga de emprego...',
                'iPhone usado barato...',
                'Sofá retrátil...',
                'Moto até R$10.000...',
            ],
            _placeholderTimer: null,

            // Efeito "máquina de escrever" no placeholder — chama atenção pro
            // campo vazio sem travar quem já está digitando (@focus desliga).
            // Roda só enquanto o campo está vazio e a conversa não começou.
            startPlaceholderAnimation() {
                if (this._placeholderTimer) return;
                let phraseIdx = 0;
                let charIdx = 0;
                let deleting = false;
                const tick = () => {
                    if (this.chatInput.trim() !== '' || this.chatMessages.length > 0) {
                        this.stopPlaceholderAnimation();
                        return;
                    }
                    const phrase = this._placeholderPhrases[phraseIdx];
                    if (!deleting) {
                        charIdx++;
                        this.animatedPlaceholder = phrase.slice(0, charIdx);
                        if (charIdx === phrase.length) {
                            deleting = true;
                            this._placeholderTimer = setTimeout(tick, 1600);
                            return;
                        }
                        this._placeholderTimer = setTimeout(tick, 55);
                    } else {
                        charIdx--;
                        this.animatedPlaceholder = phrase.slice(0, charIdx);
                        if (charIdx === 0) {
                            deleting = false;
                            phraseIdx = (phraseIdx + 1) % this._placeholderPhrases.length;
                            this._placeholderTimer = setTimeout(tick, 300);
                            return;
                        }
                        this._placeholderTimer = setTimeout(tick, 25);
                    }
                };
                this._placeholderTimer = setTimeout(tick, 400);
            },

            stopPlaceholderAnimation() {
                if (this._placeholderTimer) {
                    clearTimeout(this._placeholderTimer);
                    this._placeholderTimer = null;
                }
            },

            _now() {
                const d = new Date();
                return d.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });

            async _geocodeCity(lat, lon) {
                try {
                    const res = await fetch(
                        `https://nominatim.openstreetmap.org/reverse?lat=${lat}&lon=${lon}&format=json&accept-language=pt-BR`,
                        { headers: { 'Accept-Language': 'pt-BR' } }
                    );
                    const data = await res.json();
                    return data.address?.city || data.address?.town || data.address?.municipality || data.address?.county || null;
                } catch (e) {
                    return null;
                }
            },

            async sendMessage() {
                const msg = this.chatInput.trim();
                if (!msg || this.chatLoading) return;

                this.chatMessages.push({ role: 'user', content: msg, time: this._now() });
                this.chatInput = '';

                // Solta o foco em vez de recolocar — se não, o teclado do celular
                // nunca recolhe (o usuário toca a setinha de enviar e o teclado
                // continua tampando a tela, atrapalhando ver a resposta/resultado).
                const el = document.getElementById('aiChatInput');
                if (el) el.blur();

                await this._sendToAI(msg);
                this._scrollToLastMessage();
            },

            async loadMore(msg) {
                if (msg.loadingMore) return;
                msg.loadingMore = true;
                try {
                    const res = await fetch('/api/ia/busca/mais', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ filters: msg.filters, offset: msg.results.length })
                    });
                    if (res.ok) {
                        const data = await res.json();
                        msg.results.push(...(data.results || []));
                        msg.resultsTotal = data.results_total ?? msg.resultsTotal;
                    }
                } catch (e) {
                    // silencioso — o botão "Ver mais" só fica disponível de novo pra tentar outra vez
                } finally {
                    msg.loadingMore = false;
                }
            },

            async _sendToAI(msg) {
                this.lastQuery = msg;
                this.chatLoading = true;
                // Propaga pro escopo do <section> (ancestral) que a conversa começou,
                // pra ele esconder o título grande e esticar o painel de chat pra tela cheia.
                this.chatActive = true;
                this._scrollChat();

                try {
                    const res = await fetch('/api/ia/busca', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            message: msg,
                            conversation_history: this.chatHistory.slice(),
                            detected_city: this.detectedCity
                        })
                    });

                    if (!res.ok) {
                        throw new Error('Erro na IA');
                    }

                    const data = await res.json();
                    const text = data.response_text || '';

                    if (msg) {
                        this.chatHistory.push({ role: 'user', content: msg });
                        this.chatHistory.push({ role: 'assistant', content: text });
                    } else if (text) {
                        this.chatHistory.push({ role: 'assistant', content: text });
                    }

                    // Balão próprio de confirmação (ver confirm_expansion em
                    // app/api/ia.py) — aparece ANTES do balão de resultado quando
                    // o backend detecta que a IA expandiu uma gíria/abreviação
                    // curta (ex: "jet" virou "jet ski"). Texto determinístico,
                    // não depende da IA escrever a frase certa (já visto falhar).
                    if (data.confirm_expansion && data.confirm_expansion.q) {
                        this.chatMessages.push({
                            role: 'confirm',
                            time: this._now(),
                            confirmExpansion: { q: data.confirm_expansion.q, resolved: null }
                        });
                    }

                    if (text) {
                        const aiMsg = {
                            role: 'ai',
                            content: text,
                            time: this._now(),
                            results: data.results || null,
                            resultsUrl: data.results_url || null,
                            resultsTotal: data.results_total || null,
                            filters: data.filters || null,
                            loadingMore: false,
                            alternatives: data.alternatives || null,
                            suggestedFilters: data.suggested_filters || null,
                            consultaId: data.consulta_id || null
                        };
                        this.chatMessages.push(aiMsg);

                        // Sugestão de instalar o app: um balão PRÓPRIO (não um detalhe
                        // dentro do balão de texto — assim fica visível de verdade, igual
                        // a qualquer outra mensagem). Só 1x por sessão, e só se a pessoa
                        // ainda não instalou. Normalmente só a partir da 2ª busca (pra não
                        // interromper a primeira interação) — EXCETO quando a própria
                        // pergunta já é sobre o app/instalação, caso em que mostra na hora.
                        // O botão chama exatamente a mesma função do item "Instalar App"
                        // do menu ☰ (pwaManualInstall()) — mesmo link, mesmo comportamento.
                        if (!this.installNudgeShown) {
                            const userTurns = this.chatMessages.filter(m => m.role === 'user').length;
                            // Propositalmente NÃO olha pro flag salvo em localStorage aqui —
                            // ele fica "preso" em true pra sempre depois do 1º install (mesmo
                            // que a pessoa desinstale depois), já causou o balão nunca mais
                            // aparecer de novo num celular real. isStandaloneMode() reflete o
                            // estado ATUAL (está rodando instalado agora?), é a checagem certa.
                            const standalone = typeof isStandaloneMode === 'function' && isStandaloneMode();
                            const askedAboutApp = /\b(app|aplicativo|instalar|instala|baixar|baixa)\b/i.test(msg);
                            if ((userTurns >= 2 || askedAboutApp) && !standalone) {
                                let nudgeType = null;
                                if (typeof pwaDeferredPrompt !== 'undefined' && pwaDeferredPrompt) {
                                    nudgeType = 'android';
                                } else if (typeof isIOSSafari === 'function' && isIOSSafari()) {
                                    nudgeType = 'ios';
                                } else if (askedAboutApp) {
                                    // Perguntou explicitamente sobre o app, mas o navegador
                                    // (ex: Chrome desktop) ainda não liberou o popup nativo —
                                    // mesmo assim mostra o botão, pra sempre dar algo clicável
                                    // em resposta à pergunta (pwaManualInstall() sabe lidar
                                    // com esse caso e orientar pelo menu do navegador).
                                    nudgeType = 'generic';
                                }
                                if (nudgeType) {
                                    this.chatMessages.push({ role: 'install', installNudgeType: nudgeType, time: this._now() });
                                    this.installNudgeShown = true;
                                }
                            }
                        }
                    }

                } catch (e) {
                    this.chatMessages.push({
                        role: 'ai',
                        content: '🙏 Ops, tive um problema. Pode repetir?',
                        time: this._now()
                    });
                } finally {
                    this.chatLoading = false;
                    this._scrollToLastMessage();
                    // No celular o foco fica solto de propósito (ver sendMessage() —
                    // senão o teclado nunca recolhe e tampa os resultados). No desktop
                    // não existe esse teclado virtual, então não tem motivo pra fazer a
                    // pessoa clicar de novo no campo a cada pergunta.
                    if (this.isDesktop) this.focusChatInput();
                }
            },

            _scrollChat() {
                this.$nextTick(() => {
                    const el = document.getElementById('aiChatMessages');
                    if (el) el.scrollTop = el.scrollHeight;
                });
            },

            // Usado quando a resposta termina de chegar — rola só até o TOPO do
            // balão novo (não o fundo), pra quem acabou de receber um resultado
            // comece lendo do início dele em vez de já ver o fim (ex: "Ver mais"
            // e os chips de filtro, depois dos cards). Mira sempre na ÚLTIMA
            // mensagem da IA (data-msg-role="ai"), nunca no último elemento cru
            // — a partir da 2ª pergunta, o nudge de instalar o app pode ser
            // empurrado DEPOIS do balão de resultados, e lastElementChild
            // pegaria o nudge em vez do balão.
            _scrollToLastMessage() {
                this.$nextTick(() => {
                    const container = document.getElementById('aiChatMessages');
                    if (!container) return;
                    const aiMsgs = container.querySelectorAll('[data-msg-role="ai"]');
                    const last = aiMsgs.length ? aiMsgs[aiMsgs.length - 1] : container.lastElementChild;
                    if (!last) { container.scrollTop = container.scrollHeight; return; }
                    const delta = last.getBoundingClientRect().top - container.getBoundingClientRect().top;
                    container.scrollTop += delta;
                });
            },

        }
    }

    // Preço de anúncio em card (chat, Stories): mesma regra do filtro price_tag

    window.formatCardPrice = function(price) {
        let text = price.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });

        return 'R$ ' + text;
    }

    async function saveCurrentSearch() {
        const label = prompt('Como você quer chamar essa busca? (ex: "Gato doação em Palmas")');
        if (!label || !label.trim()) return;

        const params = new URLSearchParams(window.location.search);
        const toFloat = (v) => (v === null || v === '') ? null : parseFloat(v);

        try {
            const res = await fetch('/api/saved-searches', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                credentials: 'include',
                body: JSON.stringify({
                    label: label.trim(),
                    q: params.get('q') || null,
                    category: params.get('category') || null,
                    ad_type: params.get('ad_type') || null,
                    city: params.get('city') || null,
                    neighborhood: params.get('neighborhood') || null,
                    price_min: toFloat(params.get('price_min')),
                    price_max: toFloat(params.get('price_max')),
                }),
            });
            if (res.status === 401) {
                showToast('Faça login para salvar buscas');
                setTimeout(() => window.location.href = '/login', 1200);
                return;
            }
            if (!res.ok) {
                const err = await res.json().catch(() => ({}));
                showToast(err.detail || 'Não foi possível salvar a busca');
                return;
            }
            showToast('Busca salva! Você será avisado por e-mail sobre anúncios novos. 🔔');
        } catch (e) {
            showToast('Erro ao salvar busca. Tente novamente.');
        }
    }

    async function copyToClipboard(text) {
        if (navigator.clipboard && window.isSecureContext) {
            try {
                await navigator.clipboard.writeText(text);
                return true;
            } catch (err) {
                console.error("navigator.clipboard failed:", err);
            }
        }
        
        // Fallback robusto para contextos não seguros (HTTP) ou navegadores antigos
        try {
            const textArea = document.createElement("textarea");
            textArea.value = text;
            textArea.style.position = "fixed";
            textArea.style.left = "-9999px";
            textArea.style.top = "0";
            document.body.appendChild(textArea);
            textArea.focus();
            textArea.select();
            const success = document.execCommand('copy');
            document.body.removeChild(textArea);
            return success;
        } catch (err) {
            console.error("Cópia falhou:", err);
            return false;
        }
    }

    window.shareAdNativo = async function(event, id, title, quem) {
        if (event) {
            event.preventDefault();
            event.stopPropagation();
        }
        const url = `${window.location.origin}/c/${id}`;
        // Beacon best-effort — nunca trava o fluxo de compartilhar por causa
        // disso (ver app/api/ia.py::api_busca_interacao). Marca intenção real
        // de compartilhar, tanto pelo menu nativo quanto pela cópia manual.
        const trackShare = () => {
            try { navigator.sendBeacon('/api/ia/busca/interacao', new Blob([JSON.stringify({ad_id: parseInt(id), tipo: 'compartilhou'})], {type:'application/json'})); } catch(e) {}
        };

        if (navigator.share) {
            try {
                await navigator.share({
                    title: title,
                    // 'dono': o próprio anunciante divulgando (Meu Painel).
                    // Curto: o título e o preço já vêm na prévia do link (dono, 22/11/2026).
                    text: 'Confira esta oportunidade:',
                    url: url,
                });
                trackShare();
                return;
            } catch (err) {
                if (err.name !== 'AbortError') {
                    console.error("Erro ao compartilhar:", err);
                } else {
                    return; // Usuário cancelou
                }
            }
        }

        // Tenta copiar para o clipboard (com fallback robusto)
        const copied = await copyToClipboard(url);
        if (copied) {
            trackShare();
            window.showToast("Link copiado!");
        } else {
            window.showToast("Erro ao copiar link.");
        }
    }

    // Compartilha a URL ATUAL da página (não um id fixo como shareAd acima) —
    // usada no perfil público, onde o filtro/ordenação já atualiza a URL
    // sozinho via hx-push-url (ver perfil.html), então isso já manda o link
    // filtrado certinho pro cliente, sem o vendedor precisar saber que dá
    // pra copiar da barra de endereço. Se `ids` vier preenchido (seleção
    // manual de anúncios dentro do filtro atual), acrescenta ?ids=1,2,3 na
    // URL antes de compartilhar, mantendo os outros filtros já na barra.
    window.shareCurrentPage = async function(title, ids) {
        let url = window.location.href;
        if (ids && ids.length) {
            const u = new URL(url);
            u.searchParams.set("ids", ids.join(","));
            url = u.toString();
        }
        if (navigator.share) {
            try {
                await navigator.share({ title: title, text: title, url: url });
                return;
            } catch (err) {
                if (err.name === 'AbortError') return;
            }
        }
        const copied = await copyToClipboard(url);
        if (copied) {
            window.showToast("Link copiado!");
        } else {
            window.showToast("Erro ao copiar link.");
        }
    }

function openReportModal() {
    var modal = document.getElementById('report-modal');
    modal.classList.remove('hidden');
    modal._prevFocus = document.activeElement;
    var first = modal.querySelector('select, button, input');
    if (first) first.focus();
    modal._trapHandler = function(e) {
        if (e.key === 'Escape') { closeReportModal(); return; }
        if (e.key !== 'Tab') return;
        var focusable = modal.querySelectorAll('button, select, input, textarea, [tabindex]:not([tabindex="-1"])');
        var firstEl = focusable[0], lastEl = focusable[focusable.length - 1];
        if (e.shiftKey) { if (document.activeElement === firstEl) { e.preventDefault(); lastEl.focus(); } }
        else { if (document.activeElement === lastEl) { e.preventDefault(); firstEl.focus(); } }
    };
    document.addEventListener('keydown', modal._trapHandler);
}
function closeReportModal() {
    var modal = document.getElementById('report-modal');
    modal.classList.add('hidden');
    if (modal._trapHandler) document.removeEventListener('keydown', modal._trapHandler);
    if (modal._prevFocus) modal._prevFocus.focus();
}

async function toggleFavorite(adId) {
    const btn = document.getElementById('fav-btn');
    const icon = document.getElementById('fav-icon');
    const label = document.getElementById('fav-label');
    const isFav = icon.textContent === '♥';

    const url = `/api/users/me/favorites/${adId}`;
    const method = isFav ? 'DELETE' : 'POST';
    const res = await fetch(url, { method });

    if (res.ok) {
        if (isFav) {
            icon.textContent = '♡';
            label.textContent = 'Salvar';
            btn.classList.remove('border-red-300', 'bg-red-50', 'text-red-500');
            btn.classList.add('border-gray-200', 'text-gray-500');

            icon.textContent = '♥';
            label.textContent = 'Salvo';
            btn.classList.add('border-red-300', 'bg-red-50', 'text-red-500');
            btn.classList.remove('border-gray-200', 'text-gray-500');
        }
        window.showToast(isFav ? 'Removido dos favoritos' : 'Adicionado aos favoritos');
    } else if (res.status === 401) {
        window.location.href = '/login';
    }
}
