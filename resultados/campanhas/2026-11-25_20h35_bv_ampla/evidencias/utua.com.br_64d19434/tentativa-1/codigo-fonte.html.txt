    // manager (cookie-first, localStorage fallback). An existing UUID is reused

    var UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;

            var escapedName = name.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
            var match = document.cookie.match(
                new RegExp("(?:^|; )" + escapedName + "=([^;]*)")
            );

            if (!match) return "";

            try {
                return decodeURIComponent(match[1]);
            } catch (error) {
                return "";
            }
        }

        function setCookie(name, value) {
            var secure = window.location.protocol === "https:" ? "; Secure" : "";
            document.cookie =
                name +
                "=" +
                encodeURIComponent(value) +
                "; path=/; max-age=" +
                SESSION_COOKIE_MAX_AGE_SEC +
                "; SameSite=Lax" +
                secure;
        }

        function readLocalStorage(key) {
            try {
                return window.localStorage.getItem(key) || "";
            } catch (error) {
                return "";
            }
        }

        function writeLocalStorage(key, value) {
            try {
                window.localStorage.setItem(key, value);
            } catch (error) {
                // Cookies remain available if localStorage is blocked.
            }
        }

        function generateSessionId() {
            if (window.crypto && typeof window.crypto.randomUUID === "function") {
                return window.crypto.randomUUID();
            }

            if (window.crypto && typeof window.crypto.getRandomValues === "function") {
                var bytes = new Uint8Array(16);
                window.crypto.getRandomValues(bytes);
                bytes[6] = (bytes[6] & 0x0f) | 0x40;
                bytes[8] = (bytes[8] & 0x3f) | 0x80;

                return Array.from(bytes)
                    .map(function (byte) {
                        return byte.toString(16).padStart(2, "0");
                    })
                    .join("")

// =========================================================================
// [FIX 3] Uma única fila global (window.__cdpConversionQueue), compartilhada
// pelas duas IIFEs deste bloco via window. Substitui a assimetria antiga
// (bloco 12 fazia polling por evento; bloco 13 descartava quando a lib ainda
// não existia). A fila preserva ordem FIFO, faz retry por no máximo ~10s e
// envia cada item exatamente uma vez quando window.ConversionAnalytics.track
// existir. Existe no máximo UM timer compartilhado por página.
(function initConversionAnalyticsQueue() {
  'use strict';

  var QUEUE = window.__cdpConversionQueue;

  if (QUEUE) {
    // Já inicializada por uma execução anterior do bloco unificado.
    return;
  }

  QUEUE = {
    items: [],
    timerId: null,
    startedAt: null
  };

  window.__cdpConversionQueue = QUEUE;

  function hasConversionAnalytics() {
    return !!(
      window.ConversionAnalytics &&
      typeof window.ConversionAnalytics.track === 'function'
    );
  }

  /**
   * Drena a fila em ordem FIFO enquanto houver itens.
   *
   * Atribui event_timestamp_usec no envio efetivo, apenas se ainda não
   * existir (caso dos itens do bloco 13) — nunca muta o mesmo payload
   * duas vezes de forma incorreta.
   */
  function drainQueue() {
    if (!hasConversionAnalytics()) {
      return;
    }

    var item;

    while (QUEUE.items.length > 0) {
      item = QUEUE.items.shift();

      try {
        if (
          item.payload &&
          typeof item.payload === 'object' &&
          !item.payload.event_timestamp_usec
        ) {
          item.payload.event_timestamp_usec = Date.now() * 1000;
        }

        window.ConversionAnalytics.track(
          item.eventName,
          item.payload || {}
        );
      } catch (error) {
        if (window.console) {
          console.log(
            '[CDP-GAM] Error sending event (conversion analytics):',
            error
          );
        }
      }
    }
  }

  /**
   * Garante no máximo um timer compartilhado.
   *
   * Para quando a fila esvazia ou depois de ~10s de retry sem a lib
   * ConversionAnalytics disponível.
   */
  function startDrainTimer() {
    if (QUEUE.timerId) {
      return;
    }

    QUEUE.timerId = setInterval(function () {
      drainQueue();

      if (QUEUE.items.length === 0) {
        clearInterval(QUEUE.timerId);
        QUEUE.timerId = null;
        QUEUE.startedAt = null;
      } else if (Date.now() - QUEUE.startedAt >= 10000) {
        // Retry por no máximo ~10s: descarta o restante e encerra o timer.
        clearInterval(QUEUE.timerId);
        QUEUE.timerId = null;
        QUEUE.startedAt = null;
        QUEUE.items.length = 0;

        if (window.console) {
          console.log(
            '[CDP-GAM] ConversionAnalytics indisponível (timeout ~10s) — itens descartados'
          );
        }
      }
    }, 100);
  }

  /**
   * Enfileira um evento na fila única.
   *
   * Quando a lib já está disponível, drena de forma síncrona e imediata
   * (ordem FIFO preservada entre os blocos 12 e 13). Quando não está,
   * inicia/garante o timer compartilhado de retry.
   */
  function enqueue(eventName, payload) {
    QUEUE.items.push({
      eventName: eventName,
      payload: payload || {}
    });

    if (!QUEUE.startedAt) {
      QUEUE.startedAt = Date.now();
    }

    if (hasConversionAnalytics()) {
      drainQueue();
    } else {
      startDrainTimer();
    }
  }

  window.__cdpConversionEnqueue = enqueue;
})();

    // Armazenar dados da resposta para uso em eventos posteriores (snake_case).
    window.cdpSlotData[slotId] = {
      line_item_id: responseInfo ? responseInfo.lineItemId : null,
      advertiser_id: responseInfo ? responseInfo.advertiserId : null,
      creative_id: responseInfo ? responseInfo.creativeId : null,
      campaign_id: responseInfo ? responseInfo.campaignId : null,
      source_agnostic_creative_id: responseInfo ? responseInfo.sourceAgnosticCreativeId : null,
      source_agnostic_line_item_id: responseInfo ? responseInfo.sourceAgnosticLineItemId : null,
      is_sold: !!responseInfo
    };

    sendToCDP('slotResponseReceived', {
      slot_id: slotId,
      ad_unit_path: adUnitPath,
      // Join keys para Data Transfer
      line_item_id: responseInfo ? responseInfo.lineItemId : null,
      advertiser_id: responseInfo ? responseInfo.advertiserId : null,
      creative_id: responseInfo ? responseInfo.creativeId : null,
      campaign_id: responseInfo ? responseInfo.campaignId : null,
      source_agnostic_creative_id: responseInfo ? responseInfo.sourceAgnosticCreativeId : null,
      source_agnostic_line_item_id: responseInfo ? responseInfo.sourceAgnosticLineItemId : null,
      is_sold: !!responseInfo,
      response_time_ms: timestamps.requested_at ? Date.now() - timestamps.requested_at : null
    });

    // Atualizar label se vendeu
    if (responseInfo) {
      updateAdLabel(adUnitPath, true);
    }
  }

  /**
   * slotRenderEnded - Slot renderizado (PRINCIPAL para join com Data Transfer)
   */
  function handleSlotRenderEnded(event) {
    var slotId = event.slot.getSlotElementId();
    var adUnitPath = event.slot.getAdUnitPath();
    // [FIX 3] rendered_at agora e' persistido no state global.
    var timestamps = slotTimestamps(slotId);
    timestamps.rendered_at = Date.now();

    sendToCDP('slotRenderEnded', {
      slot_id: slotId,
      ad_unit_path: adUnitPath,
      response_identifier: event.responseIdentifier || null,
      // CRITICAL: Join keys para Data Transfer (aqui vem do evento do GPT, camelCase e' o correto)
      line_item_id: event.lineItemId || null,
      creative_id: event.creativeId || null,
      source_agnostic_creative_id: event.sourceAgnosticCreativeId,
      source_agnostic_line_item_id: event.sourceAgnosticLineItemId,
      advertiser_id: event.advertiserId || null,
      campaign_id: event.campaignId || null,
      // Detalhes do render
      is_empty: event.isEmpty,
      size_rendered: event.size ? event.size[0] + 'x' + event.size[1] : null,
      is_backfill: event.isBackfill || false,
      creative_template_id: event.creativeTemplateId || null,
      // Posicionamento
      slot_visible_on_render: isElementInViewport(slotId),
      slot_top_offset: getElementTopOffset(slotId),
      // Timing
      render_time_ms: timestamps.requested_at ? Date.now() - timestamps.requested_at : null
    });

    // Rastrear slots vazios como oportunidades perdidas
    if (event.isEmpty) {
      sendToCDP('slotEmpty', {
        slot_id: slotId,
        ad_unit_path: adUnitPath,
        reason: 'no_fill',
        floor_price: getMagicAdsPrice(adUnitPath)
      });
    }
  }

  /**

  function handleImpressionViewable(event) {

      line_item_id: slotData.line_item_id,
      creative_id: slotData.creative_id,
      // [FIX 2] cdpSlotData grava em snake_case — ler em camelCase dava sempre undefined.
      source_agnostic_creative_id: slotData.source_agnostic_creative_id,
      source_agnostic_line_item_id: slotData.source_agnostic_line_item_id,
      // Contexto de viewability
      time_to_viewable_ms: timestamps.rendered_at ? Date.now() - timestamps.rendered_at : null,
      scroll_depth: window.scrollY,
      viewport_visible_pct: getVisiblePercentage(slotId),
      // Magic Ads floor price
      floor_price: getMagicAdsPrice(adUnitPath)
    });

    // Push para dataLayer (top slot)
    if (adUnitPath.indexOf('_top') > -1) {
      window.dataLayer = window.dataLayer || [];
      window.dataLayer.push({ event: 'top_impression' });
    }
  }

  /**

    });
  }

  function handleSlotOnload(event) {
    var slotId = event.slot.getSlotElementId();
    var adUnitPath = event.slot.getAdUnitPath();
    var timestamps = slotTimestamps(slotId);

    sendToCDP('slotOnload', {
      slot_id: slotId,
      ad_unit_path: adUnitPath,
      load_time_ms: timestamps.rendered_at ? Date.now() - timestamps.rendered_at : null,
      total_time_ms: timestamps.requested_at ? Date.now() - timestamps.requested_at : null
    });
  }

  // ========== AD LABEL ==========
  function updateAdLabel(adUnitPath, isSold) {
    if (!isSold) return;
    try {
      // Child content slots
      if (adUnitPath.indexOf('_content_0') > -1) {
        var parts = adUnitPath.split('content_0');
        if (parts.length >= 2) {
          var position = parseInt(parts[1], 10) - 1;
          var labels = document.getElementsByClassName(CONFIG.DOMAIN + '-mob-content-child-label');
          var labelDiv = Array.prototype.slice.call(labels)[position];
          if (labelDiv) {
            labelDiv.innerHTML = 'ADVERTISEMENT';
            labelDiv.style.display = 'block';
          }
        }
        return;
      }
      // Main slots
      var labelMap = {};
      labelMap['/21862753527/' + CONFIG.DOMAIN + '_mob_top'] = CONFIG.DOMAIN + '-mob-top-label';
      labelMap['/21862753527/' + CONFIG.DOMAIN + '_mob_content'] = CONFIG.DOMAIN + '-mob-content-label';
      labelMap['/21862753527/' + CONFIG.DOMAIN + '_desk_top'] = CONFIG.DOMAIN + '-desk-top-label';
      labelMap['/21862753527/' + CONFIG.DOMAIN + '_desk_content'] = CONFIG.DOMAIN + '-desk-content-label';
      labelMap['/21862753527/' + CONFIG.DOMAIN + '_desk_sidebar'] = CONFIG.DOMAIN + '-desk-sidebar-label';

      var labelId = labelMap[adUnitPath];
      if (labelId) {
        var lbl = document.getElementById(labelId);
        if (lbl) {
          lbl.innerHTML = 'ADVERTISEMENT';
          lbl.style.display = 'block';
        }
      }
    } catch (e) {
      log('Error updating label:', e);
    }
  }

  // ========== INIT ==========
  function init() {
    window.googletag = window.googletag || { cmd: [] };

    googletag.cmd.push(function() {
      var handlers = {
        'slotRequested': handleSlotRequested,
        'slotResponseReceived': handleSlotResponseReceived,
        'slotRenderEnded': handleSlotRenderEnded,
        'impressionViewable': handleImpressionViewable,
        'slotVisibilityChanged': handleSlotVisibilityChanged,
        'slotOnload': handleSlotOnload
      };

      CONFIG.EVENTS_TO_TRACK.forEach(function(eventName) {
        if (handlers[eventName]) {
          googletag.pubads().addEventListener(eventName, handlers[eventName]);
          log('Registered:', eventName);
        }
      });

      log('CDP GAM Event Tracker v2.1 initialized');
    });
  }

  init();
})();

   * O bloco 12 é quem alimenta o Growth Control (window.Analytics). Este bloco

  function track(eventName, properties) {
    try {
      // [FIX 3] Caminho único: fila compartilhada do unified. Preserva ordem

        return;
      }

      // Fallback defensivo — fila não inicializada.
      if (
        window.ConversionAnalytics &&
        typeof window.ConversionAnalytics.track === 'function'
      ) {
        window.ConversionAnalytics.track(
          eventName,

      log(
        'ConversionAnalytics indisponível. Evento não enviado:',

        'Erro ao enviar evento:',
        eventName,

    }

   * Eventos gerais (pageAdsRenderCompleted e AdsImpression) continuam
   * funcionando nos dois casos.

  function isInterstitialDomainActive() {

    // emitido neste domínio — nem mesmo se existirem slots com "interstitial".
    if (!CONFIG.INTERSTITIAL_ENABLED) {
      return false;
    }

    if (window.interstitialSlot) {
      return true;
    }

    if (sawInterstitialSlot) {
      return true;
    }

      if (
        window.googletag &&
        typeof window.googletag.pubads === 'function'
      ) {
        var pubads = window.googletag.pubads();

        if (
          pubads &&
          typeof pubads.getSlots === 'function'

            try {
              adUnitPath = slots[i].getAdUnitPath() || '';
            } catch (error) {
              adUnitPath = '';

            if (/interstitial/i.test(adUnitPath)) {
              return true;
            }
          }
        }

    } catch (error) {
      // googletag indisponível — trata como domínio sem interstitial
    }

    // de forma confiável (o piloto utua.com.br tem interstitial). Isso garante
    // que o clique elegível não seja perdido só porque o slot ainda não foi

    return true;

  function isInterstitialSlot(slot) {

  /**

  function roundDimension(value) {

  /**
   * Mede o estado visual atual de um elemento.

  function getElementVisualState(element) {
    var rect;
    var style;

      style = window.getComputedStyle(element);

        height: 0,

      height: roundDimension(rect.height),
      display: style.display || null,
      visibility: style.visibility || null,
      hidden: !!element.hidden
    };

   * Seleciona os elementos cuja expansão pode representar a abertura do slot.
   *
   * - observa a própria div GPT;
   * - quando existe, observa também o ancestral marcado como collapse.
   *
   * O seletor fica restrito à árvore do slot, portanto não captura menus ou
   * accordions da página que também usem a classe genérica .collapse.
   */
  function getAdExpansionTargets(slotElement) {
    var targets = [];
    var collapseElement = null;

    if (
      slotElement &&
      typeof slotElement.closest === 'function'
    ) {
      try {
        collapseElement = slotElement.closest(
          '.collapse, .ad-collapse, ' +
          '[data-collapse], [data-ad-collapse]'
        );
      } catch (error) {
        collapseElement = null;
      }
    }

    if (
      collapseElement &&
      collapseElement !== document.body &&
      collapseElement !== document.documentElement
    ) {
      targets.push({
        element: collapseElement,
        role: 'collapse_ancestor'
      });
    }

    targets.push({
      element: slotElement,
      role: 'gpt_slot'
    });

    return targets;
  }

  /**
   * Desliga observers e timer de um slot.
   */
  function cleanupAdExpansionObservation(slotKey) {
    var state =
      adExpansionState.slots[slotKey];

    if (!state) {
      return;
    }

    if (state.resizeObserver) {
      state.resizeObserver.disconnect();
      state.resizeObserver = null;
    }

    if (state.mutationObserver) {
      state.mutationObserver.disconnect();
      state.mutationObserver = null;
    }

    if (state.timeoutId) {
      clearTimeout(state.timeoutId);
      state.timeoutId = null;
    }
  }

  /**
   * Encerra todos os observers depois que o evento global já foi enviado.
   * Isso evita trabalho desnecessário nos demais anúncios da página.
   */
  function cleanupAllAdExpansionObservations() {
    Object.keys(adExpansionState.slots)
      .forEach(function (slotKey) {
        cleanupAdExpansionObservation(slotKey);
      });
  }

  /**
   * Envia AdsImpression somente quando as duas condições foram confirmadas:
   *
   * 1. houve transição DOM colapsado -> expandido;
   * 2. slotRenderEnded informou fill (isEmpty=false).
   *
   * A deduplicação é global: o primeiro slot válido envia o evento e encerra
   * todos os observers. A ordem entre expansão e fill pode variar; por isso a
   * expansão fica pendente até o fill ser conhecido.
   */
  function maybeSendAdsImpression(slotKey) {
    var state =
      adExpansionState.slots[slotKey];

    if (
      !state ||
      adExpansionState.eventSent ||
      state.eventSent ||
      !state.filled ||
      !state.pendingExpansion
    ) {
      return;
    }

    state.eventSent = true;
    adExpansionState.eventSent = true;
    adExpansionState.sourceSlotKey = slotKey;

    var responseInfo =
      getResponseInformation(state.slot);

    var expansion =
      state.pendingExpansion;

    track(
      'AdsImpression',
      {
        event_scope: 'page',
        deduplication: 'first_expanded_filled_ad',
        slot_id: state.slotId,
        ad_unit_path: state.adUnitPath,
        observed_element_role: expansion.element_role,
        observed_element_id: expansion.element_id,
        observed_element_class: expansion.element_class,
        previous_height_px: expansion.previous_height_px,
        expanded_height_px: expansion.expanded_height_px,
        height_delta_px:
          roundDimension(
            expansion.expanded_height_px -
            expansion.previous_height_px
          ),
        expanded_width_px: expansion.expanded_width_px,
        requested_at: state.requestedAt,
        rendered_at: state.renderedAt,
        expanded_at: expansion.expanded_at,
        time_request_to_expand_ms:
          state.requestedAt
            ? expansion.expanded_at -
              state.requestedAt
            : null,
        time_render_to_expand_ms:
          state.renderedAt
            ? expansion.expanded_at -
              state.renderedAt
            : null,
        line_item_id:
          state.renderMetadata.line_item_id ||
          getResponseValue(responseInfo, 'lineItemId'),
        creative_id:
          state.renderMetadata.creative_id ||
          getResponseValue(responseInfo, 'creativeId'),
        advertiser_id:
          state.renderMetadata.advertiser_id ||
          getResponseValue(responseInfo, 'advertiserId'),
        campaign_id:
          state.renderMetadata.campaign_id ||
          getResponseValue(responseInfo, 'campaignId'),
        size_rendered:
          state.renderMetadata.size_rendered
      }
    );

    cleanupAllAdExpansionObservations();
  }

  /**
   * Reavalia todos os elementos observados e registra a primeira transição
   * válida de altura ~0 para uma altura visual útil.
   */
  function evaluateAdExpansion(slotKey) {
    var state =
      adExpansionState.slots[slotKey];

    if (
      !state ||
      adExpansionState.eventSent ||
      state.eventSent ||
      state.isEmpty
    ) {
      return;
    }

    state.targets.forEach(function (targetState) {
      if (state.pendingExpansion) {
        return;
      }

      var current =
        getElementVisualState(
          targetState.element
        );

      var previous =
        targetState.lastVisualState;

      /**
       * Se o elemento voltar a colapsar antes da primeira expansão válida,
       * mantém a memória de que houve estado fechado.
       */
      if (isElementCollapsed(current)) {
        targetState.wasCollapsed = true;
      }

      if (
        targetState.wasCollapsed &&
        isElementExpanded(current) &&
        previous &&
        !isElementExpanded(previous)
      ) {
        state.pendingExpansion = {
          element_role: targetState.role,
          element_id: targetState.element.id || null,
          element_class:
            typeof targetState.element.className ===
              'string'
              ? targetState.element.className
              : null,
          previous_height_px: previous.height,
          expanded_height_px: current.height,
          expanded_width_px: current.width,
          expanded_at:
            Date.now()
        };

        log(
          'Expansão visual detectada:',
          slotKey,
          state.pendingExpansion
        );

        maybeSendAdsImpression(slotKey);
      }

      targetState.lastVisualState = current;
    });
  }

  /**
   * Começa a observar o slot no slotRequested, antes do retorno do anúncio.
   * Isso é essencial para guardar o estado fechado inicial.
   */
  function startAdExpansionObservation(slot) {
    if (
      !slot ||
      adExpansionState.eventSent
    ) {
      return;
    }

    var slotKey = getSlotKey(slot);
    var slotId = getSlotId(slot);

    if (
      !slotKey ||
      !slotId ||
      adExpansionState.slots[slotKey]
    ) {
      return;
    }

    var slotElement =
      document.getElementById(slotId);

    if (!slotElement) {
      log(
        'Slot não encontrado no DOM para observar expansão:',
        slotId
      );
      return;
    }

    var targets =
      getAdExpansionTargets(slotElement)
        .map(function (target) {
          var visualState =
            getElementVisualState(
              target.element
            );

          return {
            element: target.element,
            role: target.role,
            wasCollapsed: isElementCollapsed( visualState ),
            lastVisualState:
              visualState
          };
        });

    var state = {
      slot: slot,
      slotId: slotId,
      adUnitPath: getAdUnitPath(slot),
      requestedAt: Date.now(),
      renderedAt: null,
      filled: false,
      isEmpty: false,
      eventSent: false,
      pendingExpansion: null,
      renderMetadata: {},
      targets: targets,
      resizeObserver: null,
      mutationObserver: null,
      timeoutId:
        null
    };

    adExpansionState.slots[slotKey] = state;

    /**
     * ResizeObserver é a fonte principal: reage à mudança real de dimensão.
     */
    if (typeof window.ResizeObserver === 'function') {
      state.resizeObserver =
        new window.ResizeObserver(function () {
          evaluateAdExpansion(slotKey);
        });

      targets.forEach(function (targetState) {
        state.resizeObserver.observe(
          targetState.element
        );
      });
    }

    /**
     * MutationObserver é fallback e também cobre display/class/hidden.
     */
    if (typeof window.MutationObserver === 'function') {
      state.mutationObserver =
        new window.MutationObserver(function () {
          window.requestAnimationFrame(function () {
            evaluateAdExpansion(slotKey);
          });
        });

      targets.forEach(function (targetState) {
        state.mutationObserver.observe(
          targetState.element,
          {
            attributes: true,
            attributeFilter: [
              'class',
              'style',
              'hidden'
            ],
            childList: true,
            subtree: true
          }
        );
      });
    }

    state.timeoutId =
      setTimeout(function () {
        cleanupAdExpansionObservation(
          slotKey
        );
      }, CONFIG.AD_EXPANSION_OBSERVER_TIMEOUT_MS);

    log(
      'Observando expansão do slot:',
      {
        slot_id: slotId,
        ad_unit_path: state.adUnitPath,
        targets: targets.map(function (targetState) {
          return {
            role: targetState.role,
            id: targetState.element.id || null,
            class_name:
              typeof targetState.element.className ===
                'string'
                ? targetState.element.className
                : null,
            initial_height_px: targetState.lastVisualState.height,
            initially_collapsed:
              targetState.wasCollapsed
          };
        })
      }
    );
  }

  /**
   * Entrega ao observer o resultado comercial do slotRenderEnded.
   */
  function handleAdExpansionRenderEnded(
    event,
    renderedAt
  ) {
    if (!event || !event.slot) {
      return;
    }

    var slotKey = getSlotKey(event.slot);
    var state =
      adExpansionState.slots[slotKey];

    if (!state) {
      return;
    }

    state.renderedAt = renderedAt;
    state.isEmpty = !!event.isEmpty;
    state.filled = !event.isEmpty;

    state.renderMetadata = {
      line_item_id: event.lineItemId || null,
      creative_id: event.creativeId || null,
      advertiser_id: event.advertiserId || null,
      campaign_id: event.campaignId || null,
      size_rendered:
        event.size
          ? event.size[0] +
            'x' +
            event.size[1]
          : null
    };

    if (state.isEmpty) {
      cleanupAdExpansionObservation(slotKey);
      return;
    }

    /**
     * A expansão pode ter acontecido milissegundos antes do slotRenderEnded.
     */
    maybeSendAdsImpression(slotKey);
  }

  /**
   * Cria um identificador único para cada clique.
   */
  function createClickId() {
    return (
      Date.now().toString(36) +
      '-' +
      Math.random().toString(36).slice(2, 10)
    );
  }

  /**
   * Retorna o último clique quando ele aconteceu dentro da janela
   * de tempo configurada.
   */
  function getRecentClick() {
    if (
      !interstitialState.lastClickId ||
      !interstitialState.lastClickAt
    ) {
      return null;
    }

    var elapsedMs =
      Date.now() - interstitialState.lastClickAt;

    if (
      elapsedMs >
      CONFIG.CLICK_RENDER_WINDOW_MS
    ) {
      return null;
    }

    return {
      click_id: interstitialState.lastClickId,
      clicked_at: interstitialState.lastClickAt,
      click_href: interstitialState.lastClickHref,
      click_text: interstitialState.lastClickText,
      time_from_click_ms:
        elapsedMs
    };
  }

  // =========================================================================
  // CLIQUES ELEGÍVEIS PARA O INTERSTITIAL
  // =========================================================================

  /**
   * Monitora cliques em links.
   *
   * Links com:
   *
   * data-google-interstitial="false"
   *
   * são ignorados.
   *
   * Este clique é apenas potencialmente elegível.
   * Ele não comprova que o Google apresentou o interstitial.
   *
   * NÃO é duplicidade: a origem é o DOM, não um evento do GPT. O bloco 12
   * não escuta cliques.
   */
  document.addEventListener(
    'click',

    function handleEligibleLinkClick(event) {
      var target = event.target;

      if (
        !target ||
        typeof target.closest !== 'function'
      ) {
        return;
      }

      var link = target.closest('a[href]');

      if (!link) {
        return;
      }

      /**
       * Ignora links explicitamente excluídos do web interstitial.
       */
      var excludedElement = target.closest(
        '[data-google-interstitial="false"]'
      );

      if (excludedElement) {
        return;
      }

      /**
       * [UNIFICADO — GUARDA DE DOMÍNIO] Sem interstitial de verdade nesta
       * instalação (sem window.interstitialSlot e sem nenhum slot definido com
       * ad unit contendo "interstitial"), não existe clique elegível: o evento
       * interstitialEligibleLinkClick NÃO é enviado. Eventos gerais
       * (pageAdsRenderCompleted e AdsImpression) continuam funcionando.
       */
      if (!isInterstitialDomainActive()) {
        return;
      }

      var clickId = createClickId();

      var clickText = (
        link.textContent || ''
      )
        .replace(/\s+/g, ' ')
        .trim()
        .slice(0, 150);

      var clickHref = null;

      try {
        clickHref =
          link.href ||
          link.getAttribute('href') ||
          null;
      } catch (error) {
        clickHref = null;
      }

      interstitialState.lastClickId =
        clickId;

      interstitialState.lastClickAt =
        Date.now();

      interstitialState.lastClickHref =
        clickHref;

      interstitialState.lastClickText =
        clickText || null;

      track(
        'interstitialEligibleLinkClick',
        {
          click_id: clickId,
          click_href: clickHref,
          click_text: clickText || null,
          clicked_at:
            interstitialState.lastClickAt
        }
      );
    },

    true
  );

  /**
   * Retorna um resumo do estado atual dos slots da página.
   */
  function getPageAdsSummary() {
    var requestedKeys = Object.keys(
      pageAdsState.requestedSlots
    );

    var completedKeys = Object.keys(
      pageAdsState.completedSlots
    );

    var requestedSlots = [];
    var filledSlots = [];
    var emptySlots = [];
    var pendingSlots = [];

    requestedKeys.forEach(function (key) {
      var requested =
        pageAdsState.requestedSlots[key];

      requestedSlots.push({
        slot_id: requested.slot_id,
        ad_unit_path:
          requested.ad_unit_path
      });

      var completed =
        pageAdsState.completedSlots[key];

      // [FIX 4] Um slot só é "pendente" se ainda não concluiu OU se a
      // conclusão registrada pertence a uma geração/request anterior do
      // mesmo slot.
      if (
        !completed ||
        completed.generation !== requested.generation
      ) {
        pendingSlots.push({
          slot_id: requested.slot_id,
          ad_unit_path:
            requested.ad_unit_path
        });
      }
    });

    completedKeys.forEach(function (key) {
      var completed =
        pageAdsState.completedSlots[key];

      var requested =
        pageAdsState.requestedSlots[key];

      // [FIX 4] Ignora conclusões de gerações/requests anteriores: nunca
      // contam como concluídas para o request vigente do slot.
      if (
        requested &&
        completed.generation !== requested.generation
      ) {
        return;
      }

      var slotInformation = {
        slot_id: completed.slot_id,
        ad_unit_path:
          completed.ad_unit_path
      };

      if (completed.is_empty) {
        emptySlots.push(slotInformation);
      } else {
        filledSlots.push(slotInformation);
      }
    });

    return {
      requested_count: requestedKeys.length,
      completed_count: completedKeys.length,
      filled_count: filledSlots.length,
      empty_count: emptySlots.length,
      pending_count: pendingSlots.length,
      requested_slots: requestedSlots,
      filled_slots: filledSlots,
      empty_slots: emptySlots,
      pending_slots:
        pendingSlots
    };
  }

  /**
   * Atualiza o momento da última atividade relacionada a anúncios.
   */
  function markPageAdsActivity() {
    pageAdsState.lastActivityAt = Date.now();
  }

  /**
   * Cancela o timer de estabilidade existente.
   */
  function clearPageAdsQuietTimer() {
    if (pageAdsState.quietTimerId) {
      clearTimeout(
        pageAdsState.quietTimerId
      );

      pageAdsState.quietTimerId = null;
    }
  }

  /**
   * Cancela o timer máximo existente.
   */
  function clearPageAdsMaxTimeout() {
    if (pageAdsState.maxTimeoutId) {
      clearTimeout(
        pageAdsState.maxTimeoutId
      );

      pageAdsState.maxTimeoutId = null;
    }
  }

  /**
   * Envia o evento consolidado dos anúncios da página.
   *
   * completion_reason pode ser:
   *
   * quiet_period_completed
   * Todos os slots solicitados terminaram e não houve nova atividade
   * durante o período configurado.
   *
   * max_timeout
   * O tempo máximo terminou, mesmo havendo slots pendentes.
   */
  function sendPageAdsRenderResult(
    completionReason
  ) {
    if (pageAdsState.reportSent) {
      return;
    }

    var summary = getPageAdsSummary();

    /**
     * Não envia o evento se nenhum anúncio foi solicitado.
     */
    if (summary.requested_count === 0) {
      return;
    }

    var allAdsRendered =
      summary.requested_count ===
        summary.completed_count &&
      summary.pending_count === 0;

    /**
     * Fora do timeout máximo, somente envia se todos os slots
     * solicitados tiverem terminado.
     */
    if (
      completionReason !== 'max_timeout' &&
      !allAdsRendered
    ) {
      return;
    }

    pageAdsState.reportSent = true;

    clearPageAdsQuietTimer();
    clearPageAdsMaxTimeout();

    track(
      'pageAdsRenderCompleted',
      {
        /**
         * Todos os slots solicitados dispararam slotRenderEnded.
         *
         * Slots vazios também são considerados concluídos.
         */
        all_ads_rendered: allAdsRendered,

        /**
         * Todos terminaram e nenhum ficou vazio.
         */
        all_ads_filled: allAdsRendered && summary.empty_count === 0,
        requested_count: summary.requested_count,
        completed_count: summary.completed_count,
        filled_count: summary.filled_count,
        empty_count: summary.empty_count,
        pending_count: summary.pending_count,
        requested_slots: summary.requested_slots,
        filled_slots: summary.filled_slots,
        empty_slots: summary.empty_slots,
        pending_slots: summary.pending_slots,
        completion_reason: completionReason,

        /**
         * Tempo entre a primeira solicitação e o fechamento do relatório.
         */
        total_page_ads_time_ms:
          pageAdsState.firstRequestAt
            ? Date.now() -
              pageAdsState.firstRequestAt
            : null,

        /**
         * Tempo desde a última solicitação ou renderização.
         */
        time_since_last_activity_ms:
          pageAdsState.lastActivityAt
            ? Date.now() -
              pageAdsState.lastActivityAt
            : null,
        checked_at:
          Date.now()
      }
    );
  }

  /**
   * Aguarda um período sem novas solicitações ou renderizações.
   *
   * Isso evita concluir o relatório antes de o interstitial ser solicitado
   * pelo bloco que espera 3 segundos.
   */
  function schedulePageAdsQuietCheck() {
    if (pageAdsState.reportSent) {
      return;
    }

    clearPageAdsQuietTimer();

    pageAdsState.quietTimerId =
      setTimeout(function () {
        var summary =
          getPageAdsSummary();

        var allAdsRendered =
          summary.requested_count > 0 &&
          summary.requested_count ===
            summary.completed_count &&
          summary.pending_count === 0;

        if (allAdsRendered) {
          sendPageAdsRenderResult(
            'quiet_period_completed'
          );
        }
      }, CONFIG.PAGE_ADS_QUIET_PERIOD_MS);
  }

  /**
   * Inicia o tempo máximo de espera.
   *
   * Esse timer começa somente quando o primeiro anúncio é solicitado.
   */
  function startPageAdsMaxTimeout() {
    if (
      pageAdsState.maxTimeoutId ||
      pageAdsState.reportSent
    ) {
      return;
    }

    pageAdsState.maxTimeoutId =
      setTimeout(function () {
        sendPageAdsRenderResult(
          'max_timeout'
        );
      }, CONFIG.PAGE_ADS_MAX_TIMEOUT_MS);
  }

  // =========================================================================
  // GPT: SLOT REQUESTED
  // =========================================================================

  /**
   * Executado sempre que qualquer slot da página é solicitado.
   *
   * [AU-192] Só alimenta o estado do pageAdsRenderCompleted. O evento
   * individual `interstitialSlotRequested` FOI REMOVIDO — era o mesmo
   * `slotRequested` que o bloco 12 já emite (filtre por ad_unit_path).
   */
  function handleSlotRequested(event) {
    if (!event || !event.slot) {
      return;
    }

    var slotKey =
      getSlotKey(event.slot);

    if (!slotKey) {
      return;
    }

    /**
     * [UNIFICADO — GUARDA DE DOMÍNIO] Qualquer slot de interstitial visto
     * aqui (via isInterstitialSlot) confirma que o domínio tem interstitial.
     */
    if (isInterstitialSlot(event.slot)) {
      sawInterstitialSlot = true;
    }

    /**
     * [FIX 4] Novo request do mesmo slot: invalida a conclusão anterior
     * ANTES de sobrescrever o estado solicitado. Sem isso, o
     * pageAdsRenderCompleted trataria o request novo como concluído pelo
     * render do ciclo anterior.
     */
    delete pageAdsState.completedSlots[slotKey];

    /**
     * [FIX 4] Geração/request token por slot: cada request incrementa o
     * contador do slot. A conclusão registrada no slotRenderEnded carrega a
     * geração vigente — entradas órfãs de gerações antigas nunca contam
     * como concluídas para o request atual.
     */
    var generation =
      (pageAdsState.requestGenerations[slotKey] || 0) + 1;

    pageAdsState.requestGenerations[slotKey] = generation;

    pageAdsState.requestedSlots[slotKey] = {
      slot_id: getSlotId(event.slot),
      ad_unit_path: getAdUnitPath(event.slot),
      requested_at: Date.now(),
      generation:
        generation
    };

    if (!pageAdsState.firstRequestAt) {
      pageAdsState.firstRequestAt =
        pageAdsState.requestedSlots[slotKey].requested_at;
    }

    markPageAdsActivity();
    startPageAdsMaxTimeout();
    schedulePageAdsQuietCheck();

    /**
     * Começa antes da resposta do GAM para capturar a transição 0 -> altura.
     * Qualquer slot com elemento no DOM pode participar da detecção.
     */
    startAdExpansionObservation(event.slot);
  }

  // =========================================================================
  // GPT: SLOT RENDER ENDED
  // =========================================================================

  /**
   * Executado quando o GAM termina o processo de renderização de um slot.
   *
   * Esse evento também acontece quando o slot ficou vazio.
   *
   * [AU-192] Os emissores `interstitialRendered` e `interstitialEmpty` FORAM
   * REMOVIDOS — são o mesmo `slotRenderEnded` / `slotEmpty` do bloco 12.
   * O handler continua indispensável por duas razões:
   *
   *   1. marca `filledSlots`, que é a trava anti-fantasma do
   *      interstitialPresented (só apresenta o que preencheu);
   *   2. registra `completedSlots` para o pageAdsRenderCompleted;
   *   3. dispara a correlação clique -> render (interstitialRenderedAfterClick),
   *      que o bloco 12 não faz.
   */
  function handleSlotRenderEnded(event) {
    if (!event || !event.slot) {
      return;
    }

    var slotKey =
      getSlotKey(event.slot);

    var slotId =
      getSlotId(event.slot);

    var adUnitPath =
      getAdUnitPath(event.slot);

    var renderedAt =
      Date.now();

    /**
     * Marca qualquer anúncio da página como concluído.
     */
    if (slotKey) {
      // [FIX 4] Registra a geração/request vigente do slot: a conclusão só
      // conta para o request atual quando a geração bate (ver summary).
      var requestedForRender =
        pageAdsState.requestedSlots[slotKey] || {};

      pageAdsState.completedSlots[slotKey] = {
        slot_id: slotId,
        ad_unit_path: adUnitPath,
        is_empty: !!event.isEmpty,
        rendered_at: renderedAt,
        generation:
          requestedForRender.generation || null
      };

      markPageAdsActivity();
      schedulePageAdsQuietCheck();
    }

    /**
     * Confirma fill/empty para o evento global AdsImpression.
     */
    handleAdExpansionRenderEnded(
      event,
      renderedAt
    );

    /**
     * Daqui para baixo, processa somente o interstitial.
     */
    if (!isInterstitialSlot(event.slot)) {
      return;
    }

    // [FIX 2] Domínios sem interstitial não emitem nenhum evento específico
    // do interstitial, mesmo que outro bloco tenha criado esse slot no GPT.
    if (!CONFIG.INTERSTITIAL_ENABLED) {
      return;
    }

    /**
     * O GAM terminou, mas não retornou anúncio.
     *
     * Sem fill não há o que apresentar: não marca filledSlots e não há
     * correlação de clique a registrar. (O `slotEmpty` do bloco 12 já
     * registra o acontecimento.)
     */
    if (event.isEmpty) {
      return;
    }

    /**
     * Marca que este slot de interstitial preencheu neste ciclo.
     * Habilita o interstitialPresented (ver handleSlotVisibilityChanged).
     */
    if (slotKey) {
      interstitialState.filledSlots[slotKey] = true;
    }

    var recentClick =
      getRecentClick();

    /**
     * Relaciona a renderização ao último clique elegível.
     *
     * É uma correlação temporal. Não representa uma confirmação absoluta
     * de que o clique foi o gatilho interno usado pelo Google.
     */
    if (
      recentClick &&
      !interstitialState
        .renderedClickIds[
          recentClick.click_id
        ]
    ) {
      interstitialState
        .renderedClickIds[
          recentClick.click_id
        ] = true;

      track(
        'interstitialRenderedAfterClick',
        {
          click_id: recentClick.click_id,
          click_href: recentClick.click_href,
          click_text: recentClick.click_text,
          slot_id: slotId,
          ad_unit_path: adUnitPath,
          line_item_id: event.lineItemId || null,
          creative_id: event.creativeId || null,
          advertiser_id: event.advertiserId || null,
          campaign_id: event.campaignId || null,
          response_identifier: event.responseIdentifier || null,
          size_rendered: event.size ? event.size[0] + 'x' + event.size[1] : null,
          time_from_click_ms: recentClick.time_from_click_ms,
          rendered_at:
            renderedAt
        }
      );
    }
  }

  // =========================================================================
  // GPT: SLOT VISIBILITY CHANGED  (APRESENTAÇÃO DO INTERSTITIAL)
  // =========================================================================

  /**
   * Executado quando a porcentagem visível de QUALQUER slot muda.
   *
   * Para o interstitial, este é o ÚNICO sinal client-side que marca o momento
   * da APRESENTAÇÃO (display:none -> block, tela cheia). Enquanto o interstitial
   * fica escondido, a visibilidade é 0; a única subida para ~100 acontece quando
   * o usuário clica num link elegível e o Google o apresenta.
   *
   * REGRA (provada no navegador em 21/07/2026):
   * - Conta como apresentação a 1a subida para inViewPercentage >= 50.
   * - Deduplicado por slot: uma apresentação por ciclo.
   * - A descida posterior (100 -> 0, ao fechar) é ignorada.
   *
   * interstitialPresented equivale à métrica "Impressions" do GAM para o
   * interstitial — NÃO o slotRenderEnded (que é o preenchimento escondido
   * no load, e era a causa do gap 197 renders x 45 impressions no GAM).
   *
   * ⚠️ Este é o ÚNICO listener de `slotVisibilityChanged` em produção. O bloco
   *    12 tem o handler escrito mas não o registra. Por isso este emissor NÃO
   *    é duplicidade — nenhum outro bloco produz este acontecimento.
   */
  function handleSlotVisibilityChanged(event) {
    if (
      !event ||
      !CONFIG.INTERSTITIAL_ENABLED ||
      !isInterstitialSlot(event.slot)
    ) {
      return;
    }

    var inViewPercentage =
      typeof event.inViewPercentage === 'number'
        ? event.inViewPercentage
        : null;

    /**
     * Só a subida para >= 50% conta como apresentação.
     * Valores baixos (interstitial ainda escondido, ou descida ao fechar)
     * são ignorados.
     */
    if (
      inViewPercentage === null ||
      inViewPercentage < 50
    ) {
      return;
    }

    var slotKey =
      getSlotKey(event.slot) ||
      CONFIG.INTERSTITIAL_AD_UNIT_PATH;

    /**
     * Trava de segurança: só conta apresentação se o interstitial de fato
     * preencheu neste ciclo (slotRenderEnded isEmpty=false já ocorreu).
     * Sem fill não há criativo para apresentar — visibilidade alta aqui seria
     * um falso positivo, e contaria impressão que o GAM não conta.
     */
    if (
      !interstitialState.filledSlots[slotKey]
    ) {
      return;
    }

    /**
     * Dedup: uma apresentação por slot por ciclo.
     */
    if (
      interstitialState.presentedSlots[slotKey]
    ) {
      return;
    }

    interstitialState.presentedSlots[slotKey] = true;

    var responseInfo =
      getResponseInformation(event.slot);

    var recentClick =
      getRecentClick();

    /**
     * Confirmação cruzada síncrona: no instante da apresentação o Google já
     * adicionou #google_vignette à URL (via history.replaceState, sem disparar
     * o evento DOM hashchange). Ler location aqui é seguro e confirma o sinal.
     */
    var currentHash = null;
    var isVignetteHash = false;

    try {
      currentHash = window.location.hash || null;
      isVignetteHash = /#google_vignette/.test(window.location.href);
    } catch (error) {
      currentHash = null;
    }

    track(
      'interstitialPresented',
      {
        slot_id: getSlotId(event.slot),
        ad_unit_path: getAdUnitPath(event.slot),
        in_view_percentage: inViewPercentage,
        url_hash: currentHash,
        is_vignette_hash: isVignetteHash,
        line_item_id: getResponseValue(responseInfo, 'lineItemId'),
        creative_id: getResponseValue(responseInfo, 'creativeId'),
        advertiser_id: getResponseValue(responseInfo, 'advertiserId'),
        campaign_id: getResponseValue(responseInfo, 'campaignId'),
        source_agnostic_creative_id: getResponseValue(responseInfo, 'sourceAgnosticCreativeId'),
        source_agnostic_line_item_id: getResponseValue(responseInfo, 'sourceAgnosticLineItemId'),
        click_id: recentClick ? recentClick.click_id : null,
        presented_after_click: !!recentClick,
        time_from_click_ms: recentClick ? recentClick.time_from_click_ms : null,
        presented_at:
          Date.now()
      }
    );
  }

  // =========================================================================
  // INICIALIZAÇÃO
  // =========================================================================

  /**
   * googletag.cmd registra os listeners quando o GPT estiver pronto.
   *
   * [AU-192] Os listeners de `slotOnload` e `impressionViewable` foram
   * REMOVIDOS: depois de tirar interstitialLoaded e
   * interstitialImpressionViewable eles não emitiam mais nada. Esses dois
   * acontecimentos vêm do bloco 12, com os nomes oficiais.
   */
  window.googletag =
    window.googletag || { cmd: [] };

  window.googletag.cmd.push(
    function registerAdsListeners() {
      try {
        var pubads =
          window.googletag.pubads();

        /**
         * Alimenta o estado do pageAdsRenderCompleted.
         */
        pubads.addEventListener(
          'slotRequested',
          handleSlotRequested
        );

        /**
         * Trava do filledSlots + estado da página + correlação de clique.
         */
        pubads.addEventListener(
          'slotRenderEnded',
          handleSlotRenderEnded
        );

        /**
         * APRESENTAÇÃO do interstitial (impression do GAM).
         * Único listener de slotVisibilityChanged em produção.
         */
        pubads.addEventListener(
          'slotVisibilityChanged',
          handleSlotVisibilityChanged
        );

        log(
          'Tracker v2.2.0 inicializado com sucesso.',
          {
            interstitial_ad_unit_path: CONFIG.INTERSTITIAL_AD_UNIT_PATH,
            eventos_emitidos: [
              'interstitialPresented',
              'interstitialEligibleLinkClick',
              'interstitialRenderedAfterClick',
              'pageAdsRenderCompleted',
              'AdsImpression'
            ],
            click_render_window_ms: CONFIG.CLICK_RENDER_WINDOW_MS,
            page_ads_quiet_period_ms: CONFIG.PAGE_ADS_QUIET_PERIOD_MS,
            page_ads_max_timeout_ms:
              CONFIG.PAGE_ADS_MAX_TIMEOUT_MS
          }
        );
      } catch (error) {
        log(
          'Erro ao registrar listeners do GPT:',
          error
        );
      }
    }
  );
})();
