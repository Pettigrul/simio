<script>
document.getElementById('btnSimulate').addEventListener('click', function() {
    // Get values using the helper to parse "10.000,00" -> 10000.00
    const price = parseMoney(document.getElementById('sim_price').value) || 0;
    const downPayment = parseMoney(document.getElementById('sim_downpayment').value) || 0;
    const months = parseInt(document.getElementById('sim_months').value) || 48;
    
    // Validate
    if(price <= 0) {
        alert('Por favor, informe o valor do veículo.');
        return;
    }

    if(downPayment >= price) {
        alert('A entrada deve ser menor que o valor do veículo.');
        return;
    }

    // Calculation Logic (PMT Formula)
    // Rate: 2.5% monthly (0.025) - taxa mais realista
    const rate = 0.025;
    const principal = price - downPayment;
    
    // PMT = P * (r * (1 + r)^n) / ((1 + r)^n - 1)
    const pmt = principal * (rate * Math.pow((1 + rate), months)) / (Math.pow((1 + rate), months) - 1);
    
    // Format Result Brazilian Standard
    const formatter = new Intl.NumberFormat('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    
    // Update UI
    document.getElementById('sim_empty_state').classList.add('hidden');
    document.getElementById('sim_result_state').classList.remove('hidden');
    
    document.getElementById('res_value').innerText = formatter.format(pmt);
    document.getElementById('res_months').innerText = months + 'x';
});
</script>
