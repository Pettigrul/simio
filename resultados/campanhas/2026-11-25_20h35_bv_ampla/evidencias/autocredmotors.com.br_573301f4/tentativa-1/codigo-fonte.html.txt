(function(){
  var estoque = (window.ESTOQUE && window.ESTOQUE.veiculos) ? window.ESTOQUE.veiculos : [];

  var simParcela = document.querySelector('#simParcela');
  var simFinanciado = document.querySelector('#simFinanciado');

  var params = new URLSearchParams(window.location.search);

  if (simVeiculo) {

      simVeiculo.appendChild(opt);

      if (achado) {
        simVeiculo.value = achado.id;
        if (simValor) simValor.value = Number(achado.preco).toLocaleString('pt-BR');
      }
    }
  }

  function parseMilhar(str){
    if (!str) return 0;
    var limpo = String(str).replace(/[^\d]/g, '');
    return limpo ? parseInt(limpo, 10) : 0;
  }
  function fmtMilhar(num){
    if (!num && num !== 0) return '';
    return Number(num).toLocaleString('pt-BR');
  }
  function fmtPreco(v){
    return 'R$ ' + Number(Math.round(v)).toLocaleString('pt-BR');
  }

  function aplicarMascara(input){
    if (!input) return;
    input.addEventListener('input', function(){
      var num = parseMilhar(input.value);
      input.value = num ? fmtMilhar(num) : '';
      calcular();
    });
  }
  aplicarMascara(simValor);
  aplicarMascara(simEntrada);

  if (simVeiculo) {

      var preco = opt ? Number(opt.getAttribute('data-preco')) : 0;
      if (preco && simValor) {

    var valorVeiculo = parseMilhar(simValor ? simValor.value : 0);

})();
