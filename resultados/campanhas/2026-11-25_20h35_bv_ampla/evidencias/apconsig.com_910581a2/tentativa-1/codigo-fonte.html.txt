            showGlobalSpinner("Acessando " + name + "...", "Redirecionando com segurança...");
            setTimeout(() => {
                hideGlobalSpinner();
                window.open(url, '_blank', 'noopener,noreferrer');
            }, 400);
        }

        // CONTROLE DO DROPDOWN "LINKS"
        function toggleLinksDropdown(event) {
            event.stopPropagation();
            const navItem = document.getElementById('navItemLinks');
            navItem.classList.toggle('open-click');
        }

        // Fecha ao clicar fora
        document.addEventListener('click', (e) => {
            const navItem = document.getElementById('navItemLinks');
            if (navItem && !navItem.contains(e.target)) {
                navItem.classList.remove('open-click');
            }
        });

        // CONTROLE DOS ITENS DO ACORDEÃO (CLIQUE PARA EXPANDIR / RECOLHER)
        function toggleAccItem(event, el) {
            event.stopPropagation();
            const item = el.closest('.acc-item');
            if (!item) return;

            // Se for do mesmo nível, podemos opcionalmente fechar os irmãos
            const parent = item.parentElement;
            const siblings = parent.querySelectorAll(':scope > .acc-item');
            siblings.forEach(sib => {
                if (sib !== item) sib.classList.remove('is-open');
            });

            item.classList.toggle('is-open');
        }

        // FILTRO DE BUSCA NO DROPDOWN
        function filterMenuAccordion(term) {
            const filter = term.toLowerCase().trim();
            const items = document.querySelectorAll('.links-dropdown-panel .acc-item, .links-dropdown-panel .acc-leaf');
            
            if (!filter) {
                // Reseta estado
                items.forEach(el => {
                    el.style.display = '';
                    if (el.classList.contains('acc-item')) {
                        el.classList.remove('is-open');
                    }
                });
                return;
            }

            // Se houver busca, abre tudo que tiver match e mostra
            document.querySelectorAll('.links-dropdown-panel .acc-leaf').forEach(leaf => {
                const text = leaf.textContent.toLowerCase();
                if (text.includes(filter)) {
                    leaf.style.display = '';
                    // Abre todos os pais
                    let p = leaf.closest('.acc-item');
                    while (p) {
                        p.classList.add('is-open');
                        p.style.display = '';
                        p = p.parentElement.closest('.acc-item');
                    }
                } else {
                    leaf.style.display = 'none';
                }
            });
        }

        // SIMULADOR
        function calcularSimulacao() {
            const valor = parseFloat(document.getElementById('val-credito').value);
            const meses = parseInt(document.getElementById('val-prazo').value);
            const convenio = document.getElementById('sim-convenio').value;

            document.getElementById('val-credito-display').textContent = valor.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL', maximumFractionDigits: 0 });
            document.getElementById('val-prazo-display').textContent = meses + 'x';

            let taxa = 0.0185;
            if (convenio === 'inss') taxa = 0.0168;
            else if (convenio === 'publico') taxa = 0.0175;
            else if (convenio === 'fgts') taxa = 0.0149;

            const parcela = (valor * (taxa * Math.pow(1 + taxa, meses))) / (Math.pow(1 + taxa, meses) - 1);
            document.getElementById('val-parcela').textContent = parcela.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' });
        }
