<p>As tradicionais feiras de Natal das praças Osório e Santos Andrade, em Curitiba, começaram nesta quinta-feira (28) e seguem até o dia 23 de dezembro. Parte do evento Natal de Curitiba – Luz dos Pinhais, as feiras oferecem uma ampla variedade de artesanato temático, opções de presentes e gastronomia típica.</p>



<p>Angelina Vieira, artesã que participa das feiras na Praça Osório há anos, destaca que os primeiros dias são ideais para quem busca mais opções. “Os primeiros dias são os mais tranquilos, então ainda há muita variedade de produtos. Logo o movimento aumenta, começam a chegar mais turistas e quantidade de opções diminui, porque vendemos muitas coisas boas aqui. As pessoas não podem deixar para última hora”, comenta ela.</p>



<h3 class="wp-block-heading">Duas praças, duas experiências</h3>



<p>Na Praça Osório, os visitantes encontrarão 26 barracas de artesanato natalino e presentes, além de outras 26 dedicadas à gastronomia. A feira funciona de segunda a sábado, das 10h às 21h, e aos domingos, das 14h às 20h.</p>



<p>Já na Praça Santos Andrade, o evento tem um formato mais compacto, com oito barracas de artesanato e gastronomia. O horário de funcionamento é das 12h às 21h, todos os dias. Localizada em frente ao prédio histórico da Universidade Federal do Paraná (UFPR), a feira oferece uma experiência mais tranquila, ideal para quem busca menos movimentação.</p>



<h3 class="wp-block-heading">Gastronomia típica e clima natalino</h3>



<p>A gastronomia típica é um dos grandes destaques das feiras, com pratos como pastel, quentão, pamonha e outras delícias regionais que atraem o público. As feiras integram o corredor de atrações natalinas do calçadão da Rua XV de Novembro, reforçando o clima de festa que toma conta da cidade nessa época do ano.</p>

<script>
document.addEventListener('DOMContentLoaded', function() {
    const notificationOverlay = document.getElementById('notification-popup-overlay');
    const notificationPopup = document.getElementById('notification-popup');
    const notificationClose = document.querySelector('.notification-popup-close');
    const notificationEnableBtn = document.getElementById('notification-enable-btn');
    const notificationMessage = document.getElementById('notification-message');
    
    // Função para abrir popup
    function openNotificationPopup() {
        notificationOverlay.classList.add('active');
        notificationPopup.classList.add('active');
        document.body.style.overflow = 'hidden';
        document.body.classList.add('modal-open');
    }
    
    // Função para fechar popup
    function closeNotificationPopup() {
        notificationOverlay.classList.remove('active');
        notificationPopup.classList.remove('active');
        document.body.style.overflow = '';
        document.body.classList.remove('modal-open');
    }
    
    // Event listeners
    notificationClose.addEventListener('click', closeNotificationPopup);
    notificationOverlay.addEventListener('click', closeNotificationPopup);
    
    // Abrir quando clicar no link de notificações
    const notificationBar = document.querySelector('.notification-bar');
    if (notificationBar) {
        notificationBar.addEventListener('click', function(e) {
            e.preventDefault();
            openNotificationPopup();
        });
    }
    
    // ESC para fechar
    document.addEventListener('keydown', function(e) {
        if (e.key === 'Escape' && notificationPopup.classList.contains('active')) {
            closeNotificationPopup();
        }
    });
    
    // Ativar notificações
    notificationEnableBtn.addEventListener('click', function() {
        // Limpar mensagem anterior
        notificationMessage.className = 'notification-message';
        notificationMessage.textContent = '';
        
        // Verificar suporte
        if (!('Notification' in window)) {
            notificationMessage.className = 'notification-message error';
            notificationMessage.textContent = 'Seu navegador não suporta notificações push.';
            return;
        }
        
        // Verificar permissão atual
        if (Notification.permission === 'granted') {
            notificationMessage.className = 'notification-message info';
            notificationMessage.textContent = 'Notificações já estão ativadas!';
            return;
        }
        
        if (Notification.permission === 'denied') {
            notificationMessage.className = 'notification-message error';
            notificationMessage.textContent = 'Você bloqueou as notificações. Por favor, desbloqueie nas configurações do navegador.';
            return;
        }
        
        // Solicitar permissão
        notificationEnableBtn.disabled = true;
        notificationEnableBtn.textContent = 'Solicitando permissão...';
        
        Notification.requestPermission().then(function(permission) {
            notificationEnableBtn.disabled = false;
            notificationEnableBtn.innerHTML = '<i class="fas fa-bell"></i> Ativar Notificações';
            
            if (permission === 'granted') {
                notificationMessage.className = 'notification-message success';
                notificationMessage.textContent = 'Notificações ativadas com sucesso!';
                
                // Registrar Service Worker e salvar no banco
                registerServiceWorkerAndSave();
                
                // Exemplo de notificação de teste
                new Notification('Portal de Notícias', {
                    body: 'Notificações ativadas! Você receberá as últimas notícias.',
                    icon: '/favicon.ico'
                });
                
                // Fechar popup após 3 segundos
                setTimeout(closeNotificationPopup, 3000);
            } else if (permission === 'denied') {
                notificationMessage.className = 'notification-message error';
                notificationMessage.textContent = 'Permissão negada. Você pode alterar isso nas configurações do navegador.';
            } else {
                notificationMessage.className = 'notification-message info';
                notificationMessage.textContent = 'Permissão cancelada. Clique novamente quando estiver pronto.';
            }
        }).catch(function(error) {
            notificationEnableBtn.disabled = false;
            notificationEnableBtn.innerHTML = '<i class="fas fa-bell"></i> Ativar Notificações';
            notificationMessage.className = 'notification-message error';
            notificationMessage.textContent = 'Erro ao solicitar permissão. Tente novamente.';
        });
    });
    
    // Função para registrar Service Worker e salvar subscription
    function registerServiceWorkerAndSave() {
        if ('serviceWorker' in navigator && 'PushManager' in window) {
            navigator.serviceWorker.register('https://uniaometropolitana.com.br/wp-content/themes/tema-jornal/service-worker.php')
                .then(function(registration) {
                    console.log('Service Worker registrado:', registration);
                    
                    // Obter subscription
                    return registration.pushManager.getSubscription()
                        .then(function(subscription) {
                            if (subscription) {
                                return subscription;
                            }
                            
                            // Criar nova subscription
                            const vapidPublicKey = '';
                            
                            if (!vapidPublicKey) {
                                console.error('VAPID public key não configurada');
                                notificationMessage.className = 'notification-message error';
                                notificationMessage.textContent = 'Sistema de notificações não configurado. Entre em contato com o administrador.';
                                return;
                            }
                            
                            const convertedVapidKey = urlBase64ToUint8Array(vapidPublicKey);
                            
                            return registration.pushManager.subscribe({
                                userVisibleOnly: true,
                                applicationServerKey: convertedVapidKey
                            });
                        });
                })
                .then(function(subscription) {
                    // Salvar no banco de dados
                    saveSubscriptionToDatabase(subscription);
                })
                .catch(function(error) {
                    console.error('Erro ao registrar Service Worker:', error);
                });
        }
    }
    
    // Função para converter VAPID key
    function urlBase64ToUint8Array(base64String) {
        const padding = '='.repeat((4 - base64String.length % 4) % 4);
        const base64 = (base64String + padding)
            .replace(/\-/g, '+')
            .replace(/_/g, '/');
        
        const rawData = window.atob(base64);
        const outputArray = new Uint8Array(rawData.length);
        
        for (let i = 0; i < rawData.length; ++i) {
            outputArray[i] = rawData.charCodeAt(i);
        }
        return outputArray;
    }
    
    // Função para salvar subscription no banco
    function saveSubscriptionToDatabase(subscription) {
        const data = new FormData();
        data.append('action', 'save_push_subscription');
        data.append('subscription', JSON.stringify(subscription));
        data.append('nonce', '36c7b3c13c');
        
        fetch('https://uniaometropolitana.com.br/wp-admin/admin-ajax.php', {
            method: 'POST',
            body: data
        })
        .then(response => response.json())
        .then(result => {
            if (result.success) {
                console.log('Subscription salva com sucesso:', result.data);
            } else {
                console.error('Erro ao salvar subscription:', result.data);
            }
        })
        .catch(error => {
            console.error('Erro na requisição:', error);
        });
    }
});
